\chapter{Неассоциативные обертывающие алгебры нильтреугольных алгебр Шевалле}\label{ch:enveloping}

\section{Точные обертывающие алгебры и стандартные идеалы}\label{sec:enveloping}
%
В теории Картана"--~Киллинга каждой простой комплексной конечномерной алгебре
Ли $L$ сопоставляют единственную (с точностью до эквивалентности) неразложимую
систему корней $\Phi$ евклидова пространства $V$, построенного на подалгебре
Картана \cite[глава 3]{carter72}. Выбор простых корней или базы $\Pi$ в $\Phi$
определяет систему положительных корней $\Phi^+\supseteq \Pi$.

Элементы $e_r$ $(r \in \Phi)$ и подходящая база подалгебры Картана алгебры Ли
$L=\mathfrak{L}(\Phi, \mathbb{C})$ задают \cite{Chev-55} базу Шевалле с
целочисленными структурными константами и тем самым определяют алгебру Ли
$\mathfrak{L}(\Phi, \mathbb{K})$ над любым полем $\mathbb{K}$ (алгебра
Шевалле). Ее подалгебру $N\Phi(\mathbb{K})$ с базой $\{e_r\ | \ r \in \Phi^+
    \}$ называем \emph{нильтреугольной}.

\medskip

Согласно теореме Пуанкаре"--~Биркгофа"--~Витта каждая алгебра Ли может быть
получена при помощи выбора подходящей ассоциативной обертывающей алгебры,
замены в ней операции на коммутирование и дальнейшего перехода к подходящей
подалгебре Ли, см. \cite[\S\ 17]{humphreys-ru}, \cite[\S~V.9]{Kurosh1973}.
Более трудная теорема Адо"--~Ивасавы (принадлежащая Адо для характеристики $0$
и Ивасаве для характеристики $p$, \cite[глава VI]{jacobson-ru}) устанавливает,
что конечномерные алгебры Ли могут быть получены таким способом из
конечномерных ассоциативных алгебр.

В случае, когда алгебра Ли $\mathfrak{L}(\Phi, \mathbb{K})$ является простой,
присоединенное представление $\ad \colon \mathfrak{L}(\Phi, \mathbb{K})
    \hookrightarrow \mathfrak{gl}(\mathfrak{L}(\Phi, \mathbb{K}))$ является точным
и ассоциативная алгебра, порожденная образом $N\Phi(\mathbb{K})$, дает ее
конечномерную обертывающую алгебру.

\begin{definition}[\cite{DAN2018}]
    Алгебра $A$ называется \emph{точной обертывающей} алгебры Ли $L$, если $L$ и
    $A^{(-)}$ изоморфны.
\end{definition}
%
Алгебру Ли $L$ и ее точную обертывающую алгебру $A$ можно определять
структурными константами в одной базе, в отличие от универсальной ассоциативной
обертывающей алгебры.

В \cite{DAN2018} были построены семейства точных обертывающих алгебр для
нильтреугольных алгебр $N\Phi(\mathbb{K})$. Нам потребуются предварительные
сведения.

\medskip

По теореме Шевалле о базе \cite[теорема 4.2.1]{carter72} при $r,s\in \Phi$
имеем
\begin{equation*}%\label{eq:chevalley-basis}
    [e_r ,e_s] = N_{rs}e_{r+s}=-[e_s ,e_r] \ \ (r+s\in \Phi),
    \qquad [e_r,e_s]=0 \  \  (r+s\notin \Phi \cup \{0\}),
\end{equation*}
%
где либо $N_{rs}=\pm 1$, либо $|r|=|s|<|r+s|$ и $N_{rs}=\pm 2$, либо
$N_{rs}=\pm 2$ или $\pm 3$ и $\Phi$ типа $G_2$. Согласно \cite[предложение
    4.2.2]{carter72}, знаки структурных констант $N_{rs}$ имеют определенный
произвол для алгебры Шевалле, а также для $N\Phi(\mathbb{K})$.

\medskip

Фиксируя выбор знаков структурных констант $N_{rs}$, через $R_{\Phi}$ обозначим
$\mathbb{K}$-алгебру с базой $\{e_r\ | \ r \in \Phi^+\}$ и умножением:\
$e_re_s=0$ при $r+s \not \in \Phi$ и
\[
    e_r e_s= e_{r+s},\qquad e_se_r=(1-N_{rs}) e_{r+s}\qquad (r,s,r+s\in \Phi^+,\ N_{rs}\geq 1).
\]
Заметим, что структурные константы $N_{rs}$ в базисе Шевалле являются целыми
числами. При работе над полем $\mathbb{K}$ мы отождествляем $N_{rs}$ с его
образом $N_{rs}\cdot 1_{\mathbb{K}}\in \mathbb{K}$. Поэтому условия и сравнения
вида $N_{rs}\geq 1$ следует понимать как утверждения о соответствующем целом
числе $N_{rs}$ (до редукции по характеристике), а коэффициенты в формулах
рассматриваются в поле $\mathbb{K}$.

\medskip

При фиксированном $\Phi$ точные обертывающие алгебры $R_{\Phi}$ и оценку их
числа выявляет

\begin{proposition}\label{p:mult-enveloping}
    %
    Для алгебры Ли $N\Phi(\mathbb{K})$ каждая алгебра $R_{\Phi}$ является точной
    обертывающей алгеброй и их число не превосходит $2^{|\Phi^+ \setminus \Pi |}$.
\end{proposition}

\begin{proof}
    %
    Таблицы умножения базисных элементов $N\Phi(\mathbb{K})$ и любой алгебры
    $R_{\Phi}^{(-)}$, очевидно, совпадают. Поэтому, как и в {\cite[предложение
        1]{DAN2018}}, алгебра $R_{\Phi}$ есть точная обертывающая алгебры Ли
    $N\Phi(\mathbb{K})$.

    Зафиксируем произвольный линейный порядок на $\Pi$ и продолжим его до линейного
    порядка $\prec$ на $\Phi^+$. Когда $r,s,r+s \in \Phi^+$ и $0\prec r \prec s$,
    пару корней $r, s$ называют \emph{специальной}, а также
    \emph{экстраспециальной}, если $r \preccurlyeq r_1$ для каждой специальной пары
    $r_1, s_1$ с суммой $r_1+s_1 =r+s$. Согласно ~\cite[предложение
        4.2.2]{carter72}, знаки констант $N_{rs}$ можно выбрать для экстраспециальных
    пар $(r,s)$ произвольно, с точностью до изоморфизмов алгебры Ли $\mathcal{L}
        (\Phi, \mathbb{K})$ (аналогично, $N\Phi(\mathbb{K})$); тогда знаки остальных
    констант $N_{rs}$ определяются однозначно.

    Любую из построенных точных обертывающих алгебр $R_{\Phi}$ алгебры Ли
    $N\Phi(\mathbb{K})$ однозначно определяет выбор знака $\pm$ (одна из двух
    возможностей) константы $N_{rs}$ для каждой экстраспециальной пары корней
    $(r,s)$ в $\Phi^+$.

    С другой стороны, любой положительный непростой корень в $\Phi$ есть сумма
    $r+s$ единственной экстраспециальной пары корней $r,s$. Поэтому множество
    экстраспециальных пар взаимнооднозначно множеству $\Phi^+ \setminus \Pi$.

    Отсюда вытекает требуемая в предложении оценка числа алгебр $R_{\Phi}$.
\end{proof}

\begin{lemma}\label{l:Const-RPhi-Non-assoc}
    %
    Все ненулевые структурные константы алгебр $R_{\Phi}$ в выбранной базе равны \
    $\pm 1$ для типа $\neq G_2$ и равны $1$, когда все корни в $\Phi$ одной длины.
    Если $\Phi$ имеет корни разных длин, то все алгебры $R_{\Phi}$ неассоциативные.
\end{lemma}

\begin{proof}
    %
    Когда система корней $\Phi$ имеет корни разных длин, найдутся корни $r, s \in
        \Phi^+$, для которых пересечение $\Phi\cap (Zr+ Zs)=\Psi$ есть подсистема
    корней типа $B_2$ или $G_2$. Определение алгебры $R_{\Phi}$ позволяет считать,
    что $\{r, s \}$ есть база в $\Psi$ и $2r+s$ "--- корень. Если $N_{rs} \geq 1$,
    то получаем
    \[
        e_r(e_re_s) =e_re_{r+s} = \pm e_{2r+s},\quad (e_re_r)e_s=0.
    \]
    Аналогично при $N_{sr}\ (=- N_{rs})\ \geq 1$ произведения $(e_se_r)e_r =e_{r+s}
        e_r = \pm e_{2r+s}$ и $e_s(e_re_r)=0$ не совпадают. Следовательно, алгебра
    $R_{\Phi}$ "--- неассоциативная.

    \medskip

    Первое утверждение в лемме сейчас сразу следует из определения алгебр
    $R_{\Phi}$. Таким образом, доказательство завершено.
\end{proof}

\medskip

Для каждого типа $\Phi$ мы исследуем (\S~\ref{sec:uniqtheorems}) условия
однозначности алгебр $R_{\Phi}$ и, взаимосвязано, структурных констант $N_{rs}$
алгебр Ли $N\Phi(\mathbb{K})$ над полем $\mathbb{K}$.

\medskip

Идеал произвольного кольца $A$ всегда есть идеал и кольца $A^{(-)}$, так как
основные операции в $A^{(-)}$ производны от операций в $A$. Верно и обратное,
когда умножение в $A$ есть также производная операция от операций в $A^{(-)}$,
например, если для алгебры Ли $L$ с умножением $*$ выберем обертывающую алгебру
$A$ с умножением $\alpha \cdot \beta=(\alpha* \beta)t$, где $t$ "--- обратимый
скаляр.

\medskip

Как и во введении, используем отношения частичного порядка $r>s$ корней и
инцидентности ($r\geq s$ или $s\geq r$), понятие множества углов $\mathcal{L}=
    \mathcal{L}(H)$ подмножеств $H$ в $N\Phi(\mathbb{K})$, а также идеалы $T(r), \
    Q(r), \ T(\mathcal{L})$ и $Q(\mathcal{L})$.

\begin{definition}\label{def:stand-enval}
    %
    Идеал $H$ кольца Ли $N\Phi(\mathbb{K})$ называем \emph{стандартным}, если $Q
        (\mathcal{L}(H))\subset H$. Точную обертывающую алгебру $R$ алгебры Ли
    $N\Phi(\mathbb{K})$ над полем $\mathbb{K}$ называем \emph{стандартной}, если
    все ее идеалы стандартны.
\end{definition}

Из свойств систем корней и определений легко вытекает

\begin{lemma}\label{l:stand-one-corner}
    В кольце $R_{\Phi}$ каждый идеал с единственным углом стандартен.
    Когда $\Phi$ ранга $2$, любая алгебра $R_{\Phi}$ стандартна.
\end{lemma}

\begin{proof}
    Доказательство первого утверждения леммы для идеала $H$ в $R_{\Phi}$ c
    множеством углов $\mathcal{L}(H)=\{r\}$ проводим с помощью индукции по
    $\mathrm{h}(\Phi)-\height(r)$. Ясно, что $Q(\rho) = 0 \subset H \subset T(r)$. Пусть $r$ "---
    не максимальный корень. Тогда существует простой корень $p$ с условием $r+p \in
        \Phi^+$ и одно из произведений $e_pe_r$, $e_re_p$ равно $e_{r+p}$. Поэтому
    множество $(\mathbb{K}e_p)H + H(\mathbb{K}e_p)$ по модулю $Q(r+p)$ совпадает с $\mathbb{K}e_{r+p}$ и, по
    индукции, порождает идеал, содержащий $Q(r+p)$, откуда $T(r+p) \subset H$.
    Учитывая произвол в выборе $p$, получаем включение $Q(r) \subset H$.

    Второе утверждение леммы сразу следует из первого для случая систем корней
    $\Phi$ типа $A_2$, $B_2$ и $G_2$.
\end{proof}

Хорошо известно, что алгебра Ли $N\Phi(\mathbb{K})$ типа $A_{n-1}$ изоморфна
алгебре Ли, ассоциированной с алгеброй $NT(n,\mathbb{K})$ (нижних)
нильтреугольных $n\times n$ матриц над $\mathbb{K}$. Таким образом, для алгебры
Ли $N\Phi(\mathbb{K})$ типа $A_{n-1}$ существует ассоциативная точная
обёртывающая алгебра "--- алгебра $R=NT(n,\mathbb{K})$.

Здесь обычные матричные единицы $e_{ij} \ (1 \leq j < i \leq n)$ составляют
базу алгебры $R$, а также базу Шевалле из элементов $e_r= e_{ij}\ (r \in
    \Phi^+)$ алгебры Ли $R^{(-)}= N\Phi(\mathbb{K})$ после соответствующей
нумерации корней $r=r_{ij}$. В этом случае известна стандартность всех идеалов
алгебры $R$ над полем $\mathbb{K}$ (Р.~Дюбиш и С.~Перлис \cite[Теорема
    8]{Dub-Perl-1951} и даже идеалов кольца $R$, \cite[теорема 2]{vL76}).

Алгебры Ли $N\Phi(\mathbb{K})$ остальных классических типов заданы в
\cite[лемма 2]{vl90a_ru} аналогично в базе Шевалле из <<матричных единиц>> \
$e_{iv}$ с ограничениями
\[
    B_n: \ -i < v < i \leq n;\quad
    D_n: \ 1 \leq |v| < i \leq n;\quad
    C_n: \ -i \leq v < i \leq n, \ v \neq 0.
\]

К суммам двух корней, являющихся корнями, помимо сумм $r_{ij}+ r_{jv}=r_{iv}$
(аналогично типу $A_n$), здесь относятся еще суммы
\[
    r_{kv}+ r_{m,-v}=r_{k, -m}\quad (k>m>|v|),
\]
а для типа $C_n$ при $k =m>|v|\geq 1$ также суммы $r_{kv}+ r_{k,-v} =
    r_{k,-k}$.

Любой элемент из $N\Phi(\mathbb{K})$ представляется суммой $\sum a_{iv}e_{iv}$
и $\Phi^+$-матрицей $||a_{iv}||$ над $\mathbb{K}$ соответствующего типа. Так,
$B_n^+$-матрица имеет вид
%
\[a_{10}\]
\[a_{2,-1}\ a_{20}\ a_{21}\]
\[\dots\ \ \dots\ \ \dots\]
\[a_{n,-n+1}\ \dots\  a_{n,-1}\ a_{n0}\ a_{n1}\ \dots\ a_{n,n-1}.\]
%
Отбрасывая столбец с нулевым номером, получаем $D_n^+$-матрицу.

В \cite[лемма 2]{vl90a_ru} доказана

\begin{lemma}\label{l:StructConst-vl90a_ru}
    Знаки структурных констант $N_{rs}$ алгебры Ли  $N\Phi(\mathbb{K})$ можно
    выбрать так, что $[e_{ij},e_{jv}]=e_{iv}=-[e_{jv},e_{ij}]$ и
    выполняются равенства:
    \[
        [e_{jv},e_{i,-v}]=e_{i,-j}\quad (\Phi=B_n,D_n),\quad i>j>|v|>0;\]
    %
    \[[e_{i0},e_{j0}]=2e_{i,-j}\quad(\Phi=B_n),\quad
        [e_{ij},e_{i,-j}]=2e_{i,-i}\quad(\Phi=C_n),\quad i>j\geq1;\]
    %
    \[[e_{jm},e_{i,-m}]=[e_{im},e_{j,-m}]=e_{i,-j}\quad(\Phi=C_n),\quad
        i>j>m\geq1.\]
\end{lemma}

\medskip

Соответственно типу ${\Phi}$ для алгебр Ли $N\Phi(\mathbb{K})$ используем
обозначения $NB_n(\mathbb{K})$, $NC_n(\mathbb{K})$ и так далее. Очевидным
следствием леммы~\ref{l:StructConst-vl90a_ru} является

\begin{lemma}\label{l:ReductBnCnDn}
    Факторалгебра $ND_n(\mathbb{K})/T(r)$ изоморфна $NT(n,\mathbb{K})^{(-)}$ точно для
    двух корней $r= r_{2,-1}$ и $r=r_{21}$ при $n>4$, и точно для трех
    простых корней $r$ при $n=4$.
    Кроме того, $NB_n(\mathbb{K})/T(r) \simeq NT(n+ 1,\mathbb{K})^{(-)}$ и $NC_n(\mathbb{K})/T(r) \simeq NT(n+
        1,\mathbb{K})^{(-)}$ для единственного корня $r=r_{2,-1}$ и $r=r_{2,-2}$ соответственно.
\end{lemma}

\begin{remark}\label{remark:A3-D3}
    Для ${\Phi}$ типа $B_n$ и $C_n$ произвольная обертывающая алгебра
    $R_{\Phi}$ алгебры Ли $N\Phi(\mathbb{K})$ имеет $\mathbb{K}$-подмодуль $M_\Phi$ с
    базой $\{e_{iv}\ |\ 1 \leq |v| < i \leq n\}$. Он не является
    подалгеброй для ${\Phi}$ типа $C_n\ (n>1)$, а в алгебре $R_{\Phi}$
    типа $B_n$ есть подалгебра, изоморфная обертывающей алгебре алгебры Ли $ND_n(\mathbb{K})$.
\end{remark}

Точные обертывающие алгебры $R_{\Phi}$ классического типа с выбором знаков
структурных констант по лемме \ref{l:StructConst-vl90a_ru} обозначаем
соответственно типу
%
\begin{equation}\label{eq:RAn-RDn}
    RA_n(\mathbb{K})=NT(n+1, \mathbb{K}),\ \ RB_n(\mathbb{K}),\ \ RC_n(\mathbb{K}),\ \ RD_n(\mathbb{K}).
\end{equation}
%
\begin{lemma}\label{l:mult-RAn-RDn}
    %
    В алгебрах $R_{\Phi}$ из списка \eqref{eq:RAn-RDn} умножение определяют правила
    \[
        e_{ij}e_{jv}= e_{iv},\quad e_{iu}e_{jv}=0\quad (u\neq j, \ u\neq -v);
    \]
    %
    \[
        \Phi = B_n, D_n:\ \
        e_{jv}e_{i,-v}=e_{i,-j}\ \ (i>j>\mid v \mid > 0),\quad
        e_{iv}e_{j,-v}=0\ \ (i\geq j>\mid v \mid > 0);
    \]
    %
    \[
        \Phi=C_n:\ \ e_{ij}e_{i,-j}=-e_{i,-j}e_{ij}=e_{i,-i}\quad (i>j\geq 1),
    \]
    %
    \[
        e_{jm}e_{i,-m} = e_{im}e_{j,-m} = e_{i,-j},\quad
        e_{i,-m}e_{jm}=e_{j,-m}e_{im}=0\quad (i>j>m \geq 1);
    \]
    %
    \[
        \Phi = B_n:\qquad e_{i0}e_{j0} = e_{i,-j}=-e_{j0}e_{i0}\quad (i>j\geq 1).
    \]
\end{lemma}

\begin{lemma}\label{l:R-A3-D3}
    %
    Точная обертывающая алгебра $RD_3(\mathbb{K})$ алгебры Ли $N\Phi(\mathbb{K})$
    типа $A_3$ является неассоциативной и нестандартной.
\end{lemma}

\begin{proof}
    %
    Алгебры Ли $N\Phi(\mathbb{K})$ типа $A_3$ и $D_3$ изоморфны, поскольку их
    системы корней эквивалентны. Однако, алгебра $RD_3(\mathbb{K})$, в отличие от
    алгебры $RA_3(\mathbb{K})= NT(4,\mathbb{K})$, имеет нестандартный идеал
    \[
        \mathbb{K}(e_{2 1}+ e_{2,-1})+ \mathbb{K}(e_{3 1}+ e_{3,-1})+ \mathbb{K}e_{3,-2}
    \]
    и является неассоциативной, так как $e_{2,-1} (e_{32} e_{21}) = e_{3,-2}$ и
    $(e_{2,-1}e_{32}) e_{21}=0$.
\end{proof}

\begin{lemma}
    Идеал $H \subset L_4$ точной обертывающей алгебры $R_\Phi$ типа $F_4$
    является стандартным.
\end{lemma}
\begin{proof}
    %
    Пусть $r \in \mathcal{L}(H)$, $r \neq \rho$. Тогда существует простой корень
    $p$ с условием $r+p \in \Phi^+$ и одно из произведений $e_pe_r$, $e_re_p$ равно
    $e_{r+p}$.

    В случае, когда корни $q_{3,-2}$ и $p_{41}$ не лежат одновременно в
    $\mathcal{L}(H)$, множество $(\mathbb{K}e_p)H + H(\mathbb{K}e_p)$ по модулю
    $Q(r+p)$ совпадает с $\mathbb{K}e_{r+p}$ и порождает идеал, содержащий
    $Q(r+p)$, откуда $T(r+p) \subset H$. Учитывая произвол в выборе $p$, получаем
    включение $Q(r) \subset H$.

    Остается рассмотреть случай, когда $r \in \{q_{3,-2}, p_{41}\} \subset
        \mathcal{L}(H)$. Включение $T(p_{4,-1}) \in H$ получаем аналогично первому
    случаю, выбрав в качестве $p$ корень $q_{32}$. Полагая теперь $p=p_{32}$,
    включение $T(q_{42}) \subset H$ получаем, рассматривая множество
    $(\mathbb{K}e_p)H + H(\mathbb{K}e_p)$ по модулю $T(p_{4,-1}) + Q(q_{42})$.
\end{proof}

\begin{theorem}[{{\cite[теорема 1.1]{jsfu2018}}}]\label{th:stand-f4}
    %
    Существует как стандартные, так и нестандартные точные обертывающие алгебры
    $R_\Phi$ типа $F_4$.
\end{theorem}
\begin{proof}
    Переобозначая $N_{rs}$ через $N(r,s)$, выберем значения $N(r,s)$ на следующих
    парах корней:
    \begin{equation}\label{eq:StructConst-Stand-F4}
        \begin{gathered}
            N(q_{21},q_{3,-1}) = 1,\ N(q_{21},p_{31}) = -1, \
            N(q_{32},p_{3,-1}) = 1, \ N(q_{10},p_{3,-1}) = -1, \\
            %
            N(q_{32},q_{2,-1}) = -1, \ N(q_{32},q_{21}) = -1, \ N(q_{10},p_{32}) = -1, \
            N(q_{21},q_{10}) = -1.
        \end{gathered}
    \end{equation}

    Для получения нестандартной обертывающей алгебры $R_1$ дополнительно полагаем
    $N(q_{32},q_{20}) = -1$, \ $N(q_{10},q_{20}) = 2$ и
    \[
        N(p_{32},q_{2,-1}) = -1,\quad N(p_{32},q_{30}) = -1,\quad N(q_{10},q_{31}) = 1.
    \]
    Выбирая произвольно для оставшихся экстраспециальных пар корней $r$ и $s$ знаки
    структурных констант $N_{rs}$, получаем алгебру $R_1$. Легко видеть, что в
    алгебре $R_1$ идеалы вида
    \begin{equation}\label{eq:f4-nonstandard-ideal}
        \mathbb{K}(e_{q_{30}} + ce_{q_{2,-1}}) + \mathbb{K}(e_{p_{42}} + ce_{p_{3,-2}}) + T(q_{3,-1}) +
        T(q_{43}) \quad (c \in \mathbb{K}^*)
    \end{equation}
    являются нестандартными. Кроме того, легко проверить, что все остальные идеалы в $R_1$
    являются стандартными.

    \medskip

    Покажем, что знаки структурных констант $N_{rs}$ алгебры $N\Phi(\mathbb{K})$
    типа $F_4$ можно выбрать так, что все идеалы кольца $R_\Phi$ будут
    стандартными. Прямая вычислительная проверка показывает, что алгебра $R_\Phi$
    типа $F_4$ является стандартной, если выполняются следующие условия:
    \begin{equation*}\label{eq:F4-stand-condition}
        \begin{aligned}
            N(q_{21},q_{3,-1}) = -N(q_{21},p_{31}), \quad & N(q_{32},p_{3,-1}) =
            -N(q_{32},q_{2,-1}),                                                 \\
            N(q_{10},p_{3,-1}) = -N(q_{10},q_{31}), \quad & N(p_{32},q_{30}) =
            -N(p_{32},q_{2,-1}),                                                 \\
            N(q_{21},p_{31}) = N(q_{32},q_{21}), \quad    & N(q_{10},q_{31}) =
            -N(q_{10},p_{32}),                                                   \\
            N(q_{32},q_{21}) = N(q_{21},q_{10}), \quad    & N(q_{21},q_{10}) =
            N(q_{10},p_{32}) .
        \end{aligned}
    \end{equation*}

    Для построения стандартной обертывающей алгебры $R_2$, снова используем
    ~\eqref{eq:StructConst-Stand-F4}. Полагаем также $N(q_{32},q_{20}) = -1$ и
    $N(q_{10},q_{20})=- 2$. Далее, по тождеству Якоби получаем $N(p_{32},q_{20}) =
        1$ и
    \begin{equation*}\label{eq:Jacobi-F4}
        N(p_{32},q_{2,-1}) = 1,\quad
        N(p_{32},q_{30}) = -1,\quad
        N(q_{10},q_{31}) = 1.
    \end{equation*}
    Выбирая произвольно для оставшихся экстраспециальных пар корней $r$ и $s$ знаки
    структурных констант $N_{rs}$, получаем стандартную обертывающую алгебру $R_2$.
\end{proof}

\section{Теоремы единственности}\label{sec:uniqtheorems}
%
В этом параграфе мы исследуем для классических типов $\Phi$ условия
однозначности точных обертывающих алгебр нильтреугольной алгебры
$N\Phi(\mathbb{K})$.

\medskip

Ясно, что для любой подсистемы $\Psi$ системы корней $\Phi$ точная обёртывающая
алгебры Ли $N\Psi(\mathbb{K})$ определена в алгебре $R_{\Phi}$ как подалгебра
\[
    R(\Psi)\coloneq\sum_{a \in \Psi^+}\mathbb{K}e_a.
\]

Когда в системе $\Phi$ все корни одной длины и база $\Pi(\Psi)$ подсистемы
$\Psi$ лежит в базе $\Pi(\Phi)$, нестандартный идеал подалгебры $R(\Psi)$
порождает в алгебре $R_{\Phi}$ также нестандартный идеал. Отсюда вытекает

\begin{lemma}\label{l:stand-subsystem}
    Пусть $\Pi(\Psi) \subset \Pi(\Phi)$ для подсистемы $\Psi$
    системы корней $\Phi$ одной длины. Если точная обёртывающая
    $R_{\Phi}$ алгебры Ли $N\Phi(\mathbb{K})$ стандартная, то подалгебра
    $R(\Psi)$ алгебры $R_{\Phi}$ также стандартна.
\end{lemma}

Как и в \cite{JA-12}, корни $r, s$ называем \emph{$p$-связанными} для простого
корня $p$, если $r+p$, $s+p$ $\in \Phi^+$. Подсистему корней $\Psi(r, p, s)$ в
$\Phi$ определяем равенствами
\[
    \Psi=\Psi(r, p, s)\coloneq\Phi\cap (Zr+Zp+Zs).
\]

Алгебры $R_{\Phi}$ типа $A_{2}$ ассоциативны, поскольку для них $R_{\Phi}^3=0$.
В системах корней $\Phi$ одной длины подсистема $\Psi$ ранга 3 всегда типа
$A_3$. Очевидна

\begin{lemma}\label{l:subsystem-rank3}
    Если подсистема корней $\Psi=\Psi(r,p,s)$ ранга $>2$, то либо она
    типа $A_3$ с базой $\{r, p, s\}$, либо $\Psi$ типа $B_3$ или $C_3$.
\end{lemma}

Замена $a\circ b\coloneq ba$ умножения в $R$ дает \emph{противоположную
    алгебру} $R^{(op)}$, антиизоморфную $R$.

\begin{lemma}\label{l:NT-opposite}
    Алгебра $NT(n,\mathbb{K})$ нижних нильтреугольных $n\times n$ матриц над полем $\mathbb{K}$
    изоморфна своей противоположной алгебре $NT(n,\mathbb{K})^{(op)}$.
\end{lemma}
\begin{proof}
    Пусть $J=(j_{ab})$ "--- матрица перестановки с единицами на побочной диагонали,
    то есть $j_{a,n+1-a}=1$ и $j_{ab}=0$ иначе. Тогда $J^{-1}=J$.

    Рассмотрим линейное отображение
    \[
        \varphi\colon NT(n,\mathbb{K})\to NT(n,\mathbb{K}),\qquad \varphi(X)=JX^{t}J^{-1}=JX^tJ,
    \]
    где $X^t$ "--- транспонированная матрица. Если $X$ нижнетреугольна, то $X^t$
    верхнетреугольна, а сопряжение матрицей $J$ переводит верхнетреугольные матрицы
    в нижнетреугольные; значит, $\varphi(NT(n,\mathbb{K})) = NT(n,\mathbb{K})$.

    Для любых $X,Y\in NT(n,\mathbb{K})$ имеем
    \[
        \varphi(XY)=J(XY)^tJ=JY^tX^tJ=(JY^tJ)(JX^tJ)=\varphi(Y)\varphi(X),
    \]
    то есть $\varphi$ "--- антиавтоморфизм алгебры $NT(n,\mathbb{K})$.
    Следовательно, отображение $X\mapsto \varphi(X)$ задает изоморфизм
    $NT(n,\mathbb{K})\simeq NT(n,\mathbb{K})^{(op)}$.
\end{proof}

\begin{lemma}\label{l:stand-opposite}
    %
    Если точная обертывающая алгебра $R$ алгебры Ли $N\Phi(\mathbb{K})$ стандартна,
    то $R^{(op)}$ также является стандартной обертывающей алгеброй.
\end{lemma}
\begin{proof}
    %
    Очевидно, что множества идеалов в $R$ и в $R^{(op)}$ совпадают. Изоморфизм $x
        \mapsto -x \ (x \in R)$ модулей $R$ и $R^{(-)}$ является антиавтоморфизмом
    алгебры $R^{(-)}$ в силу равенств
    \[
        [-a, -b] = [a, b] = ab - ba = -[b, a] \quad (a,b \in R).
    \]
    Его композиция $-\mu\colon x\mapsto -x^{\mu}$ с любым антиизоморфизмом $\mu$
    алгебры $R$ дает изоморфизм алгебр $R^{(-)}$ и $\mu (R)^{(-)}$. (В важном
    частном случае этот изоморфизм хорошо известен \cite[\S 11.2.1]{carter72}.)

    Остается заметить, что алгебры $R$ и $R^{(op)}$ определены на одном множестве и
    их антиизоморфизмом является тождественное отображение.
\end{proof}

Отметим, что переход к противоположной алгебре $R^{(op)}$ сохраняет и
стандартность, и ассоциативность.

\begin{lemma}\label{l:stand-A3}
    Пусть $\Psi=\Psi(r, p, s)$ "--- подсистема типа $A_3$ для
    $p$-связанных корней $r, s$ в $\Phi$. Тогда стандартность
    подалгебры $R(\Psi)$ в $R_{\Phi}$ равносильна условию $N_{s,p} = N_{p, r}$ и
    дает равенство $N_{s+p,r}=N_{s,r+p}$, причем
    $R(\Psi) \simeq NT(4,\mathbb{K})$, когда $N_{s+p,r} =1$. Если $N_{s+p,r}= -1$, то
    $R(\Psi)$ "--- неассоциативная алгебра.
\end{lemma}

\begin{proof}
    В подсистеме $\Psi=\Psi(r,p,s)$ типа $A_3$ корень $r+p+s$ "--- единственный,
    представимый неоднозначно суммой специальной пары корней.

    Любую обертывающую алгебру $R(\Psi)$ алгебры Ли $N\Psi(\mathbb{K})$ порождают
    $\mathbb{K}e_r$, $\mathbb{K}e_s$ и $\mathbb{K}e_p$. Ненулевые структурные
    константы для $R(\Psi)$ типа $A_3$ равны $\pm 1$.

    % TODO определение аннулятора повторяется
    Для любого кольца $A$ наряду с его односторонними аннуляторами выделяют
    аннулятор
    \[
        \Ann\ (A)= \{\alpha\in A\ | \ \alpha A=0= A\alpha\}.
    \]
    Аннулятор кольца $R(\Psi)$ есть идеал $\mathbb{K}e_{r+p+s}=R(\Psi)^3$, причем
    \[
        R(\Psi)=\mathbb{K}e_r+(T(p)\cap R(\Psi))+\mathbb{K}e_s.
    \]
    Пересечение $T(p)\cap R(\Psi)$ и его аннулятор в $R(\Psi)$ совпадают с
    аннулятором идеала $R(\Psi)^2$. Если по модулю $ R(\Psi)^3$ односторонний
    аннулятор пересечения (равносильно, элемента $e_p$) совпадает с $R(\Psi)$ и,
    например, $e_p e_r=e_{r+p},\ e_re_p=0$, то $\mathbb{K}(e_p+ e_s)$ порождает в
    $R(\Psi)$ нестандартный идеал, как и в лемме~\ref{l:R-A3-D3}.

    Таким образом, при условии стандартности $R(\Psi)$ элементы $e_r, e_s$ лежат в
    разных односторонних аннулляторах элемента $e_p$. Это означает, что совпадают
    знаки констант $N_{p,r}$ и $N_{s,p}$, равносильно, $N_{p,r} N_{s,p}=1$.

    С учетом леммы \ref{l:stand-opposite}, с точностью до перехода к
    противоположной алгебре, имеем
    \begin{equation}\label{eq:An-Nsp-Npr-1}
        N_{s,p} =N_{p,r}=1,\quad
        e_se_p=e_{s+p},\quad
        e_pe_r=e_{r+p},\quad
        e_re_p=0=e_pe_s.
    \end{equation}

    Тождество Якоби в кольце Ли $R(\Psi)^{(-)}$ дает равенства
    \[
        [e_s,[e_p,e_r]] = [[e_s,e_p],e_r],\quad N_{pr}N_{s,r+p}=N_{sp}N_{s+p,r}.
    \]
    Отсюда $N_{s,r+p}=N_{s+p,r}\ (=\pm 1)$, то есть $N_{s,r+p}N_{s+p,r} = 1$. В
    случае
    \begin{equation}\label{eq:An-Nspr-1}
        N_{s+p,r} = 1, \quad
        e_{s+p}e_r=e_{r+s+p}=e_{s}e_{r+p}, \quad
        e_re_{s+p}=0=e_{r+p}e_s,
    \end{equation}
    приходим к изоморфизму алгебр $R(\Psi)\simeq NT(4,\mathbb{K})$, действующему по правилу:
    \[
        e_p \mapsto e_{32}, \ \ e_r \mapsto e_{21}, \ \ e_s \mapsto
        e_{43}, \ \ e_{r+p} \mapsto e_{31}, \ \ e_{s+p} \mapsto e_{42}, \
        \ e_{s+r+p } \mapsto e_{41}.
    \]

    В оставшемся случае, когда $N_{s+p,r} = -1$, имеем $e_p(e_re_s)=0$ и
    $(e_pe_r)e_s = e_{s+r+p}$. Поэтому алгебра $R(\Psi)$ неассоциативная; по модулю
    аннуллятора она изоморфна алгебре $NT(4,\mathbb{K})/\mathbb{K}e_{41}$. Лемма
    доказана.
\end{proof}

\begin{lemma}\label{l:stand-AnMod3}
    Если точная обертывающая алгебра $R$ алгебры Ли $N\Phi(\mathbb{K})$ типа $A_n$
    $(n>2)$ стандартна, то по модулю третьих степеней алгебр имеем
    \[
        R\simeq NT(n+1, \mathbb{K})\quad {\mbox {или } }\ \ R^{(op)}\simeq NT(n+1, \mathbb{K}).
    \]
\end{lemma}
\begin{proof}
    Исследуем произвольную стандартную обертывающую алгебру $R$
    алгебры Ли $N\Phi(\mathbb{K})$ типа $A_{n}\ (n> 2)$. Для любой подсистемы
    корней $\Psi= \Psi(r, p,s)$ типа $A_3$ в $\Phi$, по лемме
    \ref{l:stand-subsystem}, свойство стандартности $R$ наследуется
    подалгеброй $R(\Psi)$ "--- обертывающей алгебры Ли $N\Psi(\mathbb{K})$.

    \medskip

    Ясно, что пара $p$-связанных корней $r,s\in \Phi^+$ с простым корнем $p$
    существует только если $p$ "--- промежуточный корень в графе Кокстера

    Пусть $\Psi=\Psi(r, p,s)$ "--- подсистема в $\Phi$ с базой из простых корней
    $r,p,s$. Учитывая леммы \ref{l:stand-one-corner} и \ref{l:stand-A3}, для

    аннулляторах элемента $e_p$ "--- в левом $\Ann^{(l)} (e_p)$ и правом
    $\Ann^{(r)}(e_p)$. В этом случае знаки структурных констант $N_{s,p}$ и
    $N_{p,r}$ совпадают и $N_{s,p}N_{p,r}=1$.

    Когда корень $s$ соответствует последней вершине графа Кокстера, в силу леммы

    \medskip

    Выберем далее корень $q$, соседний слева с $r$ в графе Кокстера, и подсистему
    $\Psi=\Psi (q, r, p)$ с базой $q, r, p$. Применяя леммы
    \ref{l:stand-one-corner} и \ref{l:stand-A3} к подалгебре $R(\Psi)$, аналогично
    получаем $e_re_q = e_{r+q}$.

    Указанный процесс продолжаем, завершая подалгеброй $R(\Psi)$ с базой в $\Psi$,
    начинающейся с первой вершины графа Кокстера.

    Базу обертывающей алгебры $R=NT(n+1, \mathbb{K})$ и базу Шевалле алгебры
    Ли~$R^{(-)}$ дают матричные единицы $e_r= e_{ij}$ $(1 \leq j < i \leq n+1)$ при
    соответствующей нумерации корней $r=r_{ij}$ системы $\Phi$ типа $A_{n}$.

    В матричной индексации и обозначениях $e_{r}=e_{ij}$ при $r=r_{ij}$ получаем
    $e_{ij}e_{jm}=e_{im}$ для случаев $i-m=2$. Это дает изоморфизм алгебр $R$ и
    $NT(n+1,\mathbb{K})$ по модулю их третьих степеней. Лемма доказана.
\end{proof}

Все стандартные и все ассоциативные точные обертывающие алгебры $R$ алгебры Ли
$N\Phi(\mathbb{K})$ типа $A_{n-1}\ (n> 3)$ над полем $\mathbb{K}$
классифицирует

\begin{theorem}[{{\cite[теорема 1]{imm2020}}}]\label{th:Stand-An}
    Точная обертывающая алгебра $R$ алгебры Ли $N\Phi(\mathbb{K})$ типа
    $A_{n-1}$ стандартна тогда и только тогда, когда алгебра $R$ или
    $R^{(op)}$ по модулю аннулятора изоморфна $NT(n,\mathbb{K})/\mathbb{K}e_{n1}$.
    Ассоциативная точная обертывающая алгебра алгебры Ли $N\Phi(\mathbb{K})$
    типа $A_{n-1}$, с точностью до изоморфизма
    единственна и изоморфна алгебре $NT(n,\mathbb{K})$.
\end{theorem}

\begin{proof}
    Исследуем произвольную стандартную обертывающую алгебру $R$
    алгебры Ли $N\Phi(\mathbb{K})$ типа $A_{n}\ (n> 2)$. Для $n=3$ теорему
    \ref{th:Stand-An} доказывают леммы \ref{l:stand-A3} и \ref{l:stand-AnMod3}; в этом случае
    $\Phi=\Psi=\Psi(r, p, s)$ типа $A_3$, а пара $p$-связанных корней $r,s$ в
    $\Phi^+$ и простой корень $p$ определены однозначно.

    Каждая подалгебра $R(\Psi)$ в $R$ с подсистемой корней $\Psi= \Psi(r, p,s)$
    типа $A_3$ в $\Phi$ стандартна в силу леммы \ref{l:stand-subsystem}. С
    точностью до перехода от $R$ к противоположной алгебре, по лемме
    \ref{l:stand-A3} имеем соотношения \eqref{eq:An-Nsp-Npr-1}, а в случае
    ассоциативности $R$ также \eqref{eq:An-Nspr-1}. При $n > 3$ по индукции можем
    считать теорему доказанной для каждого типа $A_{k}$, $3\leq k <n$.

    Выберем в $\Phi$ подсистему $\Phi_{1}$ (аналогично $\Phi_{n}$) с базой,
    полученной из $\Pi$ отбрасыванием первого (соответственно, последнего) простого
    корня в графе Кокстера; обе подсистемы типа $A_{n-1}$. Ясно, что стандартность
    алгебры $R$ наследуется ее подалгебрами $R(\Phi_{1})$ и $R(\Phi_{n})$, по
    индуктивному предположению, и дает их ассоциативность, а также изоморфность
    алгебр $R$ и $NT(n+1,\mathbb{K})$ по модулю $n$-х степеней.

    Стандартность подалгебр $R(\Psi)$ в $R$ для подсистем корней $\Psi = \Psi(r,
        p,s)$ типа $A_3$ в $\Phi_{1} \cap \Phi_{n}$ приводит к уточнению. Матричная
    индексация корней и обозначения $e_{r}=e_{ij}$ при $r=r_{ij}$ $(1\leq j<i\leq
        n)$, наряду с ассоциативностью произведения $e_{n+1 n}e_{n n-1} \cdots
        e_{32}e_{21}=e_{n+1 1}$ и подалгебр $R(\Psi(r_{j1}, r_{ ij}, r_{n+1 i}))$ при
    $1< j<i< n+1$, по лемме \ref{l:stand-A3}, дают также изоморфность алгебр $R$ и
    $NT(n+1,\mathbb{K})$.

    Это завершает доказательство теоремы.
\end{proof}

\subsection{Градуированные алгебры}\label{sec:graded-algebras}
Базис Шевалле алгебры $\mathfrak{L}(\Phi,{\mathbb{K}})$ ранга $r$ задает на ней
стандартную тонкую \emph{$\mathbb{Z}\Phi$-градуировку Картана} целочисленной
решеткой корней $\mathbb{Z}\Phi \simeq \mathbb{Z}^r$. Компонент с нулевой
степенью в этой градуировке является подалгеброй Картана $H$, а остальные
нетривиальные компоненты являются одномерными $H$-инвариантными подалгебрами в
$\mathfrak{L}(\Phi,{\mathbb{K}})$. В частности, на подалгебре
$N\Phi(\mathbb{K})$ все нетривиальные компоненты индуцированной градуировки
являются одномерными.

Как выявляет \cref{th:Stand-Graded-An} в \S~\ref{sec:new-An}, алгебра $NT(n,
    \mathbb{K})$ является единственной градуированной нильпотентной ассоциативной
точной обертывающей алгеброй для алгебры Ли $N\Phi(\mathbb{K})$ типа $A_{n-1}$.

\medskip

Пусть $G$ "--- абелева группа. Будем говорить, что на алгебре $A$ задана
\emph{$G$-градуировка $\Gamma$}, если $A$ представлена в виде прямой суммы
подпространств
\[
    \Gamma\colon A = \bigoplus_{g \in G}A_g
\]
и включение $A_{g}A_{h} \subset A_{gh}$ выполняется для всех $g, h \in G$.
Множество элементов $\Supp\Gamma \coloneq \{g \in G \mid A_g \neq 0 \}$
называется \emph{носителем} градуировки $\Gamma$.

\medskip

Градуировки алгебр Ли абелевыми группами систематически изучались
в~\cite{patera1989}, см. также \cite{kochetov2013, latyshev1988}.

\medskip

Элемент $a \in A_g$ будем называть \emph{однородным степени $g$} и писать
$\mathrm{deg}_\Gamma(a) = g$. Отметим, что при таком определении $0 \in A$
является однородным элементом произвольной степени. Подпространство $A_g$
называется \emph{однородным компонентом степени $g$}. При фиксированной
градуировке $\Gamma$ алгебру $A$ будем называть \emph{градуированной}.

\noindent
%
Произвольный элемент $a \in A$ можно однозначно представить в виде суммы
$\sum_{g \in G}a_g$, где элементы $a_g \in A_g$ называются \emph{однородными
    компонентами $a$}. Подалгебра $R \subset A$ называется градуированной, если она
содержит однородные компоненты всех своих элементов, т.\;е.
\[ R = \bigoplus_{g \in G} (A_g \cap R).\]

\begin{lemma}\label{l:graded-factor}
    %
    Фактор-алгебра $A/I$ по градуированному идеалу $I$ наследует градуировку
    алгебры $A$.
\end{lemma}
\begin{proof}
    % В литературе подробного д-ва нет. Сделал как упражнение, думаю можно выкинуть.
    % Здесь просто дал развенутое д-во из R. Moody, A. Pianzola, Lie algebras with triangular decompositions, 1995, p.18
    Алгебру $A/I$ можно представить в виде прямой суммы
    \[
        A/I = \bigoplus_{g \in G} {(A/I)}_g,
    \]
    где ${(A/I)}_g \coloneq (A_g + I)/I \subset A/I$. Действительно, произвольный
    класс $a + I \in A/I$ раскладывается в сумму $\sum_{g \in G}(a_g + I)$ и каждое
    слагаемое лежит в $(A_g + I)/I$.

    \medskip

    Покажем, что сумма $\sum_{g \in G} (A_g + I)/I$ является прямой. Пусть
    \[
        \sum_{g\in G}(a_g + I) = \sum_{g\in G}a_g + I = 0 \text{ в } A/I.
    \]
    Это эквивалентно $\sum_{g\in G}a_g \in I$. Так как $I$ градуирован, то $a_g \in
        I$ и $a_g + I = 0$ для всех $g \in G$.

    Остается проверить согласованность с умножением. Для $g,h \in G$ имеем $(a_g +
        I)(a_h + I) = a_ga_h + I \in (A_{gh} + I)/I$, то есть
    \[
        (A/I)_g(A/I)_h \subset (A/I)_{gh}.\qedhere
    \]
\end{proof}

%Отметим также, что каноническая проекция $\pi \colon A_g \twoheadrightarrow
%    (A_g + I)/I$ имеет ядро $A_g \cap I = I_g$ и, по теореме об изоморфизме,
%\[
%    (A/I)_g = (A_g + I)/I \simeq A_g/I_g .
%\]

\medskip

Рассмотрим две градуировки алгебры A
\begin{equation*}
    \Gamma_1 \colon A = \bigoplus_{g \in G}A_g
    \quad \text{и} \quad
    \Gamma_2 \colon A = \bigoplus_{h \in H}A_h.
\end{equation*}

\medskip

Будем говорить, что $\Gamma_2$ является \emph{уточнением} градуировки
$\Gamma_1$, если существует \emph{гомоморфизм огрубления} $\phi\colon H \to G$
такой, что для каждого $h \in H$ выполняется включение $A_h \subset
    A_{\phi(h)}$. Если для некоторого $h \in H$ это включение строгое, то будем
говорить о \emph{собственном уточнении}.

Градуировка $\Gamma$ называется \emph{тонкой}, если она не допускает
собственных уточнений (\cite[определение 2]{patera1989}). Тонкую градуировку
$\Gamma$ будем называть \emph{градуировкой ширины один} \cite{mill-graded2019},
если все ее нетривиальные компоненты одномерны.

\begin{lemma}\label{l:graded-fine-existence}
    %
    Если конечномерная алгебра $A$ градуирована, то на ней существует тонкая
    градуировка.
\end{lemma}
\begin{proof}
    %
    Если градуировка $\Gamma_1$ алгебры $A$ не является тонкой, то у нее есть
    строгое уточнение $\Gamma_2$. Аналогично для $\Gamma_2$ найдем строгое
    уточнение $\Gamma_3$. Так как этот процесс не может повторяться более $\dim
        A-1$ раз, то за конечное число шагов мы получим тонкую градуировку.
\end{proof}

Алгебра $A \neq 0$ называется \emph{нильпотентной} класса нильпотентности $m$,
если $A^m \neq 0$ и $A^{m+1} = 0$. В частности, $A^m \subset \Ann(A) \neq 0$.

\medskip

Покажем, что любая тонкая градуировка конечномерной нильпотентной алгебры
является градуировкой ширины один.

\medskip

Для любой алгебры $A$ наряду с ее односторонними аннуляторами выделяют
аннулятор
\[
    \Ann(A) \coloneq \{\alpha\in A\ | \ \alpha A = 0 = A\alpha\}.
\]

\begin{lemma}\label{l:graded-ann}
    %
    Аннулятор $\Ann(A)$ градуированной алгебры $A = \oplus_{g \in G}A_g$ является
    ее градуированным идеалом.
\end{lemma}
\begin{proof}
    %
    По определению градуированного идеала нужно показать, что если $x = \sum_{g \in
            G} x_g \in \Ann(A)$, то каждый $x_g \in \Ann(A)$.

    Возьмем произвольный $h \in G$ и элемент $y \in A_h$. Так как $xA = 0$, имеем
    \[
        xy = \sum_{g \in G}x_gy = 0,
    \]
    где слагаемые $x_gy \in A_gA_h \subset A_{g+h}$, то есть лежат в различных
    однородных компонентах. Поскольку сумма $\oplus_{g \in G}A_g$ прямая, из
    равенства $\sum_{g \in G}x_gy = 0$ следует $x_gy = 0$ для всех $g$. Так как $y
        \in A_h$ произволен и $\sum_{h \in G} A_h = A$, получаем $x_gA = 0$ для всех
    $g$.

    Аналогично из $Ax=0$ получаем $Ax_g = 0$ для всех $g$. Значит, $x_g \in
        \Ann(A)$ и поэтому
    \[
        \Ann(A) = \bigoplus_{g \in G} (\Ann(A) \cap A_g),
    \]
    то есть $\Ann(A)$ "--- градуированный идеал.
\end{proof}

\emph{Точной последовательностью} (\cite[\S 13.3]{linalg1986}) будем
называть последовательность линейных пространств и отображений
% short exact sequence
\[
    \shorthandoff{"}
    \begin{tikzcd}[column sep=small]
        0 \arrow[r] &
        U \arrow[r, "\alpha"{xshift=-2pt}] &
        V \arrow[r, "\beta"{xshift=-2pt}] &
        W \arrow[r] &
        0
    \end{tikzcd}
    \shorthandon{"}
\]
с условиями $\mathrm{Im} \alpha = \mathrm{ker} \beta$, отображение $\alpha$
инъективно, а отображение $\beta$ сюръективно. В частности, из этих условий
следует, что $V=\alpha(U) \oplus S$ для некоторого подпространства $S \simeq
    W$, а значит $V \simeq U \oplus W$. Подпространство $S$ в $V$ будем называть
\emph{линейным сечением} на $W$.
\begin{lemma}\label{l:fine-graded-factor-ann}
    %
    Если на алгебре $A$ задана тонкая градуировка, то индуцированная градуировка на
    $\quot{A}{\Ann(A)}$ тоже тонкая.
\end{lemma}
\begin{proof}
    %
    Обозначим $I=\Ann(A)$. Согласно леммам~\ref{l:graded-factor} и
    \ref{l:graded-ann}, алгебра $A/I$ наследует градуировку алгебры $A$:
    \[
        \Gamma_{\pi} \colon A/I= \bigoplus_{g \in G}(A/I)_g.
    \]
    Докажем, что градуировка $\Gamma_{\pi}$ не допускает собственных уточнений.
    Предположим противное: градуировка $\Gamma_\pi$ имеет собственное уточнение
    \[
        \Gamma'_\pi\colon A/I = \bigoplus_{h \in H}(A/I)_h.
    \]
    Пусть $\pi\colon A \twoheadrightarrow A/I$ "--- каноническая проекция.
    Рассмотрим точную последовательность
    \[
        \shorthandoff{"}
        \begin{tikzcd}[column sep=small]
            0 \arrow[r] &
            I \arrow[r, "i"{xshift=-2pt}] &
            A \arrow[r, "\pi"{xshift=-2pt}] &
            A/I \arrow[r] &
            0
        \end{tikzcd}
        \shorthandon{"}
    \]
    и линейное сечение $s \colon A/I \hookrightarrow A\ (\pi \circ s =
        \mathrm{id}_{A/I}$). Тогда $s(A/I) = \oplus_{h \in H}s((A/I)_h)$ и
    \[
        A = s(A/I) \oplus I = \bigoplus_{h \in H}s((A/I)_h) \oplus I.
    \]

    У уточнения $\Gamma'_\pi$ есть гомоморфизм огрубления $\phi\colon H \to G$
    такой, что для каждого $h \in H$
    \[
        (A/I)_h \subset (A/I)_{\phi(h)}.
    \]

    Определим новую $G {\times} H$-градуировку $\Gamma'$ на $A$ следующим образом.

    \medskip

    \begin{enumerate}
        \item Если $a \in s((A/I)_h)$, то $\mathrm{deg}_{\Gamma'}(a) = (\phi(h), h)$.
        \item Если $a \in I$, то $a \in I \cap A_g$ для некоторого $g \in G$ и
              $\mathrm{deg}_{\Gamma'}(a) = (g, 0)$.
    \end{enumerate}

    \medskip

    По условию леммы, $IA = 0 = AI$ и поэтому $\Gamma'$ корректно согласована с
    умножением в $A$. Таким образом, мы получили уточнение градуировки на $A$.
    Остается показать, что полученное уточнение является собственным.

    \medskip

    Действительно, поскольку $\Gamma'_\pi$ собственно уточняет $\Gamma_\pi$,
    существует компонента $(A/I)_g$ градуировки $\Gamma_\pi$, которая строго
    содержит объединение нескольких компонент $(A/I)_h$ с $\phi(h) = g$ в
    градуировке $\Gamma'_\pi$. Их прообразы через сечение $s$ лежат в компоненте
    $A_g$ исходной градуировки $\Gamma$, но получают различные степени $(g, h)$ в
    градуировке $\Gamma'$.
\end{proof}

\begin{lemma}\label{l:graded-component-ann-intersection}
    %
    Пусть $\mathbb{K}$-алгебра $A$ имеет тонкую $G$-градуировку $\Gamma$. Если
    компонент $A_g$ имеет ненулевое пересечение с $\Ann(A)$, то $\dim A_g = 1$ и
    $A_g \subset \Ann(A)$.
\end{lemma}
\begin{proof}
    %
    Пусть $0\ne u\in A_g\cap \Ann(A)$. Если существует $v\in A_g$, не
    пропорциональный $u$, то градуировку $\Gamma$ можно уточнить следующим образом:
    положим $G' = G{\times}\mathbb{Z}$ и зададим $G'$-градуировку $\Gamma'$, вынеся
    $\mathbb{K}u$ в степень $(g,1)$, а все остальные элементы компоненты $A_g$
    оставив в степени $(g,0)$; на прочих компонентах вторую координату полагаем
    равной $0$. Это противоречит тонкости $\Gamma$. Следовательно, $\dim A_g=1$ и
    $A_g \subset\Ann(A)$.
\end{proof}

\begin{proposition}\label{p:fine-graded-nilpotent}
    %
    Любая тонкая градуировка конечномерной нильпотентной алгебры $A$ является
    градуировкой ширины один.
\end{proposition}
\begin{proof}
    %
    Пусть на конечномерной ассоциативной алгебре $A$ задана тонкая $G$-градуировка
    \[
        \Gamma\colon A = \bigoplus_{g \in G}A_g.
    \]
    Рассмотрим факторалгебру $\bar{A} = A/\Ann(A)$. Она тоже конечномерна и
    нильпотентна, причем $\dim \bar{A} < \dim A$.

    \medskip

    По леммам~\ref{l:graded-ann} и \ref{l:fine-graded-factor-ann}, градуировка на
    $A$ индуцирует тонкую градуировку
    \[
        \Gamma_\pi \colon \bar{A} = \bigoplus_{g \in G} \bar{A}_g
    \]
    и можем провести доказательство леммы индукцией по $\dim A$.

    \medskip

    По индукционному предположению для $\bar{A}$ имеем
    \[
        \dim \bar{A}_g \leq 1\ (g \in G).
    \]

    Обозначим $I=\Ann(A)$.

    \noindent
    %
    Каноническая проекция $\pi \colon A_g \twoheadrightarrow \bar{A}_g$ имеет ядро
    $A_g \cap I = I_g$ и, по теореме об изоморфизме,
    \[
        \bar{A}_g = (A_g + I)/I \simeq A_g/I_g.
    \]
    Для каждого $g \in G$, рассматривая точную последовательность
    \[
        \shorthandoff{"}
        \begin{tikzcd}[column sep=small]
            0 \arrow[r] &
            I_g \arrow[r, "i"] &
            A_g \arrow[r, "\pi"] &
            \bar{A}_g \arrow[r] &
            0,
        \end{tikzcd}
        \shorthandon{"}
    \]
    получаем
    \[
        \dim A_g = \dim(\Ann(A) \cap A_g) + \dim \bar{A}_g.
    \]
    и для завершения доказательства остается применить
    \crefVin{l:graded-component-ann-intersection}.
\end{proof}

Рассмотрим две градуированные алгебры
\begin{equation}\label{eq:two-graded-algebras}
    \Gamma_1 \colon A = \bigoplus_{g \in G}A_g
    \quad \text{и} \quad
    \Gamma_2 \colon B = \bigoplus_{h \in H}B_h.
\end{equation}

Градуировки $\Gamma_1$ и $\Gamma_2$ называются \emph{эквивалентными}, если
существует изоморфизм алгебр $\phi \colon A \to B$ и для любого $g \in G$
найдется $h \in H$ такой, что $\phi(A_g) = B_h$.

\begin{lemma}\label{l:one-dimensional-equivalence}
    Градуировки ширины один двух изоморфных алгебр эквивалентны.
\end{lemma}
\begin{proof}
    %
    Пусть $\phi \colon A \to B$ "--- изоморфизм алгебр
    \eqref{eq:two-graded-algebras}.

    Для любого $g \in G$ с $A_g \neq 0$ имеем $\dim A_g = 1$. Так как $\phi$ "---
    изоморфизм, $\phi(A_g)$ "--- одномерное подпространство в $B$ и, следовательно,
    существует единственное $h \in G$ такое, что $\phi(A_g) = B_h$. Действительно,
    если $0 \neq a \in A_g$ и $\phi(a) = \sum_{h \in G} b_h$ с $b_h \in B_h$, то
    для любого $c \in \mathbb{K}$ имеем $\phi(ca) = c\phi(a) = \sum_{h \in G}
        cb_h$. Так как $\phi(A_g) = \mathbb{K}\phi(a)$ одномерно и разложение $B =
        \bigoplus_{h \in G}B_h$ прямое, то $\phi(a)$ должен принадлежать ровно одному
    компоненту $B_h$.
\end{proof}

\subsection{Новая теорема для \texorpdfstring{$A_n$}{Aₙ}}\label{sec:new-An}
%
На алгебре $N\Phi(\mathbb{K})$ зафиксируем градуировку $\Gamma_C$ относительно
% положительной части базиса Шевалле
ее корневого базиса $\{e_r\mid r\in\Phi^+\}$:
\begin{equation*}%\label{eq:An-Cartan-Grading}
    \Gamma_C\colon N\Phi(\mathbb{K})=
    \bigoplus_{r\in\Phi^+}\mathbb{K}e_r
    %= \bigoplus_{\mathclap{1\leq j<i\leq n+1}}\mathbb{K}e_{ij}
    .
\end{equation*}

\begin{lemma}\label{l:enveloping-roots-grading}
    %
    Пусть $R$ "--- нильпотентная точная обертывающая алгебра алгебры Ли
    $N\Phi(\mathbb{K})$. Если $R$ градуирована, то $R$ можно реализовать на
    линейном пространстве $N\Phi(\mathbb{K})$ так, что $e_re_s = 0$, если $r+s
        \not\in \Phi^{+}$ и коммутатор в $R^{(-)}$ совпадает с коммутатором в
    $N\Phi(\mathbb{K})$.
\end{lemma}
\begin{proof}
    По \crefDat{l:graded-fine-existence} и \crefDat{p:fine-graded-nilpotent}
    существует градуировка $\Gamma$ ширины один
    \begin{equation}\label{eq:R-An-width-1}
        \Gamma\colon R=\bigoplus_{g\in G}R_g
    \end{equation}
    %
    алгебры $R$, имеющая ровно $\dim R$ ненулевых одномерных компонент. Градуировка
    $\Gamma$ также является градуировкой алгебры Ли $R^{(-)}$, поскольку умножение
    в $R^{(-)}$ совместимо с $\Gamma$.

    \medskip

    По лемме~\ref{l:one-dimensional-equivalence} градуировки $\Gamma$ на $R^{(-)}$
    и $\Gamma_{C}$ на $N\Phi(\mathbb{K})$ эквивалентны. Следовательно, существует
    изоморфизм алгебр Ли $\phi\colon R^{(-)}\to N\Phi(\mathbb{K})$, переводящий
    каждую ненулевую компоненту $R_g$ в некоторую корневую компоненту
    $\mathbb{K}e_{r}$.

    \medskip

    Так как $\dim R = \dim N\Phi(\mathbb{K})$, отображение $\phi$ индуцирует
    биекцию между носителем градуировки $\Gamma$ и множеством положительных корней
    $\Phi^+$. Это позволяет переиндексировать компоненты $R$ элементами $\Phi^+$:
    положим $R_r := \phi^{-1}(\mathbb{K}e_r)$. Зафиксируем в $R$ базис $\{e'_r =
        \phi^{-1}(e_{r}) \mid r \in \Phi^+\}$. Очевидно, что $e'_r \in R_r$ и $R_r =
        \mathbb{K}e'_r$.

    \medskip

    Рассмотрим произведение $e'_re'_s$ в алгебре $R$. Так как $R$ градуирована
    группой, а компоненты переиндексированы корнями аддитивно (в силу свойств
    изоморфизма $\phi$), то $e'_re'_s \in R_{r+s}$. Если $r+s \not\in \Phi^+$, то
    соответствующая компонента $R_{r+s}$ нулевая (так как носитель $\Gamma$
    совпадает с $\Phi^+$) и, следовательно, $e'_re'_s = 0$.

    \medskip

    Пусть теперь $r,s$ и $r+s \in \Phi^+$. Воспользуемся тем, что $\phi$ "---
    изоморфизм алгебр Ли. В алгебре $N\Phi(\mathbb{K})$ выполняется соотношение
    $[e_r, e_s] = N_{rs}e_{r+s}$, где $N_{rs}$ "--- структурные константы базиса
    Шевалле. Тогда
    \[
        [e'_r, e'_s]_{R^{(-)}} = \phi^{-1}([e_r, e_s]) = \phi^{-1}(N_{rs}e_{r+s}) = N_{rs}e'_{r+s}.
    \]

    \medskip

    Отождествляя векторное пространство $R$ с $N\Phi(\mathbb{K})$ посредством
    линейного отображения $e'_r \mapsto e_r$, получаем требуемую реализацию.
\end{proof}

\medskip

Для системы корней $\Phi$ типа $A_n$ будем использовать ее стандартную
реализацию~\cite[таблица 1]{bourbaki72}:
\begin{equation}\label{eq:An-Root-System}
    \begin{gathered}
        A_n=\{\epsilon_i-\epsilon_j \mid 1\leq i\neq j\leq n+1\},\qquad
        A_n^+=\{\epsilon_i-\epsilon_j \mid 1\leq j<i\leq n+1\},\\
        \Pi(A_n)=\{\alpha_i\coloneq \epsilon_{i+1}-\epsilon_i \mid 1\leq i\leq n\}.
    \end{gathered}
\end{equation}
%
В этой реализации базисные векторы $e_r=e_{\epsilon_i-\epsilon_j}$ алгебры
$N\Phi(\mathbb{K})$ удобно отождествлять с матричными единицами $e_{ij}$ в
$\mathfrak{sl}_{n+1}(\mathbb{K})$ (см., например, \cite[\S 11.2.1]{carter72}):
\begin{equation}\label{eq:An-matrix-realization}
    e_{\epsilon_i-\epsilon_j} \longleftrightarrow e_{ij} \qquad (1 \le j < i \le n+1).
\end{equation}

\begin{theorem}\label{th:Stand-Graded-An}
    %
    Если нильпотентная ассоциативная точная обертывающая алгебра $R$ алгебры Ли
    $N\Phi(\mathbb{K})$ типа $A_n\ (n \ge 3)$ градуирована, то $R \simeq
        NT(n+1,\mathbb{K})$.
\end{theorem}
\begin{proof}
    %
    Алгебру $R$ можно построить на пространстве $N\Phi(\mathbb{K})$ в матричной
    реализации \eqref{eq:An-matrix-realization}. В этой реализации коммутатор
    базисных элементов задается формулой $[e_{ij}, e_{kl}] = \delta_{jk}e_{il} -
        \delta_{li}e_{kj}$.

    Кроме того, \cref{l:enveloping-roots-grading} гарантирует, что $e_{ij}e_{kl} =
        0$, если сумма соответствующих корней не является корнем, и $e_{ij}e_{jk},
        e_{jk}e_{ij} \in \mathbb{K}e_{ik}$ в противном случае.

    \medskip

    Для типа $A_3$ это оставляет (с точностью до скаляров) только следующие
    возможные ненулевые произведения:
    \[
        e_{32}e_{21},\ e_{21}e_{32}\in\mathbb{K}e_{31};\qquad
        e_{43}e_{32},\ e_{32}e_{43}\in\mathbb{K}e_{42};
    \]
    \[
        e_{42}e_{21},\ e_{21}e_{42}\in\mathbb{K}e_{41};\qquad
        e_{43}e_{31},\ e_{31}e_{43}\in\mathbb{K}e_{41}.
    \]
    Поэтому существуют $a,a',b,b',c,c',d,d'\in\mathbb{K}$ такие, что
    \[
        e_{32}e_{21}=ae_{31},\quad e_{21}e_{32}=a'e_{31};\qquad
        e_{43}e_{32}=be_{42},\quad e_{32}e_{43}=b'e_{42};
    \]
    \[
        e_{42}e_{21}=ce_{41},\quad e_{21}e_{42}=c'e_{41};\qquad
        e_{43}e_{31}=de_{41},\quad e_{31}e_{43}=d'e_{41}.
    \]

    Коммутатор в $R^{(-)}$ совпадает с матричным коммутатором, поэтому
    \[
        a-a'=1,\qquad b-b'=1,\qquad c-c'=1,\qquad d-d'=1.
    \]

    Ассоциативность нетривиальна только для троек суммарной степени
    $\alpha_1+\alpha_2+\alpha_3$ (то есть $\epsilon_4-\epsilon_1$). Учитывая, что
    $e_{43}e_{21}=e_{21}e_{43}=0$, получаем систему условий
    \[
        a'd=0,\qquad ad'=0,\qquad bc'=0,\qquad b'c=0,\qquad bc=ad.
    \]
    Если $d\neq 0$, то из $a'd=0$ следует $a'=0$ и $a=1$, а из $ad'=0$ получаем
    $d'=0$ и $d=1$. Тогда $bc=ad=1\neq 0$, поэтому $b,c\neq 0$; равенства $bc'=0$ и
    $b'c=0$ дают $c'=0$, $b'=0$ и, следовательно, $c=1$, $b=1$. Таким образом,
    умножение в $R$ совпадает с умножением в $NT(4,\mathbb{K})$. Если же $d=0$, то
    $d'=-1$ и равенства выше аналогично влекут $a=0$, $a'=-1$, $b=0$, $b'=-1$,
    $c=0$, $c'=-1$. Тем самым, $R\simeq NT(4,\mathbb{K})^{(op)}$, и по лемме
    \ref{l:NT-opposite} снова $R\simeq NT(4,\mathbb{K})$.

    \medskip

    При $n > 3$ по индукции можем считать теорему доказанной для каждого типа
    $A_{k}$, $3\leq k <n$.

    \medskip

    Обозначим через $\Phi_1$ и $\Phi_n$ подсистемы корней типа $A_{n-1}$ в $\Phi$ с
    базами $\Pi_1 = \Pi \setminus \{\alpha_1\}$ и $\Pi_n = \Pi \setminus
        \{\alpha_n\}$. Рассмотрим подалгебры $R(\Phi_1)$ и $R(\Phi_n)$, соответствующие
    этим подсистемам. Они являются градуированными точными обертывающими алгебрами
    типа $A_{n-1}$. Следовательно, $R(\Phi_1) \simeq NT(n, \mathbb{K}) \simeq
        R(\Phi_n)$.

    Учитывая градуировку и фиксированные коммутаторы, это означает, что таблица
    умножения базисных элементов в $R(\Phi_k)$ совпадает либо с таблицей умножения
    в $NT(n, \mathbb{K})$ (стандартное умножение), либо с таблицей в $NT(n,
        \mathbb{K})^{(op)}$ (противоположное умножение).

    \medskip

    Пересечение $R(\Phi_1) \cap R(\Phi_n)$ соответствует подсистеме типа $A_{n-2}$
    с базой $\Pi \setminus \{\alpha_1, \alpha_n\}$. При $n > 3$ это пересечение
    содержит элементы с ненулевым коммутатором. Поэтому умножение на пересечении не
    может быть одновременно стандартным и противоположным. Следовательно, типы
    умножения в $R(\Phi_1)$ и $R(\Phi_n)$ совпадают.

    \medskip

    Пусть $R(\Phi_1)$ и $R(\Phi_n)$ имеют стандартное умножение. Тогда все
    произведения $e_{ij}e_{kl}$ внутри $R(\Phi_1)$ и внутри $R(\Phi_n)$ совпадают с
    матричными.

    Остается проверить произведения, где один множитель из $R(\Phi_1) \setminus
        R(\Phi_n)$ (содержит индекс $n+1$), а другой из $R(\Phi_n) \setminus R(\Phi_1)$
    (содержит индекс 1).

    \medskip

    Рассмотрим произведения вида $e_{n+1, k} e_{k, 1}$ ($k=2,\dots,n$). Используя
    ассоциативность и стандартность умножения в $R(\Phi_1)$ и $R(\Phi_n)$, можно
    показать, что их значение не зависит от $k$ и равно $X=e_{n+1, n} e_{n, 1}$.
    Например, при $k=2$:
    \[
        e_{n+1, 2} e_{21} = (e_{n+1, n} \dots e_{32}) e_{21} =
        e_{n+1, n} (\dots (e_{32} e_{21})) = e_{n+1, n} e_{n1}.
    \]
    Из коммутационных соотношений $[e_{n+1, k}, e_{k1}] = e_{n+1, 1}$
    ($k=2,\dots,n$) следует, что произведение $Y=e_{k1} e_{n+1, k}$ также принимает
    постоянное значение $X - e_{n+1, 1}$.

    Вычислим $Y$ при $k=2$:
    \[
        Y = e_{21}e_{n+1, 2} =
        e_{21}(e_{n+1, 3} e_{32}) =
        (e_{21}e_{n+1, 3})e_{32} =
        0.
    \]
    Тогда $X = e_{n+1, 1}$. Таким образом, $e_{n+1, k} e_{k1} = e_{n+1, 1}$ для
    всех $k$, что совпадает со стандартным умножением.

    Если же умножение в $R(\Phi_1)$ и $R(\Phi_n)$ противоположное, то аналогично
    получаем, что $R \simeq NT(n+1, \mathbb{K})^{(op)} \simeq NT(n+1, \mathbb{K})$.
\end{proof}
\begin{remark}\label{remark:LieAdmissible-Nonunique-Assoc-A3}
    %
    Если же в теореме \ref{th:Stand-Graded-An} не требовать наличия градуировки у
    алгебры $R$, то она, вообще говоря, не обязана быть изоморфной
    $NT(n+1,\mathbb{K})$.

    Действительно, в алгебре $NT(4,\mathbb{K})$ заменим умножение на новое $\ast$,
    полагая $x\ast y=xy$ для всех пар базисных элементов, кроме пары
    $(e_{32},e_{21})$, и дополнительно задавая
    \[
        e_{32}\ast e_{21}=e_{31}+e_{41},\qquad e_{21}\ast e_{32}=e_{41}.
    \]
    Тогда все коммутаторы базисных элементов совпадают с коммутаторами в
    $NT(4,\mathbb{K})$; значит, $(NT(4,\mathbb{K}),\ast)^{(-)}\simeq
        N\Phi(\mathbb{K})$ типа $A_3$. Ассоциативность умножения $\ast$ сохраняется,
    поскольку добавка лежит в аннуляторе $\Ann = \mathbb{K}e_{41}$.

    Полученная ассоциативная алгебра неизоморфна $NT(4,\mathbb{K})$: размерность
    левого аннулятора является инвариантом. В $NT(4,\mathbb{K})$ имеем
    $\Ann_{\ell}=\langle e_{21},e_{31},e_{41}\rangle$ и $\dim\Ann_{\ell}=3$, тогда
    как в $(NT(4,\mathbb{K}),\ast)$ элемент $e_{21}\notin\Ann_{\ell}$, поскольку
    $e_{21}\ast e_{32}=e_{41}\ne 0$, и потому $\dim\Ann_{\ell}=2$.
\end{remark}

\subsection{Прелиевы алгебры и их полупрямые суммы}\label{sec:pre-lie-algs}
%
Через $L_x\colon y\mapsto xy$ $(x,y \in R)$ обозначим линейные операторы левого
умножения в алгебре $R$.
\begin{definition}\label{def:pre-lie-algebra}
    %
    Алгебра $R$ над полем $\mathbb{K}$ называется \emph{прелиевой} или
    \emph{левосимметрической}, если ее умножение удовлетворяет тождеству
    \begin{equation*}
        (xy)z-x(yz)=(yx)z-y(xz) \qquad (x,y,z\in R)
    \end{equation*}
    или, эквивалентно, отображение $L\colon x \mapsto L_x$ задает представление
    $R^{(-)}$ в $\mathfrak{gl}(R)$:
    \begin{equation*}
        [L_x,L_y]=L_{[x,y]} \mathrlap{\qquad (x,y \in R).}
    \end{equation*}
\end{definition}

Как и в ассоциативном случае, всякая прелиева алгебра Ли-допустима: если
положить $[x,y]\coloneq xy-yx$, то $(R,[\ ,\ ])$ является алгеброй Ли.

\medskip

Левосимметрическую точную обертывающую алгебру для алгебры Ли $\mathfrak{L}$
также называют \emph{афинной} или \emph{левосимметрической структурой} на
$\mathfrak{L}$. Хорошо известно (\cite{burde-pre-lie-survey}), что полупростая
алгебра Ли над полем характеристики 0 не допускает афинных структур. В
\cite{milnor-left-sym} Дж. Милнор поставил вопрос, эквивалентный вопросу о
существовании афинных структур для разрешимых алгебр Ли. В
\cite{burde-counter-example} были построены примеры нильпотентных алгебр Ли, не
допускающих афинных структур.

Левосимметрическая алгебра $R$ называется \emph{алгеброй Новикова}, если в ней
выполняется правая коммутативность: $R_yR_z = R_zR_y$ для всех $y,z \in R$, где
$R_y\colon x\mapsto xy$. В \cite{burde-graaf-2013} разрабатывался метод
классификации алгебр Новикова с заданной ассоциированной алгеброй Ли.

Для построения прелиевых точных обертывающих алгебр алгебры Ли
$N\Phi(\mathbb{K})$ типов $B_n$, $C_n$ и $D_n$, используем их разложения в
полупрямые суммы подалгебры $NA_{n-1}(\mathbb{K})$ с подходящим идеалом.

\begin{definition}[{{\cite[\S~I.2.11]{vinberg-onishik}}}]\label{def:semidirect-lie}
    %
    \emph{Полупрямой суммой $\mathfrak{g}\ltimes_{\phi} \mathfrak{h}$} алгебр
    Ли $\mathfrak{g}$ и $\mathfrak{h}$ называется прямая сумма векторных
    пространств $\mathfrak{g}$ и $\mathfrak{h}$, снабженная операцией
    коммутирования по формуле
    \[
        [(x,a),(y,b)] = \bigl([x,y],\ \phi_x(b)-\phi_y(a) + [a,b]\bigr),
    \]
    где $\phi\colon\mathfrak{g}\to\Der(\mathfrak{h})$ "--- некоторый гомоморфизм
    алгебры Ли $\mathfrak{g}$ в алгебру Ли дифференцирований $\Der(\mathfrak{h})$ и
    $\phi_x = \phi(x)$.
\end{definition}
%
По аналогии с \crefTvor{def:semidirect-lie}, введем понятие полупрямой суммы
для прелиевых алгебр.
\begin{definition}\label{def:semidirect-pre-lie}
    %
    \emph{Полупрямой суммой}
    $R\ltimes_{\tau} I$ прелиевых алгебр $R$ и $I$ назовем прямую сумму
    векторных пространств $R$ и $I$, снабженную операцией умножения по формуле
    \[
        (x,a)(y,b) = (xy,\ \tau_x(b) + ab),
    \]
    где $\tau\colon R^{(-)} \to \Der(I)$ "--- некоторый гомоморфизм алгебры Ли
    $R^{(-)}$ в алгебру Ли дифференцирований $\Der(I)$ и $\tau_x = \tau(x)$.
\end{definition}
%
Всякое дифференцирование прелиевой алгебры $I$ является дифференцированием
алгебры Ли $I^{(-)}$, то есть $\Der(I)\subset \Der(I^{(-)})$. Таким образом,
при рассмотрении полупрямой суммы прелиевых алгебр $R\ltimes_{\tau} I$ можно
одновременно рассматривать и полупрямую сумму ассоциированных алгебр Ли
$R^{(-)}$ и $I^{(-)}$ с тем же действием $\tau$.

\medskip

Как показывают следующие две теоремы, полупрямая сумма ассоциативных алгебр
может быть неассоциативной, но всегда является прелиевой.
\begin{theorem}\label{th:semidirect-assoc-cond}
    %
    Пусть $R$ и $I$ "--- ассоциативные алгебры и $\tau\colon R^{(-)}\to\Der(I)$
    "--- гомоморфизм алгебры Ли $R^{(-)}$ в $\Der(I)$. Тогда алгебра
    $R\ltimes_{\tau} I$ ассоциативна тогда и только тогда, когда выполнены условия:
    \begin{enumerate}
        \item[\textup{(A)}] $\tau_{xy}=\tau_x\tau_y$ для всех $x,y\in R$;
        \item[\textup{(B)}] $\tau_x(I)\subset \Ann_r(I)$ для всех $x\in R$.
    \end{enumerate}
\end{theorem}
\begin{proof}
    %
    Пусть выполняются условия (A) и (B). Так как $R$ ассоциативна, достаточно
    проверить совпадение вторых компонент в равенстве
    \begin{equation}\label{eq:semidirect-assoc}
        ((x,a)(y,b))(z,c)=(x,a)((y,b)(z,c))
    \end{equation}
    %
    для всех $x,y,z \in R$ и $a,b,c \in I$. По определению умножения в
    $R\ltimes_{\tau} I$ имеем
    \[
        (x,a)(y,b)=(xy,\ \tau_x(b)+ab),\qquad (y,b)(z,c)=(yz,\ \tau_y(c)+bc).
    \]
    Поэтому вторая компонента левой части равна
    \[
        \tau_{xy}(c)+(\tau_x(b)+ab)c=\tau_{xy}(c)+\tau_x(b)c+(ab)c,
    \]
    а вторая компонента правой части равна
    \[
        \tau_x(\tau_y(c)+bc)+a(\tau_y(c)+bc)=\tau_x\tau_y(c)+\tau_x(bc)+a\tau_y(c)+a(bc).
    \]
    По \textup{(A)} имеем $\tau_x\tau_y(c) = \tau_{xy}(c)$. По условию
    $\tau_x\in\Der(I)$ и \textup{(B)} имеем
    \[
        \tau_x(bc) + a\tau_y(c) = \tau_x(b)c+b\,\tau_x(c) = \tau_x(b)c.
    \]
    Наконец, так как $I$ ассоциативна, то $a(bc)=(ab)c$. Следовательно, вторые
    компоненты совпадают, что и доказывает ассоциативность $R\ltimes_{\tau} I$.

    \medskip

    Пусть теперь алгебра $R\ltimes_{\tau} I$ ассоциативна. Подставляя в тождество
    ассоциативности \eqref{eq:semidirect-assoc} тройку $(x,0),(y,0),(0,c)$,
    получаем
    \[
        ((x,0)(y,0))(0,c)=(xy,0)(0,c)=(0,\tau_{xy}(c)),
    \]
    \[
        (x,0)((y,0)(0,c))=(x,0)(0,\tau_y(c))=(0,\tau_x\tau_y(c)),
    \]
    откуда следует (A).

    \medskip

    Далее, подставляя в \eqref{eq:semidirect-assoc} тройку $(x,0),(0,b),(0,c)$,
    получаем
    \[
        ((x,0)(0,b))(0,c)=(0,\tau_x(b))(0,c)=(0,\tau_x(b)c),
    \]
    \[
        (x,0)((0,b)(0,c))=(x,0)(0,bc)=(0,\tau_x(bc)),
    \]
    то есть $\tau_x(bc)=\tau_x(b)c$ для всех $b,c\in I$. Так как
    $\tau_x\in\Der(I)$, имеем
    \[
        \tau_x(bc)=\tau_x(b)c+b\,\tau_x(c),
    \]
    и потому $b\,\tau_x(c)=0$ для всех $x \in R$ и $b,c\in I$, что равносильно (B).
\end{proof}
\begin{theorem}\label{th:semidirect-pre-lie}
    %
    Пусть $R$ и $I$ "--- прелиевы алгебры и $\tau\colon R^{(-)}\to\Der(I)$ "---
    гомоморфизм алгебры Ли $R^{(-)}$ в $\Der(I)$. Тогда алгебра $R\ltimes_{\tau} I$
    является прелиевой. Более того, $(R\ltimes_{\tau} I)^{(-)}$ изоморфна
    полупрямой сумме $R^{(-)}\ltimes_\tau I^{(-)}$, причем
    \[
        [(x,a),(y,b)]=\bigl([x,y],\ \tau_x(b)-\tau_y(a)+[a,b]\bigr).
    \]
\end{theorem}
\begin{proof}
    %
    Непосредственным вычислением находим формулу коммутатора в $(R\ltimes_{\tau}
        I)^{(-)}$:
    \[
        [(x,a),(y,b)]=\bigl([x,y],\ \tau_x(b)-\tau_y(a)+[a,b]\bigr),
    \]
    и тем самым получаем изоморфизм $(R\ltimes_{\tau} I)^{(-)} \simeq
        R^{(-)}\ltimes_\tau I^{(-)}$.

    \medskip

    Докажем прелиевость $R\ltimes_{\tau} I$.

    \medskip

    Для $(x,a) \in R\ltimes_{\tau} I$ оператор $L_{(x,a)}$ действует на
    $R\ltimes_{\tau} I$ по формуле
    \[
        L_{(x,a)}(y,b)=(xy,\ \tau_x(b)+ab),
    \]
    то есть $L_{(x,a)}|_R = L_x$ и $L_{(x,a)}|_I = \tau_x+L_a$.

    \medskip

    Так как $R$ и $I$ прелиевы, получаем $[L_x,L_y]=L_{[x,y]}$ и
    $[L_a,L_b]=L_{[a,b]}$. Кроме того, поскольку $\tau$ "--- гомоморфизм алгебры Ли
    $R^{(-)}$ в $\Der(I)$, имеем $[\tau_x,\tau_y]=\tau_{[x,y]}$ и
    $[\tau_x,L_a]=L_{\tau_x(a)}$.

    Пользуясь полученными тождествами, в ограничениях на $R$ и $I$ получаем
    \begin{align*}
        \restr{[L_{(x,a)}, L_{(y,b)}]}{R}
         & = [L_x, L_y] = L_{[x,y]}                                        \\
         & = \restr{L_{[(x,a),(y,b)]}}{R},                                 \\
        \restr{[L_{(x,a)}, L_{(y,b)}]}{I}
         & = [\tau_x + L_a, \tau_y + L_b]                                  \\
         & = [\tau_x, \tau_y] + [L_a, \tau_y] + [\tau_x, L_b] + [L_a, L_b] \\
         & = \tau_{[x,y]} - L_{\tau_y(a)} + L_{\tau_x(b)} + L_{[a,b]}      \\
         & = \tau_{[x,y]} + L_{[a,b] + \tau_x(b) - \tau_y(a)}              \\
         & = \restr{L_{([x,y],\ \tau_x(b) - \tau_y(a) + [a,b])}}{I}        \\
         & = \restr{L_{[(x,a),(y,b)]}}{I}.
    \end{align*}

    Поскольку оба оператора $[L_{(x,a)}, L_{(y,b)}]$ и $L_{[(x,a),(y,b)]}$
    сохраняют подпространства $R$ и $I$ и совпадают на них, получаем
    \[
        [L_{(x,a)},L_{(y,b)}]=L_{[(x,a),(y,b)]}.\qedhere
    \]
\end{proof}
\begin{corollary}\label{corollary:enveloping-semidirect}
    %
    Пусть $\mathfrak{L} = \mathfrak{l}\ltimes_{\phi}\mathfrak{h}$ "--- полупрямая
    сумма алгебр Ли, $R$ и $I$ "--- прелиевы точные обертывающие алгебры для
    $\mathfrak{l}$ и $\mathfrak{h}$ соответственно. Пусть также задан гомоморфизм
    алгебр Ли $\tau\colon R^{(-)}\to\Der(I)$ такой, что индуцированное действием
    $\tau$ действие алгебры Ли $R^{(-)}$ на $I^{(-)}$ совпадает с действием $\phi$.
    Тогда алгебра $R\ltimes_{\tau} I$ является прелиевой точной обертывающей
    алгеброй алгебры Ли $\mathfrak{L}$.
\end{corollary}
\begin{proof}
    %
    По \crefDat{th:semidirect-pre-lie} алгебра $R\ltimes_{\tau}I$ прелиева, а
    значит, Ли-допустима. Кроме того,
    \[
        (R\ltimes_{\tau}I)^{(-)}\simeq
        R^{(-)}\ltimes_\tau I^{(-)}\simeq
        \mathfrak{l}\ltimes_{\phi}\mathfrak{h} = \mathfrak{L},
    \]
    откуда $R\ltimes_{\tau}I$ является точной обертывающей алгеброй алгебры Ли
    $\mathfrak{L}$.
\end{proof}

\subsection{Новая теорема для \texorpdfstring{$B_n$}{Bₙ}, \texorpdfstring{$C_n$}{Cₙ} и \texorpdfstring{$D_n$}{Dₙ}}\label{sec:new-Bn}
%
Используем \crefVin{th:semidirect-pre-lie} и ее
\crefVin{corollary:enveloping-semidirect} для построения нильпотентных
градуированных прелиевых точных обертывающих алгебр для $N\Phi(\mathbb{K})$
типов $B_n$, $C_n$ и $D_n$.

\medskip

Пусть $\Psi = A_{n-1}^+$ из \eqref{eq:An-Root-System}. Системы положительных
корней типов $D_n$, $C_n$ и $B_n$ можно представить в виде объединения
(\cite[таблицы I--IV]{bourbaki72})
\[
    \begin{aligned}
        D_n^+ & = \Psi \cup \{\epsilon_i+\epsilon_j \mid 1\le j<i\le n\}, \\
        C_n^+ & = D_n^+ \cup \{2\epsilon_i \mid 1\le i\le n\},            \\
        B_n^+ & = D_n^+ \cup \{\epsilon_i \mid 1\le i\le n\}.
    \end{aligned}
\]
%
Таким образом, $\Phi^+$ составляется из корней вида
\[
    r_{i,mj} = \epsilon_i - m\epsilon_j, \quad
    1 \leq j \leq i  \leq n,\
    m \in \{0, -1, +1\}.
\]
Базу системы корней $\Phi$ типов $D_n$, $C_n$ и $B_n$ запишем в виде $\Pi(\Phi)
    = \Pi(\Psi) \cup \{r_\Phi\}$, где
\[
    r_\Phi\coloneq
    \begin{cases}
        r_{2,-1}, & \Phi = D_n, \\
        r_{1,-1}, & \Phi = C_n, \\
        r_{10},   & \Phi = B_n.
    \end{cases}
\]

% %
% Полагая $e_r = e_{i,mj}$ при $r=p_{i,mj}$, получаем реализацию
% $NB_n(\mathbb{K})$ алгебры $N\Phi(\mathbb{K})$ типа $B_n$ с умножением:
% \begin{alignat*}{2}
%     [e_{ij},e_{jv}]   & = e_{iv},    \quad &  & i > j > |v| \geq 0; \\
%     [e_{jv},e_{i,-v}] & = e_{i,-j},  \quad &  & i > j > |v| > 0;    \\
%     [e_{i0},e_{j0}]   & = 2e_{i,-j}, \quad &  & i > j \geq 1.
% \end{alignat*}

% \medskip

Ясно, что $N\Psi = \langle e_r\mid r \in \Psi \rangle \simeq
    NT(n,\mathbb{K})^{(-)}$ является подалгеброй в $N\Phi(\mathbb{K})$ для системы
корней $\Phi$ ранга $n$.

\medskip

Обозначая через $\phi$ присоединенное действие $NT(n,\mathbb{K})^{(-)}$ на
идеале $T(r_\Phi)$, получаем разложение $N\Phi(\mathbb{K})$ в полупрямую сумму
алгебр Ли
\begin{equation}\label{eq:NBn-semi-direct-sum}
    N\Phi(\mathbb{K})\simeq NT(n,\mathbb{K})^{(-)} \ltimes_\phi T(r_\Phi).
\end{equation}

\medskip

По \crefDat{corollary:enveloping-semidirect} \crefRod{th:semidirect-pre-lie},
чтобы построить точную обертывающую алгебру $N\Phi(\mathbb{K})$ типов $D_n$,
$C_n$ и $B_n$ в виде полупрямой суммы ассоциативных алгебр, достаточно выбрать
на идеале $T(r_\Phi)$ ассоциативное умножение
\[
    \cdot_\Phi\colon T(r_\Phi) \times T(r_\Phi) \to T(r_\Phi)
\]
такое, что $(T(r_\Phi), \cdot_\Phi)^{(-)} \simeq T(r_\Phi)$. Кроме того,
присоединенное действие $\phi$ алгебры Ли $NT(n,\mathbb{K})^{(-)}$ на
$T(r_\Phi)$ должно быть согласовано с $\cdot_\Phi$: оператор $\phi_x$ для
каждого $x\in NT(n,\mathbb{K})$ должен быть дифференцированием ассоциативной
алгебры $\bigl(T(r_\Phi),\cdot_\Phi\bigr)$.

\medskip

Для типов $D_n$ и $C_n$ идеал $T(r_\Phi)$ является абелевым и в качестве
$\cdot_\Phi$ можем взять нулевое умножение.

\medskip

Рассмотрим тип $B_n$. Поскольку центр $Z$ алгебры Ли $T(r_{10})$ совпадает с ее
коммутантом, то $T(r_{10})^3 = 0$ и умножение
\[
    e_r \cdot_{B_n} e_s \coloneq
    \begin{cases}
        \dfrac12[e_r,e_s], & \Char \mathbb{K}\neq 2, \\[4pt]
        0,                 & \text{иначе}
    \end{cases}
    \mathrlap{\qquad\bigl(e_r,e_s \in T(r_{10})\bigr)}
\]
дает ассоциативную точную обертывающую алгебру $\bigl(T(r_{10}),
    \cdot_{B_n}\bigr) \simeq T(r_{10})$.

\begin{theorem}\label{th:Bn-pre-lie}
    %
    Полупрямая сумма ассоциативных алгебр
    \[
        R\Phi'(\mathbb{K}) \coloneq NT(n,\mathbb{K}) \ltimes_{\phi} \bigl(T(r_\Phi), \cdot_\Phi\bigr)
    \]
    является прелиевой точной обертывающей алгебры $N\Phi(\mathbb{K})$ типа $D_n$,
    $C_n$ или $B_n$.
\end{theorem}

\begin{remark}\label{remark:RBnprime-nonassoc}
    %
    При $\Char\mathbb{K}\neq 2$ и $n \geq 3$ алгебра $RB'_n(\mathbb{K})$
    неассоциативна независимо от конкретного выбора ассоциативного умножения на
    $T(r_{10})$. Действительно, пусть $(\cdot)$ "--- любое ассоциативное умножение
    на линейном пространстве $T(r_{10})$ такое, что $(T(r_{10}),\cdot)^{(-)}\simeq
        T(r_{10})$.

    \medskip

    Если бы алгебра $NT(n,\mathbb{K})\ltimes_{\phi} \bigl(T(r_{10}),\cdot\bigr)$
    была ассоциативной, то по \crefDat{th:semidirect-assoc-cond} имели бы
    $\phi_x\bigl(T(r_{10})\bigr)\subset \Ann_r\bigl(T(r_{10}),\cdot\bigr)$ для всех
    $x \in NT(n,\mathbb{K})$.

    \medskip

    Для любых $a,b\in \Ann_r\bigl(T(r_{10}),\cdot\bigr)$ имеем $a\cdot b=b\cdot
        a=0$, то есть $[a,b] = 0$. Если классы элементов $a,b\in T(r_{10})$ в
    $\quot{T(r_{10})}{Z}$ линейно независимы, то $[a,b]\ne 0$. Отсюда следует, что
    проекция подпространства $\Ann_r(T(r_{10}),\cdot)$ в фактор
    $\quot{T(r_{10})}{Z}$ имеет размерность не более $1$.

    С другой стороны, при $n\ge 3$, присоединенное действие $\phi$ алгебры Ли
    $NT(n,\mathbb{K})$ на факторе $\quot{T(r_{10})}{Z}$ нетривиально и имеет образ
    размерности не менее $2$. Следовательно, существуют $x_1,x_2\in
        NT(n,\mathbb{K})$ и $u_1,u_2\in T(r_{10})$ такие, что классы $\phi_{x_1}(u_1)$
    и $\phi_{x_2}(u_2)$ в $\quot{T(r_{10})}{Z}$ линейно независимы. Получили
    противоречие.
\end{remark}

\begin{remark}\label{remark:RCnprime-nonassoc}\label{remark:RDnprime-nonassoc}
    %
    Алгебра $RC'_n(\mathbb{K})$ не является ассоциативной при $n\ge 3$. Аналогично,
    алгебра $RD'_n(\mathbb{K})$ не является ассоциативной при $n\ge 4$.

    \medskip

    Действительно, если бы одна из этих алгебр была ассоциативной, то по
    \crefDat{th:semidirect-assoc-cond} выполнялось бы условие \textup{(A)}:
    $\phi_{xy}=\phi_x\phi_y$ для всех $x,y\in NT(n,\mathbb{K})$. Возьмем $x =
        e_{21}$ и $y = e_{32}$ (тогда $xy=0$ в $NT(n,\mathbb{K})$) и положим $u =
        e_{2,-1} \in T(r_\Phi)$. Тогда $\phi_{xy}(u)=0$, но $\phi_x\phi_y(u)\ne 0$, то
    есть $\phi_{xy}\ne\phi_x\phi_y$. Условие \textup{(A)} нарушается.
\end{remark}

Перечислим возможные типы $\Phi$ стандартных обертывающих алгебр $R_{\Phi}$.

\begin{proposition}\label{p:Exc-Dn-En}
    Стандартная обертывающая $R_{\Phi}$ алгебры Ли $N\Phi(\mathbb{K})$
    существует для всех типов ${\Phi}$, кроме типа $D_n$ $(n \geq 4)$
    и $E_n\ (n=6,7$ и $8)$.
\end{proposition}

\begin{proof}
    Существование стандартной обертывающей алгебры $R_{\Phi}$ над
    полем для $\Phi$ типа $B_n$ и $C_n$ указывает (\cite[Теорема
        4]{DAN2018}, \cite{imm2020}) следующая

    \begin{theorem}\label{th:classical-Exc-RDn}
        Пусть $R_{\Phi}$ есть алгебра классического типа из списка
        \ref{eq:RAn-RDn}. Если $s$ "--- угол идеала $H$ кольца $R_{\Phi}$ и
        $Q(s) \not\subset H$, то $\Phi$ типа $D_n$, $s=r_{iv}$, $2 \leq
            i<n$, $v=\pm 1$, причем идеалы $T(r_{i+1,1})$ и $T(r_{i+1,-1})$
        оба не лежат в $H$.
    \end{theorem}

    В силу теорем \ref{th:Stand-An} и \ref{th:classical-Exc-RDn}, для алгебр Ли
    $N\Phi(\mathbb{K})$ классических типов доказательство предложения
    \ref{p:Exc-Dn-En} требуется лишь для типа $D_n, \ n>3$.

    \medskip

    Допустим, что стандартная точная обертывающая алгебра $R_{\Phi}$ для $\Phi$
    типа $D_n$ существует. Граф Кокстера системы корней $\Phi$ (его вершины
    соответствуют простым корням) типа $D_n$ представляется в виде

    \medskip

В отличие от типа $D_n$, в системах ${\Phi}$ типа $B_n$ и $C_n$ корни $r_{jv}$
и $r_{j, -v}$ всегда инцидентны, то есть в $R_{\Phi}$ один из идеалов
$T(r_{jv})$ и $T(r_{j, -v})$ лежит в другом.

\medskip

Далее идеал $T(r)$ обертывающей алгебры $R_{\Phi}$ классического типа
обозначаем при $r=r_{iv}$ через $T_{iv}$, кроме случая $v=1$ для типа $D_n$,
где
\[
    T_{i1}\coloneq T_{i,-1}+T(r_{i1}).
\]
Как показывают леммы \ref{l:stand-one-corner} и \ref{l:StructConst-vl90a_ru},
стандартны все идеалы алгебры $R_{\Phi}$, лежащие в идеале вида $T_{i,-1}$ или
"--- для типа $B_n$ "--- в $T_{1 0}$.

Учитывая лемму~\ref{l:ReductBnCnDn} и теорему~\ref{th:Stand-An}, на алгебры
$R_{\Phi}$ накладываем условие

\begin{equation}\label{IsomR-Bn}
    B_n:\ R_{\Phi}/T_{2,-1}\simeq NT(n+1,\mathbb{K});
\end{equation}
\begin{equation}\label{IsomR-Cn}
    C_n:\ R_{\Phi}/T_{2,-2}\simeq NT(n+1,\mathbb{K}).
\end{equation}

Тогда элементы $e_{jm}$ алгебры $R_{\Phi}$ при $j>m>0$ умножаются по обычным
для матричных единиц правилам, причем для ${\Phi}$ типа $B_n$, в силу
предложения~\ref{p:mult-enveloping},

\begin{equation}\label{IsomR-B3}
    e_{jm}e_{m0}=e_{j0},\qquad e_{m0}e_{j0}=- e_{j0}e_{m0}=c_{jm}
    e_{j,-m}\ \ (c_{jm}=\pm 1).
\end{equation}

\begin{lemma}\label{l:Subsistem-B3}
    Алгебра $R_{\Phi}$ типа $B_3$ с условием (\ref{IsomR-Bn})
    стандартная. С точностью до выбора знака произведения $e_{10}e_{2
                0}=\pm e_{2,-1}$, она определена однозначно, когда подалгебра
    $R(\Psi)$ для подсистемы $\Psi$ типа $A_3$ в $\Phi$ ассоциативная.
\end{lemma}

\begin{proof}
    Все длинные корни системы корней $\Phi$ типа $B_3$ образуют
    единственную подсистему $\Psi$ типа $A_3$. С учетом равенств
    (\ref{IsomR-B3}), условие (\ref{IsomR-Bn}) легко дает
    стандартность алгебры $R_{\Phi}$.

    Базу в $\Psi$ дают корень $r_{32}$ и $r_{32}$-связанные корни $r_{2,-1}$ и
    $r_{21}$, то есть
    \[
        \Psi= \Psi(r_{2,-1}, r_{32}, r_{21}).
    \]
    В силу (\ref{IsomR-Bn}), имеем $e_{32} e_{21}= e_{31}$, $e_{21} e_{32} = 0$.
    Поэтому стандартность подалгебры $R(\Psi)$ по лемме \ref{l:stand-A3}
    равносильна равенствам
    \[
        e_{32}e_{2, - 1}=0,\ \ e_{2,-1}e_{32} = e_{3,-1}.
    \]
    Тождество Якоби алгебры Ли $N{\Phi}(\mathbb{K})$,
    предложение~\ref{p:mult-enveloping} и лемма~\ref{l:StructConst-vl90a_ru} дают:
    %
    \[ [e_{20},e_{10}]=[[e_{21},e_{10}],e_{10}]=2ce_{2,-1},\quad
        e_{20}e_{10}=ce_{2,-1}\quad (c=\pm 1);\]
    %
    \[ [e_{30},e_{10}]=[[e_{32},e_{20}],e_{10}]=[e_{32},[e_{20},e_{10}]]=
        2ce_{3,-1},\ \ e_{30}e_{10}=ce_{3,-1}.\]
    %
    \[ [e_{30},e_{20}]=[e_{30},[e_{21},e_{10}]]=[e_{21},[e_{30},e_{10}]]=
        2ce_{3,-2},\ \ e_{30}e_{20}=ce_{3,-2}.\]
    %
    (См. также замечание \ref{remark:Stand-Non-Assoc-An}.) Далее,
    %
    \[[e_{2,-1},e_{31}]=[e_{2,-1},[e_{32},e_{21}]]=[[e_{2,-1},e_{32}],e_{21}]=[e_{3,-1},e_{21}]
        %
        =\pm e_{3,-2},\]
    %
    то есть либо $e_{2,-1}e_{31} = e_{3,-2}$ и $e_{31}e_{2,-1} = 0$, либо
    $e_{2,-1}e_{31} = 0$ и $e_{31}e_{2,-1} = e_{3,-2}$.
    %
    К ассоциативности подалгебры $R(\Psi)$ приводят оба случая:
    %
    \[e_{2,-1}(e_{32}e_{21})=e_{2,-1}e_{31}=e_{3,-2}=
        (e_{2,-1}e_{32})e_{21}=e_{3,-1}e_{21}. \]
    %
    Однако, выбор знака константы $N_{rs}$ для специальной пары $(r,s)$ корней с
    суммой $r+s= r_{3,-2}$ произволен, согласно ~\cite[Предложение
        4.2.2]{carter72}, лишь при условии экстраспециальности пары $(r,s)$.

    Таким образом, выбор знака $c=\pm 1$ определяет алгебру $R_{\Phi}$ однозначно.
\end{proof}

\begin{lemma}\label{l:Subsistem-Bn}
    Пусть $R_n=R_{\Phi}$ есть стандартная обертывающая алгебра типа
    $B_n$ $(n > 3)$ с условием $($\ref{IsomR-Bn}$)$. Тогда $R_{\Phi}=RB_n
        (\mathbb{K})$ по модулю идеала $T_{n,-1}$, с точностью до выбора знака
    $c=\pm 1$ произведения $e_{20} e_{10}= ce_{2,-1}$.
\end{lemma}

\begin{proof}
    В системе корней ${\Phi}$ типа $B_n$ $(n>3)$ выделим подсистемы
    $\Psi_{m,j,i,k}$ типа $B_4$ с базой $\{r_{m0}, r_{jm}, r_{ij},
        r_{ki}\}$. Корень $r_{ij}$ и $r_{ij}$-связанные корни $r_{j,-m}$,
    $r_{ki}$ образуют базу подсистемы $\Psi$ типа $A_3$, то есть
    %
    \begin{equation}\label{SubsystA3-Bn}
        \Psi=\Psi (r_{j, -m}, r_{ij},  r_{ki}),\quad 1\leq m<j<i<k\leq n.
    \end{equation}
    %

    Пусть стандартная алгебра $R_{\Phi}$ выбрана с условием (\ref{IsomR-Bn}). По
    лемме \ref{l:stand-A3}, если $e_{ij} e_{j,-m} =0$ (как и в лемме
    \ref{l:Subsistem-B3} при $n=3$), то в подалгебре $R(\Psi)$, а при $i=j+1$ и в
    алгебре $R_{\Phi}$, находим нестандартные идеалы с двумя углами $r_{j,-m}$,\
    $r_{ki}$. Следовательно, $e_{ij}e_{j, -m}=e_{i,-m}$ и, по лемме
    \ref{l:stand-A3},
    %
    \[e_{j v}e_{ij}=0,\quad e_{ij} e_{j v}=e_{iv}=[e_{ij}, e_{j v}]\qquad  (i>j>|v|=m>0).\]
    %

    Корень $r_{2.-1}$ представляется суммой $r_{2.-1}=r_{10}+r_{20}$ специальной
    пары корней однозначно, причем
    %
    \[ [e_{20},e_{10}]=2ce_{2,-1},\quad e_{20}e_{10}=ce_{2,-1}\quad (c=\pm 1),\]
    %
    \[ [e_{j0},e_{10}]=[[e_{j2},e_{20}], e_{10}]=[e_{j2},[e_{20},e_{10}]]=
        2ce_{j,-1},\ \ e_{j0}e_{10}=ce_{j,-1}.\]
    %
    Для корней $r_{i,-j}$ при $1< j<i$ также имеем
    %
    \[ [e_{i0},e_{j0}]=[e_{i0},[e_{j1},e_{10}]]=[e_{j1},[e_{i0},e_{10}]]=
        2ce_{i,-j},\ \ e_{i0}e_{j0}=ce_{i,-j}.\]
    %

    Корни $r_{i,-m}$ и $r_{jm}$ инцидентны в системе $\Psi_{m,j,i,k}$, однако, в ее
    подсистеме (типа $D_4$) длинных корней не инцидентны. Вместе с $r_{kj}$ они
    образуют базу подсистемы $\Psi=\Psi(r_{i,-m}, r_{jm}, r_{kj})$ типа $A_3$.
    Подалгебра $R(\Psi)$ в $R_{\Phi}$ здесь также стандартная; в противном случае
    нестандартный идеал с двумя углами $r_{i,-m}$ и $r_{ki}$ находим в $R(\Psi)$ и
    в алгебре $R_{\Phi}$. Поэтому $e_{i,-m}e_{jm}=0$ и $e_{jm}e_{i,-m}=e_{i,-j}$.

    Используя лемму \ref{l:stand-A3} и тождество Якоби алгебры Ли
    $N{\Phi}(\mathbb{K})$, получаем
    \[
        e_{i,-j}=[e_{jm},e_{i,-m}]=[e_{jm},[e_{ij},e_{j,-m}]]= [e_{j,-m},
        [e_{ij}, e_{jm}]]=[e_{j,-m},e_{im}],
    \]
    %
    \begin{equation}\label{RelatA3-RBn}
        e_{i,-j}=e_{jm}e_{i,-m}=[e_{jm},e_{i,-m}]=[e_{j,-m},e_{im}]=e_{j,-m}e_{im}.
    \end{equation}
    %

    Учитывая произвол в выборе подсистемы $\Psi_{m,j,i,k}\ (i<k\leq n)$, с
    точностью до выбора знака $c=\pm 1$, имеем $R_{\Phi}=RB_n(\mathbb{K})$ по
    модулю идеала $T_{n,-1}$.
\end{proof}

Согласно доказательству предложения 12.2.3 в \cite{carter72} (см. также
\cite{vl90a_ru}), алгебра Ли $N\Phi (\mathbb{K})$ допускает графовый
автоморфизм $\theta$, когда граф Кокстера системы корней $\Phi$ допускает
нетривиальную симметрию и все корни в $\Phi$ одной длины. Симметрия линейно
продолжается до перестановки \ $\bar {}$ \ системы корней $\Phi$.

\medskip

Симметрия \ $\bar {}$ \ системы корней $\Phi$ типа $D_{n}$ при $n>4$
единственна и графовый автоморфизм $\theta$ алгебры Ли $N\Phi(\mathbb{K})$
порядка 2 определяется правилом
\[
    \theta : \ e_r \to e_{\bar {r}}\quad (r \in \Phi^+)
\]
\begin{lemma}\label{l:RB_n-RDn+1}
    Алгебра $RB_n(\mathbb{K})$ представляется в алгебре $RD_{n+1}(\mathbb{K})$
    централизатором графового автоморфизма $\theta$ порядка 2 алгебры
    Ли $RD_{n+1}(\mathbb{K})^{(-)}$.
\end{lemma}

\begin{proof}
    В представлении алгебры Ли $N\Phi(\mathbb{K})$ типа $D_{n+1}$ алгеброй Ли
    $RD_{n+1}(\mathbb{K})^{(-)}$ графовый автоморфизм $\theta$ переставляет в
    $D_{n+1}^+$-матрицах 1-й и (-1)-й столбцы (то есть $\theta (e_{i
            v}) =e_{i, -v}$ при $v=\pm 1$), не изменяя остальные столбцы.

    Автоморфизм $\theta$ централизует (или оставляет неподвижными) все элементы
    \begin{equation}\label{IsomRBn-Dn}
        f_{i-1,0}: = e_{i,-1}+e_{i 1}\ \ {1<i\leq n+1},\quad f_{i-1,\pm
                (v-1)}: =e_{i, \pm v}\ \ (1< v<i\leq n).
    \end{equation}
    Очевидно, централизатор графового автоморфизма $\theta$ совпадает с подалгеброй
    \[
        C(\theta)=\sum_{i=2}^{n+1} \mathbb{K}(e_{i,-1}+e_{i 1})+\sum_{1<|v|<i\leq n+1} \mathbb{K}e_{iv}
    \]
    и является даже идеалом алгебры $RD_{n+1}(\mathbb{K})$.

    Выбранная нумерация элементов $f_{iv}$ позволяет считать их базой модуля
    $RB_n(\mathbb{K})$. Умножение в алгебре $RD_{n+1}(\mathbb{K})$, по лемме
    \ref{l:mult-RAn-RDn}, превращает модуль в алгебру $RB_n(\mathbb{K})$ с базой
    Шевалле из элементов $f_{iv}$.

    Требуемый изоморфизм $C(\theta)\to RB_n (\mathbb{K})$ укажем явным действием на
    произвольную $D_{n+1}^+$-матрицу $\alpha$ из $C(\theta)$, по аналогии со
    скручивающим автоморфизмом группы Шевалле типа $D_{n+1}$ перед леммой 4 в
    \cite{vl90a_ru}. Вначале "склеиваем" \ 1-ый и (-1)-й столбцы в $\alpha$, считая
    его 0-м, а остальные столбцы не изменяем. Заменяя в новой матрице номер $i$
    каждой строки на $i-1$, а номер $\pm (v-1)$ столбца элементов из них на номер
    $\pm v\ (1<v < i\leq n)$, получаем изоморфный образ элемента $\alpha$.
    Равенства (\ref{IsomRBn-Dn}) определяют обратное изоморфное вложение.
\end{proof}

\medskip

Лемма~\ref{l:RB_n-RDn+1} и замечание~\ref{remark:A3-D3} приводят к возрастающей
последовательности с однозначно определенными изоморфными вложениями алгебр:

\begin{equation}\label{InclusRBn-Dn}
    RB_{n-1}(\mathbb{K})\subset RD_n(\mathbb{K})\subset RB_n(\mathbb{K})\subset
    RD_{n+1}(\mathbb{K})\subset \cdots ,\quad n=3,4,5,\dots \ .
\end{equation}

\begin{definition}
    Последовательность обертывающих алгебр $R_n$ одного
    классического типа (скажем, $B_n$) возрастающих рангов $n$ назовем \emph{монотонной},
    если каждый ее член изоморфно представляется подалгеброй следующего члена.
\end{definition}

\medskip

Соответствующий пример с естественными изоморфными вложениями дает
%
\begin{equation}\label{InclusRBn}
    RB_3(\mathbb{K})\subset RB_4(\mathbb{K})\subset \cdots \subset RB_n(\mathbb{K})\subset
    RB_{n+1}(\mathbb{K}) \subset \cdots .
\end{equation}
%
Монотонность здесь, очевидно, нарушается (вместе с первым включением) при
замене $RB_3(\mathbb{K})$ на алгебру $R_{\Phi}$ типа $B_3$ из леммы
\ref{l:Subsistem-B3}.

\medskip

Сейчас мы можем указать условия однозначности обертывающих алгебр $R_{\Phi}$
классических типов. Непосредственно из леммы \ref{l:Subsistem-Bn} вытекает

\begin{theorem}\label{th:Stand-Bn}
    Если последовательность $R_n= R_{\Phi}$ стандартных обертывающих алгебр типа
    $B_n$ $(n=3, 4, \cdots )$ с условием \eqref{IsomR-Bn} монотонная, то $R_n=RB_n
        (\mathbb{K})$, с точностью до выбора знака $c=\pm 1$ произведения $e_{20} e_{10}=
        ce_{2,-1}$.
\end{theorem}

Отметим, что наряду с последовательностями стандартных обертывающих алгебр типа
$B_n$, в силу \eqref{InclusRBn-Dn}, определены и монотонные последовательности
обертывающих алгебр типа $D_n$.

        RC_n(\mathbb{K})\subset NT(2n,\mathbb{K})\subset RC_{n+1}(\mathbb{K})\subset

\end{remark}

\chapter{Проблемы перечисления идеалов нильтреугольных алгебр Шевалле}\label{ch:enumeration}

Ли $N\Phi(q) \coloneq N\Phi(GF(q))$ классических типов записаны в 2001 году
\cite[проблемы 1 и 2]{sigsam2001}.

\medskip

Согласно~\cite[\S~4]{JA-12}, стандартный идеал $H$ в $N\Phi(\mathbb{K})$
характеризуют множество $\mathcal{L}=\mathcal{L}(H)$ его углов и \emph{фрейм}
$\mathcal{F}(H)$, определяемый условиями
%
\[\mathcal{F} (H)\subset \sum_{s\in \mathcal{L}}\mathbb{K}e_{s},\quad \mathcal{F}(H)=H \mod
    Q(\mathcal{L}).\]
%
Таким образом, стандартность $H$ равносильна условию $H=Q(\mathcal{L}) +
    \mathcal{F}(H)$.

Известные перечисления (\cite[следствие 4.3]{vL74}, \cite[теорема
    2.1.2]{Egorychev77}, \cite{Eg-Lev-1996,eS05}) нормальных подгрупп групп
$U\Phi(\mathbb{K})$ над полем $\mathbb{K}$, инвариантных относительно
диагональных автоморфизмов, редуцируются к перечислению стандартных идеалов $H$
колец Ли $N\Phi(\mathbb{K})$ с фреймом $\mathcal{F}(H) = \sum_{s\in
        \mathcal{L}} \mathbb{K}e_{s}$, то есть вида $T(\mathcal{L})$. Тем самым, они
редуцируются к перечислению определенных путей в решетках, зависящих от выбора
типа алгебры $N\Phi(\mathbb{K})$.

\section{Координатно полные подпространства \texorpdfstring{$\mathbb{F}_q^m$}{F₍q₎ᵐ} и их перечисление}\label{s:coordinate-complete}
%
Пусть $\mathbb{K}^m$ "--- пространство строк длины $m$ над $\mathbb{K}$.
Подпространство $S$ в $\mathbb{K}^m$ назовем \emph{координатно полным}, если
оно не лежит ни в одной координатной гиперплоскости $H_i = \{x \in \mathbb{K}^m
    \mid x_i = 0\}$ $(1 \leq i \leq m)$. (Координатно полные подпространства
$\mathbb{F}_q^m$ также называют невырожденными линейными кодами в теории
кодирования, \cite[\S 1.1.2]{acodes2007}.)

\medskip

Число всех координатно полных $t$-мерных подпространств пространства
$\mathbb{F}_q^m$ обозначим через $V_q(m,t)$. В~\cite[лемма 2]{kriv-lev2015}
найдена формула
\[
    V_q(m,t) = \mathlarger{\sum}_{\raisebox{-0.6ex}{$\scriptstyle \mathclap{1=j_1<j_2<\ldots
                        < j_{t}\leq m}$}}  \ \frac{(q^t-1)^{m-j_{t}}}{(q-1)^{t-j_{t}}}\cdot
    \prod_{k=2}^{t-1}\left(\frac{q^k-1}{q-1}\right)^{\mathrlap{j_{k+1}-j_k-1}}
    .
\]
Применяя к ней разработанный \cite{Egorychev77} метод интегрального
представления комбинаторных сумм, включающих $q$-биномиальные коэффициенты
\[
    {n \brack k}_q \coloneq\prod_{i=0}^{k-1}\frac{1-q^{n-i}}{1-q^{i+1}},
\]
Г.\:П.~Егорычев установил \cite[леммы~3 и 4]{IzvIrkGU-2016} рекуррентное
соотношение
\begin{equation}\label{eq:vmt-rec-egorychev}
    V_q(m,t) = \sum_{k=0}^{mt} (q^t -1)^k \times
    V_q(m-1-k,t-1)
\end{equation}
и следующее утверждение.

\begin{lemma}[Г.\:П. Егорычев]
    Справедлива следующая формула
    \begin{equation}\label{eq:vmt-comb-egorychev}
        V_q(m,t) = \sum_{k=0}^{m-t} (-1)^{m-t-k} q^k \binom{m-1}{t+k-1}
        {t+k-1 \brack k}_{\mathrlap{q}}.
    \end{equation}
\end{lemma}

\medskip

В~\cite[заключение]{IzvIrkGU-2016} высказывалась потребность
алгебраически-комбинаторного доказательства формул \eqref{eq:vmt-rec-egorychev}
и \eqref{eq:vmt-comb-egorychev}. Доказана

\begin{lemma}[{{\cite[лемма 2.3]{smj2023}}}]\label{l:vmt-comb}
    %
    Справедлива формула
    \begin{equation}\label{eq:vmt-comb}
        V_q(m,t) = \sum_{k=t}^{m} (-1)^{m-k} \binom{m}{k}
        {k \brack t}_{\mathrlap{q}}.
    \end{equation}
\end{lemma}

\begin{proof}
    Докажем формулу \eqref{eq:vmt-comb} для числа $V_q(m, t)$ всех координатно
    полных $t$-мерных подпространств в $\mathbb{F}_q^m$.
    Н.\,Д. Ходюня получает рекуррентную формулу \eqref{eq:vmt-rec-egorychev}
    перенесением схемы перечисления канонических баз в лемме \ref{l:LieIdeal}.

    \medskip

    Известно \cite[предложение 1.3.18]{stanley-ec1-ru}, что число всех $t$-мерных
    подпространств в $\mathbb{F}_q^m$ равно ${m \brack t}_q$. Пусть $H_i $\ ($1\leq
        i \leq m$) "--- координатная гиперплоскость всех векторов в $\mathbb{F}_q^m$ c
    нулевой $i$-й координатой. Тогда любое подпространство в $H_i$ не является
    координатно полным в $\mathbb{F}_q^m$, а каждое подпространство в
    $\mathbb{F}_q^m$, не являющееся координатно полным, лежит хотя бы в одном
    подпространстве $H_i$.

    \medskip

    Обозначим через $\hat{H}_i$ множество всех $t$-мерных подпространств в $H_i$.
    Для любого набора попарно различных индексов $i_1, \ldots, i_k$ пересечение
    $H_{i_1}\cap\ldots\cap H_{i_k}$ состоит из векторов с нулевыми координатами
    $i_1,\ldots,i_k$ и потому изоморфно $\mathbb{F}_q^{m-k}$. Следовательно,
    $|\hat{H}_{i_1}\cap\ldots\cap\hat{H}_{i_k}| = {m-k\brack t}_q$, причем это
    значение зависит лишь от числа $k$, а не от выбора индексов. Так как число
    $k$-элементных подмножеств в $\{1,\ldots,m\}$ равно $\binom{m}{k}$, формула
    включений-исключений дает
    \begin{equation}\label{eq:vmt-comb-old}
        V_q(m,t) = \sum_{k=0}^{m-t} (-1)^k \binom{m}{k}
        {m-k \brack t}_{\mathrlap{q}}.
    \end{equation}
    %
    Подстановка $j=m-k$ в \eqref{eq:vmt-comb-old} завершает доказательство леммы.
\end{proof}

\begin{remark}
    К формуле \eqref{eq:vmt-comb-old} также приводят $q$-аналог ${n \brack k}_q =
        {n \brack n-k}_q$ известной комбинаторной формулы и подстановка в
    \eqref{eq:vmt-comb-egorychev} тождества
    \[
        q^k{t+k-1 \brack k}_q = {t+k \brack t}_q - {t+k-1 \brack t}_{\mathrlap{q}}.
    \]
    %Это дает решение задачи 1 из \cite{IzvIrkGU-2016}.
\end{remark}

\medskip

Отметим в заключение параграфа равенство чисел $V_q(n,s)$ числам $A_n^{(s)}$ из
\cite[формула~(9)]{carlitz1933}:
\[
    A_n^{(s)} = \sum_{k=s}^{n}(-1)^{n-k}\binom{n}{k}\qbinom{k}{s}.
\]
Числа $A_n^{(s)}$ возникли у Л.~Карлитца при перечислении подполей абелевых
расширений поля $\mathbb{Q}$. Пусть $K/\mathbb{Q}$ "--- расширение Галуа с
элементарной абелевой группой
$\mathrm{Gal}(K/\mathbb{Q})\simeq{(\mathbb{Z}/q\mathbb{Z})}^n$ ($q$ "---
простое), порождённой автоморфизмами $\sigma_1,\ldots,\sigma_n$ поля $K$.
Обозначим через $K_i$ неподвижное поле автоморфизма $\sigma_i$; тогда
$K_1,\ldots,K_n$ "--- все максимальные промежуточные поля расширения
$K/\mathbb{Q}$. Число $A_n^{(s)}$ подсчитывает те промежуточные поля $k$
($\mathbb{Q}\subset k \subset K$, $[k:\mathbb{Q}]=q^s$), которые не содержатся
ни в одном из подполей $K_i$.

\medskip

По соответствию Галуа промежуточное поле $k$ с $[k:\mathbb{Q}]=q^s$ отвечает
подгруппе $H$ группы $\mathrm{Gal}(K/\mathbb{Q})\simeq\mathbb{F}_q^n$ индекса
$q^s$; так как подгруппы элементарной абелевой $q$-группы суть подпространства,
$H$ "--- подпространство $\mathbb{F}_q^n$ коразмерности~$s$. Подполе $K_i$ "---
координатной гиперплоскости $\{x\in\mathbb{F}_q^n \mid x_i=0\}$. Поэтому
условие $k\not\subset K_i$ для всех $i$ равносильно тому, что $H$ не содержит
ни одного стандартного базисного вектора $e_i$. Переходя к ортогональному
дополнению $H^\perp$, получаем биекцию таких подпространств с $s$-мерными
координатно полными подпространствами в $\mathbb{F}_q^n$, откуда $A_n^{(s)} =
    V_q(n,s)$.

\section{Проблема перечисления стандартных идеалов} \label{sec:Problem-1}
%
Полное решение проблемы~1 из \cite{sigsam2001} дает следующая теорема.
\begin{theorem}[{{\cite[теорема 1]{smj2023}}}]\label{th:enum-stand-id-classical}
    Число стандартных идеалов алгебры Ли $N\Phi(q)$ классического типа
    $\Phi$ лиева ранга $n$ равно
    %
    \begin{equation*}
        1+\sum_{m=1}^n B(\Phi,m) \sum_{t=1}^m
        \sum_{k=0}^{m-t} (-1)^k \binom{m}{k} {m-k \brack t}_q,
    \end{equation*}
    где $B(\Phi,m)=\binom{n}{m}^2$ для типов $B_n$ и $C_n$, а также
    \[
        B(A_{n-1},m) = \frac{1}{n}\binom{n}{m}\binom{n}{m+1}, \quad B(D_n,m) =
        \binom{n}{m}\left( \binom{n-1}{m} + \binom{n-2}{m-2} \right).
    \]
\end{theorem}
\begin{proof}
    Ясно, что в алгебре Ли $N\Phi(\mathbb{K})$ любой стандартный идеал $H$ с
    $\mathcal{L}=\mathcal{L}(H)$ порядка $m$ записывается в виде
    \begin{equation}\label{eq:hls}
        H(\mathcal{L},S) = Q(\mathcal{L}) + \{ a_1e_{r_1} + a_2e_{r_2}
        + \dots + a_me_{r_m} \,|\, (a_1,a_2,\dots,a_m)\in S \}
    \end{equation}
    %
    для однозначно определенного координатно полного подпространства $S$ в
    $\mathbb{K}^m$.

    \medskip

    Как и в \S~\ref{s:coordinate-complete}, число всех координатно полных
    $t$-мерных подпространств пространства $\mathbb{K}^m$ при $\mathbb{K}=GF(q)$
    обозначим через $V_q(m,t)$, а через $B(\Phi,m)$ "--- число всех $m$-элементных
    множеств углов $\mathcal{L}$ в $\Phi^+$. Из биективности соответствия
    \eqref{eq:hls} между стандартными идеалами и парами $(\mathcal{L}, S)$ сразу же
    вытекает

    \medskip

    \begin{lemma}\label{l:enum-stand-id-reduct-BV}
        Число стандартных идеалов алгебры Ли $N\Phi(q)$ ранга $n$ равно
        \[
            1+\sum\limits_{m=1}^{n}B(\Phi,m)\, \ \sum\limits_{t=1}^{m}V_q(m,t).
        \]
    \end{lemma}

    В~\cite{vmj2015} задача \textbf{(A)} исследовалась для типа $A_n$. Числа
    $B(\Phi,m)=B(A_{n},m)$ вычислены ранее \cite[теорема~2.1.2]{Egorychev77} (см.
    также \cite{vL74} и \cite{Tolasov-1977}):
    \[
        B(A_{n-1},m) = \frac{1}{n}\binom{n}{m}\binom{n}{m+1}.
    \]

    \Cref{l:vmt-comb,l:enum-stand-id-reduct-BV} вместе с указанной
    формулой для числа $B(A_{n-1},m)$ завершают доказательство
    \crefRod{th:enum-stand-id-classical} для типа $A_n$.

    \medskip

    Числа $B(\Phi,m)$ исследовались для классических типов в различных ситуациях с
    использованием метода интегрального представления комбинаторных сумм
    (\cite{Egorychev77}, \cite{Eg-Lev-1996}, \cite{sigsam2001} и др.). Эти числа
    указаны явно в выписанном в \crefPred{th:enum-stand-id-classical} виде для
    типов $B_n$ и $C_n$ в~\cite[предложение~17]{reiner97}, а для типа $D_n$ "---
    в~\cite[теорема~1.2]{reiner04}.

    \medskip

    Учитывая \cref{l:vmt-comb,l:enum-stand-id-reduct-BV}, доказательство теоремы
    \ref{th:enum-stand-id-classical} завершено.
\end{proof}

\medskip

\section{Специальная каноническая база идеала}\label{sec:canonicalbase}

В связи с проблемой 2 из \cite{sigsam2001} перечисления всех идеалов алгебр Ли
$N\Phi(q)$ классических типов, в \cite[\S\ 4]{vmj2015} рассматривались
специальные канонические базы лиевых идеалов алгебр $NT(n,\mathbb{K})$.

\medskip

Аналогичные базы подалгебр в алгебрах Ли $N\Phi(\mathbb{K})$ выявляет

\begin{lemma}\label{l:LieIdeal}
    Пусть $H$ "--- ненулевая подалгебра алгебры Ли $N\Phi(\mathbb{K})$ над полем
    $\mathbb{K}$ и $\mathcal{L} = \mathcal{L}(H)= \{r_1, r_2, \dots, r_m\}$. Тогда
    любую базу пересечения $Q(\mathcal{L})\cap H$ можно дополнить до базы
    в $H$ элементами

    \begin{equation}\label{eq:base1'}
        \alpha_i=  \sum_{j=1}^m a_{ij}e_{r_j}+\alpha_i' ,\quad
        i=1,2,\dots, t,\quad  t= \dim_K\ H / H \cap Q(\mathcal{L}),
    \end{equation}

    где $||a_{ij}||$ "--- $t\times m$-матрица ранга $t$ над $\mathbb{K}$\ $(t \leq
        m)$ и $\alpha_i' \in Q(\mathcal{L})$. Кроме того, при фиксированном
    упорядочении $\mathcal{L}$ для однозначно определенных номеров
    $j_1=1<j_2<\cdots <j_t \leq m$ можно считать

    \begin{equation}\label{eq:base-1}
        a_{i,j_i}=1\ (1\leq i\leq t),\quad a_{ik}=0\ (k<j_i),\quad
        a_{i,j_k}=0,\ 1\leq i<k\leq t.
    \end{equation}
\end{lemma}

\begin{proof} Ясно, что $Q(\mathcal{L})+H=Q(\mathcal{L})+\mathcal{F}(H)$
    есть стандартный идеал в $N\Phi(\mathbb{K})$, имеющий вид $H(\mathcal{L},S)$
    из \eqref{eq:hls}. Элементы $\alpha_i\in H$ и равенства
    \eqref{eq:base-1} получаем, используя известный изоморфизм
    \[
        (H+Q(\mathcal{L})) / Q(\mathcal{L}) \simeq H / H \cap Q(\mathcal{L}).
    \]

    Уточним матрицу $||a_{ij}||$ из \eqref{eq:base1'}. Очевидно, базисный элемент
    $\alpha_1\in H$ всегда можно выбрать так, что $a_{11}=1$; положим $j_1=1$ и
    обозначим через $H_1$ подалгебру элементов из $H$ с нулевым коэффициентом при
    $e_{r_1}$. Если $\alpha_i$, $j_i$ и $H_i$ уже определены при $1\leq i<t$, то
    элемент $\alpha_{i+1}\in H_i$ выбираем с наименьшим возможным номером $j_{i+1}$
    первой ненулевой координаты; можно считать $a_{i+1,k}=1$ при $k=j_{i+1}$. Через
    $H_{i+1}$ обозначаем подалгебру элементов из $H_i$ с нулевым коэффициентом при
    $e_{r_k}, \ k=j_{i+1}$.

    Продолжая аналогично, на $t$-м шаге выбираем в $H_{t-1}$ элемент $\alpha_{t}$ с
    коэффициентом $a_{t,j_t}=1$ и $H_{t}=H \cap Q(\mathcal{L})$. Элементарные
    преобразования дают оставшиеся в~\eqref{eq:base-1} равенства $a_{i,j_k}=0$
    последовательно для $i=1,2, \cdots , t-1$ и $i<k\leq t$. Это завершает
    доказательство леммы.
\end{proof}

Уточним базу \eqref{eq:base1'} для нестандартного идеала $H$ обертывающей
алгебры $RD_n(\mathbb{K})$. По теореме \ref{th:classical-Exc-RDn}, $Q(r)\subset
    H$ для всех его углов $r$, кроме двух симметричных углов $s$ и $\bar{s}$.
Простой корень $p \neq \bar{p}$, инцидентный с $s$, и простой корень $p_0=
    \bar{p}_0$ такой, что $p+p_0\in\Phi^+$, определены однозначно, причем $T(s_0 +
    \bar{p}) \subset H$, где полагаем $s_0=s+p_0$ при $s=p$ и $s_0=s$ при $s\neq
    p$.

\medskip

Если неинцидентные с $s$ и $\bar{s}$ простые корни $p_j= \bar{p}_j$ выберем
так, что
%
\begin{equation}\label{eq:r-stepsId}
    s_i=s+p_1+p_2+\cdots + p_i\in \Phi^+, \quad 1\leq i< n-\height(s),
\end{equation}
%
то $Q(s_{i})\subset H$ при $i=n-1-\height(s)$. Поэтому существует наименьший
номер $\mathbb{K}$ с условиями $Q(s_k) \subset H$ и $1\leq k\leq
    n-1-\height(s)$. Записывая $\mathcal{L}$ в виде
\begin{equation}\label{eq:R-stepsId}
    \mathcal{L}=\{r_1=s, r_2=\bar{s}, r_3, \dots, r_m\}\quad (2\leq
    m=|\mathcal{L}| <n),
\end{equation}
при $3\leq j\leq m$ имеем $\bar{r}_j= r_j$. Сопоставим $H$, по лемме
\ref{l:LieIdeal}, $t\times m$-матрицу $||a_{ij}||$ над $\mathbb{K}$ ранга $t$\ $(1\leq t
    < m)$ с условиями \eqref{eq:base-1}. Тогда $a_{12}=c\neq 0$ и
\[
    H \cap Q(\mathcal{L}) =T(s_0 + \bar{p}) + Q(s_k) + Q(\bar{s}_k)+
    \sum_{j=1}^k \mathbb{K}(e_{s_j} + ce_{\bar{s}_j})+ \sum_{j=3}^m Q(r_j)
\]
(последнее слагаемое при $m=2$ отбрасываем). Упрощая в
\eqref{eq:base1'} элементы $\alpha_i$ по модулю $H \cap Q(\mathcal{L})$,
находим $d_i\in K$ такие, что $\alpha_i'= d_ie_{s_k}, \
    i=1,2, \dots, t.$

\medskip

Таким образом, идеал $H$ представляется в виде

\begin{equation}\label{eq:IdH0}
    Id\ \{ \mathcal{L}, k, t, ||a_{ij}||,\ c, d_1, \dots d_t \}\ \coloneq
    \sum_{i=1}^t \ \mathbb{K}(d_ie_{s_k}+\sum_{j=1}^m a_{ij}e_{r_j})+
\end{equation}
\[
    +T(s_0 + \bar{p})+ Q(s_k)+ Q(\bar{s}_k)+
    \sum_{j=1}^k \mathbb{K}(e_{s_j} + ce_{\bar{s}_j})+\sum_{j=3}^m Q(r_j),
\]
где корни $s_i$ определены в \eqref{eq:r-stepsId}. Его однозначно характеризуют
множество углов $\mathcal{L}=\mathcal{L}(H)$ вида \eqref{eq:R-stepsId},
параметры $k, t$ с условиями
\[
    1\leq k\leq n-1-\height(s),\quad 1\leq t= \dim_K\ H / H \cap Q(\mathcal{L}) <|\mathcal{L}|=m <n,
\]
элементы $c\neq 0$, $d_1, \dots , d_t$ из $\mathbb{K}$ и $t\times m$-матрица
$||a_{ij}||$ над $\mathbb{K}$ ранга $t$\ с условиями $a_{12}=c$ и
\eqref{eq:base-1}. Тем самым, доказана

\begin{theorem}\label{th:ExIdForm}
    В обертывающей алгебре $RD_n(\mathbb{K})$ всякий нестандартный идеал $H$
    имеет множество углов $\mathcal{L}=\mathcal{L}(H)$ вида
    \eqref{eq:R-stepsId} и представляется как идеал \eqref{eq:IdH0},
    однозначно характеризуемый набором $\mathcal{L}, k, t, ||a_{ij}||,\
        c, d_1, \dots d_t$.
\end{theorem}

\section{Метод коэффициентов и перечисление идеалов алгебры \texorpdfstring{$RD_n(q)$}{RDₙ(q)}}\label{sec:egorychev-RDn}
%
\Cref{th:enum-stand-id-classical}, решая проблему 1 из \cite{sigsam2001}
о числе всех стандартных идеалов алгебры Ли $N\Phi(q)$ классического типа над
полем $K=GF(q)$, дает и число всех идеалов ее обертывающей алгебры
$R_{\Phi}(q)$, когда она стандартна.

В силу теорем \ref{th:enum-stand-id-classical} и \ref{th:classical-Exc-RDn},
перечисление идеалов любой стандартной обёртывающей алгебры из
предложения~\ref{p:mult-enveloping} классических типов $A_n$, $B_n$ и $C_n$
завершено. Остается перечислить нестандартные идеалы обёртывающей алгебры
$RD_n(q)$.

\medskip

Основной в этом параграфе является

\begin{theorem}\label{th:Enum-Dn}
    Число нестандартных идеалов алгебры $RD_n(q)$ равно
    \[
        \sum_{m=2}^{n-1} \binom{n-2}{m-2}\binom{n-1}{m}
        \sum_{t=1}^{m-1}q^t(q-1)
        \sum_{j=0}^{m-1-t} (-1)^j \binom{m-1}{j} {m-1-j \brack t}_q.
    \]
\end{theorem}

\begin{proof}
    Нестандартные идеалы алгебры $RD_n(q)$, по теореме~\ref{th:ExIdForm},
    исчерпываются идеалами вида~\eqref{eq:IdH0},
    которые характеризуются однозначно набором $\mathcal{L}, k, t,
        ||a_{ij}||,\ c, d_1, \dots d_t$. Записывая $\mathcal{L}$ как в
    \eqref{eq:R-stepsId} в матричном представлении, считаем
    \[
        e_{s}=e_{i1},\quad e_{\bar{s}}=e_{i,-1},\quad e_{s_k}=e_{i+k,1}\quad (2\leq i< n-k).
    \]
    Через $B(m,n,i,k)$ обозначим число всевозможных $(m+2)$-элементных множеств
    углов вида \eqref{eq:R-stepsId}, у которых два угла соответствуют базисным
    элементам $e_{s}$ и $e_{\bar{s}}$ и нет углов в строках с номерами $j$, $i < j
        \leq k$. Нам потребуется

    \begin{lemma}\label{l:rdn}
        Число нестандартных идеалов алгебры $RD_n(q)$ равно
        \begin{equation}\label{eq:enum-Dn-1}
            \sum_{m=2}^{n-1} \sum_{i=2}^{n-1} \sum_{k=1}^{n-i}
            B(m,n,i,k)\sum_{t=1}^{m-1}q^t(q-1)V(m-1,t,q).
        \end{equation}
    \end{lemma}

    \begin{proof}
        При фиксированных $|\mathcal{L}|= m$ и $\mathbb{K}$ число способов выбора
        параметра $c\neq 0$ в $K=GF(q)$ и наборов $(d_1, \dots , d_t)$
        над $\mathbb{K}$ равно $q^t(q-1)$. Согласно доказательству леммы
        \ref{l:LieIdeal}, число возможностей выбора матриц $||a_{ij}||$,
        требуемых для канонических базисов идеалов, совпадает с
        $V(m-1,t,q)$ при любых $t,m$. Отсюда для числа
        нестандартных идеалов алгебры $RD_n(q)$ получаем формулу
        \eqref{eq:enum-Dn-1}.
    \end{proof}

    Второе и третье суммирования в комбинаторной сумме~\eqref{eq:enum-Dn-1}
    проведем с помощью метода коэффициентов. Напомним некоторые понятия.

    Пусть $L$ "--- множество формальных степенных рядов Лорана над полем
    $\mathbb{C}$, содержащих конечное число членов с отрицательными степенями. По
    определению, целое число $\mathbb{K}$ есть \emph{порядок} монома $c_kw^k\ \neq
        0$ и $L_k$ "--- множество рядов Лорана из $L$, у которых наименьший порядок
    мономов равен $\mathbb{K}$. Под \emph{оператором формального вычета} ряда $C(w)
        = \sum_{k}c_kw^k$ из $L$ полагают
    \[
        \res_wC(w) \coloneq c_{-1}.
    \]

    Оператор формального вычета $\res$ (контурный интеграл) имеет ряд свойств (см.,
    например, \cite{Egorychev77}). Из них наиболее часто используются следующие
    два.

    \medskip

    \textbf{Правило линейности.} Для любых $A(w)$ и $B(w)$ из $L$ и
    $a,b \in \mathbb{C}$ имеем
    \[
        a\res_wA(w) + b\res_wB(w) = \res_w\{aA(w) + bB(w)\}.
    \]

    \textbf{Правило подстановки.} Если либо $f \in L_m \
        (m=1,2,\dots)$ и $A(w) \in L$, либо $A(w)$ "--- полином и $f(w)\in
        L$, то
    %
    \[\sum_k (f(w))^k\res_z\{A(z)z^{-k-1}\} = A(f(w)).\]

    Если $C(w)=\sum_kc_kw^k$ из $L$ есть производящая функция для
    последовательности $\{c_k\}$, то $c_k=\res_wC(w)w^{-k-1}$ для любого
    $\mathbb{K}$. Когда $n,k = 0,1,2,\dots$, приходим к биномиальным коэффициентам
    %
    \begin{equation}\label{integral_formula}
        \binom{n}{k} = \res_w\{\frac{(1+w)^n}{w^{k+1}}\} = \frac{1}{2\pi
            i}\int_{\lvert
            w \rvert=\theta} \frac{(1+w)^n}{w^{k+1}}\,dw,\ \theta > 0,
    \end{equation}
    \[
        \binom{-n}{k} \coloneq \binom{n+k-1}{k} =
        \res_w\{\frac{(1-w)^{-n}}{w^{k+1}}\} = \frac{1}{2\pi
            i}\int_{\lvert
            w \rvert=\theta} \frac{(1-w)^{-n}}{w^{k+1}}\,dw,\ 0 < \theta < 1.
    \]

    \medskip

    В доказательстве теоремы~\ref{th:Enum-Dn} основной является
    %
    \begin{lemma}\label{l:B'-Dn}
        Справедлива следующая формула суммирования
        \[
            \sum_{i=2}^{n-1} \sum_{k=1}^{n-i} B(m,n,i,k) =
            \binom{n-2}{m-2}\binom{n-1}{m}.
        \]
    \end{lemma}

    \begin{proof}
        Обозначим через $B^{(-)}(u,l)$ число $l$-элементных множеств углов
        в $D_n^+$-матрице таких, что последний угол расположен левее
        $(-1)$-го столбца и выше $(u+3)$-й строки, а через
        $B^{(+)}(u,n-u-1,l)$ "--- число $l$-элементных множеств углов таких,
        что первый угол расположен правее $1$-го столбца и ниже $(n-u)$-й
        строки. В этих обозначениях
        %
        \begin{equation}\label{Bmnik}
            B(m,n,i,k)= \sum_{l=0}^m B^{(-)}(i-3,l)
            B^{(+)}(n-i-k,i+k-1,m-2-l).
        \end{equation}
        %
        Применяя \cite[Теорема~10.14.1]{Krattenthaler2015}, находим

        \begin{equation}
            B^{(+)}(u,v,l) =
            \binom{u+v}{l}\binom{u}{l} - \binom{u+v-1}{l-1}\binom{u+1}{l+1}.
        \end{equation}
        Несложная индукция по $v$ дает также равенство
        \begin{eqnarray}\label{bminus}
            B^{(-)}(u,v) = \binom{u+1}{2v}.
        \end{eqnarray}

        В силу~\eqref{Bmnik}-\eqref{bminus}, второе и третье суммирования
        в~\eqref{eq:enum-Dn-1} приводят к 3-кратной комбинаторной сумме с биномиальными
        коэффициентами. При этом, как обычно, вычисления проводим с помощью свойств
        оператора $\res$, путем последовательного нахождения интегрального
        представления рассматриваемых выражений и вычисления полученных интегралов. В
        соответствии с \cite[теорема 10.14.1]{Krattenthaler2015} находим представление
        чисел $B^+(u,v,l)$. Пользуясь интегральной формулой~\eqref{integral_formula},
        для каждого из биномиальных коэффициентов находим
        \begin{gather*}
            B^{(+)}(u,v,l) =
            \binom{u+v}{l}\binom{u}{l} - \binom{u+v-1}{l-1}\binom{u+1}{l+1} = \\
            = \res_{z,w}\left\{ \frac{(1+z)^{u+v}}{z^{l+1}}\frac{(1+w)^u}{w^{l+1}}
            \right\}
            - \res_{z,w}\left\{ \frac{(1+z)^{u+v-1}}{z^l}\frac{(1+w)^{u+1}}{w^{(l+1)+1}}
            \right\} = \\
            = \res_{z,w}\left\{ \frac{(1+z)^{u+v-1}(1+w)^u}{z^{l+2}}(w-z) \right\}
        \end{gather*}
        и, тем самым,
        \begin{equation}\label{integral_formula_bplus}
            \begin{gathered}
                B^{(+)}(n-i-k, i+k-1, m - 2 - l) \coloneq \\
                = \res_{z,w}\left\{ \frac{(1+z)^{n-2}(1+w)^{n-i-k}}{z^{(m-2-l) +
                            1}w^{(m-1-l)
                            + 1}}(w-z) \right\}.
            \end{gathered}
        \end{equation}

        Отсюда, в силу \eqref{Bmnik}-\eqref{bminus}, получаем
        \begin{gather*}
            S \coloneq \sum_{i=2}^{n-1}\sum_{k=1}^{n-i}B(m,n,i,k) = \\
            = \sum_{i=2}^{n-1}\sum_{k=1}^{n-i}\sum_{l=0}^{m}B^{(-)}(i-3, l)
            B^{(+)}(n-i-k, i+k-1, m-2-l) =
        \end{gather*}
        (занесение знака суммы за знак $\res_{z,w}$ и добавление нулевых
        слагаемых)
        \begin{gather*}
            \sum_{i=2}^{n-1}\sum_{k=1}^{n-i}\res_{z,w}\left\{
            \frac{(1+z)^{n-2}(1+w)^{n-i-k}(w-z)}{z^{(m-2)+1}w^{(m-1)+1}}
            \times \left[ \sum_{l=0}^{m}(zw)^l \binom{i-2}{2l} \right]\right\}=
        \end{gather*}
        (суммирование в квадратных скобках и формула геометрической
        прогрессии)
        \begin{gather*}
            \sum_{i=2}^{n-1}\sum_{k=1}^{n-i}\res_{z,w}\left\{
            \frac{(1+z)^{n-2}(1+w)^{n-i-k}(w-z)}{z^{(m-2)+1}w^{(m-1)+1}}
            \times \frac{(1 + \sqrt{zw})^{i-2} + (1 - \sqrt{zw})^{i-2}}{2}
            \right\}
            = \\
            \frac{1}{2}\res_{z,w} \{
            %
            \frac{(1+z)^{n-2}(1+w)^{n-2}(w-z)}{z^{(m-2)+1}w^{(m-1)+1}} \times \\
            \left[ \sum_{i=2}^{\infty} \left(
                \frac{1+\sqrt{zw}}{1+w}\right)^{i-2} + \sum_{i=2}^{\infty} \left(
                \frac{1+\sqrt{zw}}{1+w}\right)^{i-2} \right] \times
            \sum_{k=1}^{\infty}\left(\frac{1}{1+w}\right)^k
            %
            \} =
        \end{gather*}
        ($\lvert w\rvert \gg 1$, $\lvert (1 + \sqrt{zw})/(1+w)\rvert < 1$,
        $\lvert (1-\sqrt{zw})/(1+w) \rvert < 1$, суммирование по индексам
        $i$ и $\mathbb{K}$, формула геометрической прогрессии)
        \begin{gather*}
            \frac{1}{2}\res_{z,w}\{
            \frac{(1+z)^{n-2}(1+w)^{n-2}(w-z)}{z^{(m-2)+1}w^{(m-1)+1}} \times \\
            \left[ 1/(1 - \frac{1+ \sqrt{zw}}{1+w}) + 1 / (1 - \frac{1 -
                    \sqrt{zw}}{1+w}) \right] \times \frac{1}{1+w} \times \frac{1}{1 -
                \frac{1}{1+w}}
            \} = \\
            \frac{1}{2}\res_{z,w}\left\{
            \frac{(1+z)^{n-2}(1+w)^{n-1}(w-z)}{z^{(m-2)+1}w^{(m-1)+1}} \times
            \left[ \frac{1}{w-\sqrt{zw}} + \frac{1}{w + \sqrt{zw}} \right]
            \times \frac{1}{w}
            \right\} = \\
            \frac{1}{2}\res_{z,w}\left\{
            \frac{(1+z)^{n-2}(1+w)^{n-1}(w-z)}{z^{(m-2)+1}w^{(m-1)+1}} \times
            \frac{2w}{w^2-zw} \times \frac{1}{w}
            \right\} = \\
            \res_{z,w}\left\{
            \frac{(1+z)^{n-2}(1+w)^{n-1}}{z^{(m-2)+1}w^{m+1}} \right\} =
            \\
            \res_{z,w}\left\{ \frac{(1+z)^{n-2}}{z^{(m-2)+1}} \right\} \times
            \res_{z,w}\left\{ \frac{(1+w)^{n-1}}{w^{m+1}} \right\} \coloneq
            \binom{n-2}{m-2}\binom{n-1}{m}.
        \end{gather*}
        Тем самым, доказательство \crefRod{l:B'-Dn} завершается.
    \end{proof}

    Применяя к \crefDat{l:rdn} \cref{l:vmt-comb,l:B'-Dn} получаем утверждение
    \crefRod{th:Enum-Dn}.
\end{proof}

\begin{remark}\label{remark:problem-2}
    В отличие от идеалов алгебры $RD_n(\mathbb{K})$, для идеалов $H$ алгебры Ли
    $N\Phi(\mathbb{K})$ классического типа возрастает число пар $p$-связанных
    углов $r,s$ в $H$.
    Решение проблемы 2 из \cite{sigsam2001} сложнее даже
    для типа $A_n$; с возрастанием ранга $\Phi$
    может неограниченно возрастать и число углов, $p$-связанных не с
    одним, а с двумя углами в $H$.
\end{remark}

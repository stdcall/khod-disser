\chapter{Точные обертывающие алгебры нильтреугольных алгебр Шевалле}\label{ch:enveloping}

\section{Точные обертывающие алгебры и стандартные идеалы}\label{sec:enveloping}
%
В теории Киллинга"--~Картана каждой простой комплексной конечномерной алгебре
Ли сопоставляют единственную (с точностью до эквивалентности) неразложимую
систему корней $\Phi$ \cite[глава~3]{carter72}; выбор множества простых корней
$\Pi$ в $\Phi$ определяет систему положительных корней $\Phi^+\supset\Pi$.
Элементы $e_r$ $(r\in\Phi)$ и подходящий базис подалгебры Картана задают
\cite{Chev-55} базис Шевалле с целочисленными структурными константами и
определяют \emph{алгебру Шевалле} $\mathfrak{L}(\Phi,\mathbb{K})$ над любым
полем $\mathbb{K}$. Структурные константы базиса Шевалле определяет их выбор
для \emph{нильтреугольной} подалгебры $N\Phi(\mathbb{K})$ с базисом $\{e_r\mid
r\in\Phi^+\}$ \cite[\S~4.2]{carter72}.

Согласно теореме Пуанкаре"--~Биркгофа"--~Витта каждая алгебра Ли может быть
получена при помощи выбора подходящей ассоциативной обертывающей алгебры,
замены в ней операции на коммутирование и дальнейшего перехода к подходящей
подалгебре Ли, см. \cite[\S\ 17]{humphreys-ru}, \cite[\S~V.9]{Kurosh1973}.
Более трудная теорема Адо"--~Ивасавы (принадлежащая Адо для характеристики $0$
и Ивасаве для характеристики $p$, \cite[глава VI]{jacobson-ru}) устанавливает,
что конечномерные алгебры Ли могут быть получены таким способом из
конечномерных ассоциативных алгебр.

В случае, когда алгебра Ли $\mathfrak{L}(\Phi, \mathbb{K})$ является простой,
присоединенное представление $\ad \colon \mathfrak{L}(\Phi, \mathbb{K})
\hookrightarrow \mathfrak{gl}(\mathfrak{L}(\Phi, \mathbb{K}))$ является точным
и ассоциативная алгебра, порожденная образом $N\Phi(\mathbb{K})$, дает ее
конечномерную обертывающую алгебру.

\medskip

Для алгебры $R$ через $R^{(-)}$ обозначаем алгебру на том же линейном
пространстве с умножением $[x,y]\coloneq xy-yx$.

\begin{definition}[\cite{DAN2018}]
    Алгебра $R$ называется \emph{точной обертывающей} алгебры Ли $\mathfrak{g}$,
    если $\mathfrak{g}$ и $R^{(-)}$ изоморфны.
\end{definition}
%
Алгебру Ли $\mathfrak{g}$ и ее точную обертывающую алгебру $R$ можно определять
структурными константами в одном базисе, в отличие от универсальной
ассоциативной обертывающей алгебры.

\medskip

По теореме Шевалле о базисе \cite[теорема 4.2.1]{carter72} при $r,s\in \Phi$
имеем
\begin{equation*}%\label{eq:chevalley-basis}
    [e_r ,e_s] = N_{rs}e_{r+s}=-[e_s ,e_r] \ \ (r+s\in \Phi),
    \qquad [e_r,e_s]=0 \  \  (r+s\notin \Phi \cup \{0\}),
\end{equation*}
%
где либо $N_{rs}=\pm 1$, либо $|r|=|s|<|r+s|$ и $N_{rs}=\pm 2$, либо
$N_{rs}=\pm 2$ или $\pm 3$ и $\Phi$ типа $G_2$. Согласно \cite[предложение
4.2.2]{carter72}, знаки структурных констант $N_{rs}$ имеют определенный
произвол для алгебры Шевалле, а также для $N\Phi(\mathbb{K})$.

\medskip

В \cite{DAN2018} были построены специальные семейства точных обертывающих
алгебр для нильтреугольных алгебр $N\Phi(\mathbb{K})$.

\begin{definition}\label{def:RPhi}
    Пусть фиксирован выбор знаков структурных констант $N_{rs}$.
    $\mathbb{K}$-алгебру $R_{\Phi}$ с базисом $\{e_r \mid r \in \Phi^+\}$ и
    умножением: $e_r e_s = 0$ при $r + s \notin \Phi$ и
    \[
        e_r e_s = e_{r+s}, \qquad e_s e_r = (1 - N_{rs}) e_{r+s}
        \qquad (r, s, r+s \in \Phi^+,\ N_{rs} \geq 1)
    \]
    будем называть \emph{специальной обертывающей алгеброй} алгебры Ли
    $N\Phi(\mathbb{K})$.
\end{definition}
Заметим, что структурные константы $N_{rs}$ в базисе Шевалле являются целыми
числами. При работе над полем $\mathbb{K}$ мы отождествляем $N_{rs}$ с его
образом $N_{rs}\cdot 1_{\mathbb{K}}\in \mathbb{K}$. Поэтому условия и сравнения
вида $N_{rs}\geq 1$ следует понимать как утверждения о соответствующем целом
числе $N_{rs}$ (до редукции по характеристике), а коэффициенты в формулах
рассматриваются в поле $\mathbb{K}$.

\medskip

Зафиксируем систему корней $\Phi$ и продолжим произвольный линейный порядок на
$\Pi$ до линейного порядка~$\prec$ на $\Phi^+$, согласованного с высотой
корней. Когда $r,s,r+s\in\Phi^+$ и $0\prec r \prec s$, пару корней $r, s$
называют \emph{специальной}, а также \emph{экстраспециальной}, если $r
\preccurlyeq r_1$ для каждой специальной пары $r_1, s_1$ с суммой $r_1+s_1
=r+s$.

Верхнюю оценку числа специальных обертывающих алгебр $R_{\Phi}$ и их связь с
алгеброй Ли $N\Phi(\mathbb{K})$ выявляет

\begin{proposition}\label{p:mult-enveloping}
    %<*p:mult-enveloping>
    Каждая специальная обертывающая алгебра $R_{\Phi}$ является точной обертывающей
    алгебры Ли $N\Phi(\mathbb{K})$, и число таких алгебр не превосходит $2^{|\Phi^+
        \setminus \Pi|}$.
    %</p:mult-enveloping>
\end{proposition}

\begin{proof}
    Таблицы умножения базисных элементов $N\Phi(\mathbb{K})$ и любой алгебры
    $R_{\Phi}^{(-)}$, очевидно, совпадают. Поэтому, как и в {\cite[предложение
        1]{DAN2018}}, алгебра $R_{\Phi}$ есть точная обертывающая алгебры Ли
    $N\Phi(\mathbb{K})$.

    Согласно~\cite[предложение 4.2.2]{carter72}, знаки констант $N_{rs}$ можно
    выбрать для экстраспециальных пар $(r,s)$ произвольно, с точностью до
    изоморфизмов алгебры Ли $\mathfrak{L} (\Phi, \mathbb{K})$ (аналогично,
    $N\Phi(\mathbb{K})$); тогда знаки остальных констант $N_{rs}$ определяются
    однозначно.

    Любую из построенных специальных обертывающих алгебр $R_{\Phi}$ алгебры Ли
    $N\Phi(\mathbb{K})$ однозначно определяет выбор знаков структурных констант
    $N_{rs}$ для экстраспециальных пар корней $(r,s)$ в $\Phi^+$.

    С другой стороны, любой положительный непростой корень в $\Phi$ есть сумма
    $r+s$ единственной экстраспециальной пары корней $r,s$. Поэтому множество
    экстраспециальных пар взаимно однозначно соответствует множеству $\Phi^+
    \setminus \Pi$.

    Отсюда вытекает требуемая в предложении оценка числа алгебр $R_{\Phi}$.
\end{proof}

\begin{lemma}\label{l:Const-RPhi-Non-assoc}
    %<*l:Const-RPhi-Non-assoc>
    Все ненулевые структурные константы алгебр $R_{\Phi}$ в выбранном базисе равны
    $\pm 1$ для типа, отличного от $G_2$, и равны $1$, когда все корни в $\Phi$
    одной длины. Если $\Phi$ имеет корни разных длин, то все алгебры $R_{\Phi}$
    неассоциативные.
    %</l:Const-RPhi-Non-assoc>
\end{lemma}

\begin{proof}
    Когда система корней $\Phi$ имеет корни разных длин, найдутся корни $r, s \in
    \Phi^+$, для которых пересечение $\Phi\cap (\mathbb{Z}r+ \mathbb{Z}s)=\Psi$
    есть подсистема корней типа $B_2$ или $G_2$. Определение алгебры $R_{\Phi}$
    позволяет считать, что $\{r, s \}$ есть базис в $\Psi$ и $2r+s$ "--- корень.
    Если $N_{rs} \geq 1$, то получаем
    \[
        e_r(e_re_s) =e_re_{r+s} = \pm e_{2r+s},\quad (e_re_r)e_s=0.
    \]
    Аналогично при $N_{sr}\ (=- N_{rs})\ \geq 1$ произведения $(e_se_r)e_r =e_{r+s}
    e_r = \pm e_{2r+s}$ и $e_s(e_re_r)=0$ не совпадают. Следовательно, алгебра
    $R_{\Phi}$ "--- неассоциативная.

    \medskip

    Первое утверждение в лемме сейчас сразу следует из определения алгебр
    $R_{\Phi}$. Таким образом, доказательство завершено.
\end{proof}

Для классических типов $\Phi$ мы исследуем (\S~\ref{sec:uniqtheorems}) условия
однозначности алгебр $R_{\Phi}$ и, взаимосвязано, структурных констант $N_{rs}$
алгебр Ли $N\Phi(\mathbb{K})$ над полем~$\mathbb{K}$.

\medskip

Идеал произвольного кольца $R$ всегда есть идеал и кольца $R^{(-)}$, так как
основные операции в $R^{(-)}$ производны от операций в $R$. Верно и обратное,
когда умножение в $R$ есть также производная операция от операций в $R^{(-)}$.
Например, при $2\in\mathbb{K}^*$ для алгебры Ли $\mathfrak{g}$ с умножением
$[,]$ выберем точную обертывающую алгебру $R$ с умножением $\alpha \cdot \beta
= \frac{1}{2}[\alpha, \beta]$.

\medskip

Как и во введении, для корней $r,s\in\Phi^+$ полагаем $s\geq r$, если в
разложении $s-r$ по базису $\Pi$ все коэффициенты неотрицательны. Корни
$r,s\in\Phi^+$ называем \emph{инцидентными}, если $r\geq s$ или $s\geq r$.
Множество попарно неинцидентных корней в $\Phi^+$ называем \emph{антицепью}.

Для антицепи $\mathcal{L}$ в $\Phi^+$ выделим в $N\Phi(\mathbb{K})$ идеалы
\[
    T(r)\coloneq\sum_{s\geq r}\mathbb{K}e_s,\quad
    Q(r)\coloneq\sum_{s>r}\mathbb{K}e_s\quad(r\in\Phi^+),\quad
    Q(\mathcal{L})\coloneq\sum_{r\in\mathcal{L}}Q(r).
\]
Если $H\subset T(\mathcal{L})\coloneq\sum_{r\in\mathcal{L}}T(r)$ и включение
нарушается при любой замене $T(r)$ на $Q(r)$ в сумме, то множество
$\mathcal{L}=\mathcal{L}(H)$ определено однозначно и называется
\emph{множеством углов в $H$} \cite[\S~6]{Lev1992}.

Проиллюстрируем множества углов и идеалы $T(r)$, $Q(r)$, $T(\mathcal{L})$,
$Q(\mathcal{L})$ на примере алгебры Ли $NT(n,\mathbb{K})^{(-)}$ типа $A_{n-1}$.
Ее базис Шевалле составляют матричные единицы $e_{ij}$, соответствующие корням
$r_{ij}=\epsilon_i-\epsilon_j$ $(1\leq j<i\leq n)$. Условие $r_{kl}\geq r_{ij}$
равносильно неравенствам $k\geq i$, $l\leq j$. Поэтому $T(r_{ij})$ задается
прямоугольником матричных позиций вниз и влево от $(i,j)$, включая саму
позицию; для $Q(r_{ij})$ ее исключаем.

Множество углов можно записать в виде
\[
    \mathcal{L}=\{r_{i_t,j_t}\mid 1\leq t\leq m\},\qquad
    i_1<\cdots<i_m,\quad j_1<\cdots<j_m.
\]
Объединение соответствующих прямоугольников образует матричную лестницу для
$T(\mathcal{L})$; исключение угловых позиций дает $Q(\mathcal{L})$
(рисунок~\ref{fig:corners-An}).

\begin{definition}\label{def:stand-enval}
    Идеал $H$ кольца Ли $N\Phi(\mathbb{K})$ или кольца $R_{\Phi}$ называем
    \emph{стандартным}, если $Q(\mathcal{L}(H))\subset H$. Специальную обертывающую
    алгебру $R_{\Phi}$ алгебры Ли $N\Phi(\mathbb{K})$ над полем $\mathbb{K}$
    называем \emph{стандартной}, если все ее идеалы стандартны.
\end{definition}

\begin{example}\label{example:nonstandard-NT4}
    В алгебре Ли $NT(4,\mathbb{K})^{(-)}$ рассмотрим идеал
    \[
        H=\left\{
        \begin{pmatrix}
            0 & 0  & 0 & 0 \\
            a & 0  & 0 & 0 \\
            b & 0  & 0 & 0 \\
            c & -b & a & 0
        \end{pmatrix}
        \ \middle|\ a,b,c\in\mathbb{K}
        \right\}.
    \]
    Имеем $\mathcal{L}(H)=\{r_{21},r_{43}\}$. Идеал $H$ нестандартен, поскольку
    $e_{31}\in Q(\mathcal{L}(H))$, но $e_{31}\notin H$.
\end{example}

Из свойств систем корней и определений легко вытекает

\begin{lemma}\label{l:stand-one-corner}
    В кольце $R_{\Phi}$ каждый идеал с единственным углом стандартен. Если $\Phi$
    ранга $2$, любая алгебра $R_{\Phi}$ стандартна.
\end{lemma}

\begin{proof}
    Доказательство первого утверждения леммы для идеала $H$ в $R_{\Phi}$ с
    множеством углов $\mathcal{L}(H)=\{r\}$ проводим с помощью индукции по
    $\mathrm{h}(\Phi)-\height(r)$. Ясно, что $Q(\rho) = 0 \subset H \subset T(r)$.
    Пусть $r$ "--- не максимальный корень. Тогда существует простой корень $p$ с
    условием $r+p \in \Phi^+$ и одно из произведений $e_pe_r$, $e_re_p$ равно
    $e_{r+p}$. Поэтому множество $(\mathbb{K}e_p)H + H(\mathbb{K}e_p)$ по модулю
    $Q(r+p)$ совпадает с $\mathbb{K}e_{r+p}$ и, по индукции, порождает идеал,
    содержащий $Q(r+p)$, откуда $T(r+p) \subset H$. Учитывая произвол в выборе $p$,
    получаем включение $Q(r) \subset H$.

    Второе утверждение леммы сразу следует из первого для случая систем корней
    $\Phi$ типа $A_2$, $B_2$ и $G_2$.
\end{proof}

Для алгебры Ли $N\Phi(\mathbb{K})$ типа $A_{n-1}$ алгебра $R=NT(n,\mathbb{K})$
является ассоциативной точной обертывающей. Стандартность всех идеалов
ассоциативной алгебры $R$ установлена Р.~Дюбишем и С.~Перлисом \cite[теорема
9]{Dub-Perl-1951}, а идеалов кольца $R$ "--- В.\nobreak\:М.~Левчуком
\cite[следствие~4.2]{vL74}.

Алгебры Ли $N\Phi(\mathbb{K})$ остальных классических типов заданы в
\cite[лемма 2]{vl90a_ru} аналогично в базисе Шевалле из <<матричных единиц>>
$e_{iv}$ с ограничениями
\[
    B_n: \ -i < v < i \leq n;\quad
    D_n: \ 1 \leq |v| < i \leq n;\quad
    C_n: \ -i \leq v < i \leq n, \ v \neq 0.
\]

К суммам двух корней, являющимся корнями, помимо сумм $r_{ij}+ r_{jv}=r_{iv}$
(аналогично типу $A_n$), здесь относятся еще суммы
\[
    r_{kv}+ r_{m,-v}=r_{k, -m}\quad (k>m>|v|),
\]
а для типа $C_n$ при $k =m>|v|\geq 1$ также суммы $r_{kv}+ r_{k,-v} =
r_{k,-k}$.

Любой элемент из $N\Phi(\mathbb{K})$ представляется суммой $\sum a_{iv}e_{iv}$
и $\Phi^+$-матрицей $||a_{iv}||$ над $\mathbb{K}$ соответствующего типа. Так,
$B_n^+$-матрица имеет вид
%
\[
    \begin{gathered}
        a_{10} \\
        a_{2,-1}\ a_{20}\ a_{21} \\
        \dots\ \ \dots\ \ \dots \\
        a_{n,-n+1}\ \dots\ a_{n,-1}\ a_{n0}\ a_{n1}\ \dots\ a_{n,n-1}.
    \end{gathered}
\]
%
Отбрасывая столбец с нулевым номером, получаем $D_n^+$-матрицу.

В \cite[лемма 2]{vl90a_ru} доказана

\begin{lemma}\label{l:StructConst-vl90a_ru}
    Знаки структурных констант $N_{rs}$ алгебры Ли $N\Phi(\mathbb{K})$ можно
    выбрать так, что $[e_{ij},e_{jv}]=e_{iv}=-[e_{jv},e_{ij}]$ и выполняются
    равенства:
    \[
        [e_{jv},e_{i,-v}]=e_{i,-j}\quad (\Phi=B_n,D_n),\quad i>j>|v|>0;\]
    %
    \[[e_{i0},e_{j0}]=2e_{i,-j}\quad(\Phi=B_n),\quad
        [e_{ij},e_{i,-j}]=2e_{i,-i}\quad(\Phi=C_n),\quad i>j\geq1;\]
    %
    \[[e_{jm},e_{i,-m}]=[e_{im},e_{j,-m}]=e_{i,-j}\quad(\Phi=C_n),\quad
        i>j>m\geq1.\]
\end{lemma}

Соответственно типу ${\Phi}$ для алгебр Ли $N\Phi(\mathbb{K})$ используем
обозначения $NB_n(\mathbb{K})$, $NC_n(\mathbb{K})$ и так далее. Очевидным
следствием \crefRod{l:StructConst-vl90a_ru} является

\begin{lemma}\label{l:ReductBnCnDn}
    Факторалгебра $ND_n(\mathbb{K})/T(r)$ изоморфна $NT(n,\mathbb{K})^{(-)}$ точно
    для двух корней $r= r_{2,-1}$ и $r=r_{21}$ при $n>4$, и точно для трех простых
    корней $r$ при $n=4$. Кроме того, $NB_n(\mathbb{K})/T(r) \simeq NT(n+
    1,\mathbb{K})^{(-)}$ и $NC_n(\mathbb{K})/T(r) \simeq NT(n+ 1,\mathbb{K})^{(-)}$
    для единственного корня $r=r_{2,-1}$ и $r=r_{2,-2}$ соответственно.
\end{lemma}

\begin{remark}\label{remark:A3-D3}
    Для ${\Phi}$ типа $B_n$ и $C_n$ произвольная специальная обертывающая алгебра
    $R_{\Phi}$ алгебры Ли $N\Phi(\mathbb{K})$ имеет $\mathbb{K}$-подмодуль $M_\Phi$
    с базисом $\{e_{iv}\ |\ 1 \leq |v| < i \leq n\}$. Он не является подалгеброй
    для ${\Phi}$ типа $C_n\ (n>1)$, а в алгебре $R_{\Phi}$ типа $B_n$ является
    подалгеброй, изоморфной точной обертывающей алгебре алгебры Ли
    $ND_n(\mathbb{K})$.
\end{remark}

Точные обертывающие алгебры $R_{\Phi}$ классического типа с выбором знаков
структурных констант по \crefDat{l:StructConst-vl90a_ru} обозначаем
соответственно типу
%
\begin{equation}\label{eq:RAn-RDn}
    RA_n(\mathbb{K})=NT(n+1, \mathbb{K}),\ \ RB_n(\mathbb{K}),\ \ RC_n(\mathbb{K}),\ \ RD_n(\mathbb{K}).
\end{equation}
%
\begin{lemma}\label{l:mult-RAn-RDn}
    В алгебрах $R_{\Phi}$ из списка \eqref{eq:RAn-RDn} умножение определяют правила
    \[
        e_{ij}e_{jv}= e_{iv},\quad e_{iu}e_{jv}=0\quad (u\neq j, \ u\neq -v);
    \]
    %
    \[
        \Phi = B_n, D_n:\ \
        e_{jv}e_{i,-v}=e_{i,-j}\ \ (i>j>\mid v \mid > 0),\quad
        e_{iv}e_{j,-v}=0\ \ (i\geq j>\mid v \mid > 0);
    \]
    %
    \[
        \Phi=C_n:\ \ e_{ij}e_{i,-j}=-e_{i,-j}e_{ij}=e_{i,-i}\quad (i>j\geq 1),
    \]
    %
    \[
        e_{jm}e_{i,-m} = e_{im}e_{j,-m} = e_{i,-j},\quad
        e_{i,-m}e_{jm}=e_{j,-m}e_{im}=0\quad (i>j>m \geq 1);
    \]
    %
    \[
        \Phi = B_n:\qquad e_{i0}e_{j0} = e_{i,-j}=-e_{j0}e_{i0}\quad (i>j\geq 1).
    \]
\end{lemma}

\begin{lemma}\label{l:R-A3-D3}
    Точная обертывающая алгебра $RD_3(\mathbb{K})$ алгебры Ли $N\Phi(\mathbb{K})$
    типа $A_3$ является неассоциативной и нестандартной.
\end{lemma}

\begin{proof}
    Алгебры Ли $N\Phi(\mathbb{K})$ типа $A_3$ и $D_3$ изоморфны, поскольку их
    системы корней эквивалентны. Однако алгебра $RD_3(\mathbb{K})$, в отличие от
    алгебры $RA_3(\mathbb{K})= NT(4,\mathbb{K})$, имеет нестандартный идеал
    \[
        \mathbb{K}(e_{2 1}+ e_{2,-1})+ \mathbb{K}(e_{3 1}+ e_{3,-1})+ \mathbb{K}e_{3,-2}
    \]
    и является неассоциативной, так как $e_{2,-1} (e_{32} e_{21}) = e_{3,-2}$ и
    $(e_{2,-1}e_{32}) e_{21}=0$.
\end{proof}

\leq i < h$.
\begin{lemma}
    Идеал $H \subset L_4$ специальной обертывающей алгебры $R_\Phi$ типа $F_4$
    является стандартным.
\end{lemma}
\begin{proof}
    Пусть $r \in \mathcal{L}(H)$, $r \neq \rho$. Тогда существует простой корень
    $p$ с условием $r+p \in \Phi^+$ и одно из произведений $e_pe_r$, $e_re_p$ равно
    $e_{r+p}$.

    В случае, когда корни $q_{3,-2}$ и $p_{41}$ не лежат одновременно в
    $\mathcal{L}(H)$, множество $(\mathbb{K}e_p)H + H(\mathbb{K}e_p)$ по модулю
    $Q(r+p)$ совпадает с $\mathbb{K}e_{r+p}$ и порождает идеал, содержащий
    $Q(r+p)$, откуда $T(r+p) \subset H$. Учитывая произвол в выборе $p$, получаем
    включение $Q(r) \subset H$.

    Остается рассмотреть случай, когда $r \in \{q_{3,-2}, p_{41}\} \subset
    \mathcal{L}(H)$. Включение $T(p_{4,-1}) \subset H$ получаем аналогично первому
    случаю, выбрав в качестве $p$ корень $q_{21}$. Полагая теперь $p=p_{32}$,
    включение $T(q_{42}) \subset H$ получаем, рассматривая множество
    $(\mathbb{K}e_p)H + H(\mathbb{K}e_p)$ по модулю $T(p_{4,-1}) + Q(q_{42})$.
\end{proof}

% \begin{theorem}[{{\cite[теорема 1.1]{jsfu2018}}}]\label{th:stand-f4}
\begin{theorem}\label{th:stand-f4}
    %<*th:stand-f4>
    Существуют как стандартные, так и нестандартные специальные обертывающие
    алгебры $R_\Phi$ типа $F_4$.
    %</th:stand-f4>
\end{theorem}
\begin{proof}
    Переобозначая $N_{rs}$ через $N(r,s)$, выберем значения $N(r,s)$ на следующих
    парах корней:
    \begin{equation}\label{eq:StructConst-Stand-F4}
        \begin{gathered}
            N(q_{21},q_{3,-1}) = 1,\ N(q_{21},p_{31}) = -1, \
            N(q_{32},p_{3,-1}) = 1, \ N(q_{10},p_{3,-1}) = -1, \\
            %
            N(q_{32},q_{2,-1}) = -1, \ N(q_{32},q_{21}) = -1, \ N(q_{10},p_{32}) = -1, \
            N(q_{21},q_{10}) = -1.
        \end{gathered}
    \end{equation}

    Для получения нестандартной обертывающей алгебры $R_1$ дополнительно полагаем
    $N(q_{32},q_{20}) = -1$, \ $N(q_{10},q_{20}) = 2$ и
    \[
        N(p_{32},q_{2,-1}) = -1,\quad N(p_{32},q_{30}) = -1,\quad N(q_{10},q_{31}) = 1.
    \]
    Выбирая произвольно для оставшихся экстраспециальных пар корней $r$ и $s$ знаки
    структурных констант $N_{rs}$, получаем алгебру $R_1$. Легко видеть, что при
    $c\in\mathbb{K}^*$ и $d\in\mathbb{K}$ в алгебре $R_1$ идеалы вида
    \begin{equation*}%\label{eq:f4-nonstandard-ideal}
        \mathbb{K}(e_{q_{30}} + ce_{q_{2,-1}} + de_{p_{42}}) +
        \mathbb{K}(e_{p_{42}} + ce_{p_{3,-2}}) + T(q_{3,-1}) + T(q_{43})
    \end{equation*}
    являются нестандартными. Кроме того, легко проверить, что все остальные
    идеалы алгебры $R_1$ являются стандартными.

    \medskip

    Покажем, что знаки структурных констант $N_{rs}$ алгебры $N\Phi(\mathbb{K})$
    типа $F_4$ можно выбрать так, что все идеалы кольца $R_\Phi$ будут
    стандартными. Прямая вычислительная проверка показывает, что алгебра $R_\Phi$
    типа $F_4$ является стандартной, если выполняются следующие условия:
    \begin{equation*}%\label{eq:F4-stand-condition}
        \begin{aligned}
            N(q_{21},q_{3,-1}) = -N(q_{21},p_{31}), \quad & N(q_{32},p_{3,-1}) =
            -N(q_{32},q_{2,-1}),                                                 \\
            N(q_{10},p_{3,-1}) = -N(q_{10},q_{31}), \quad & N(p_{32},q_{30}) =
            -N(p_{32},q_{2,-1}),                                                 \\
            N(q_{21},p_{31}) = N(q_{32},q_{21}), \quad    & N(q_{10},q_{31}) =
            -N(q_{10},p_{32}),                                                   \\
            N(q_{32},q_{21}) = N(q_{21},q_{10}), \quad    & N(q_{21},q_{10}) =
            N(q_{10},p_{32}) .
        \end{aligned}
    \end{equation*}

    Для построения стандартной обертывающей алгебры $R_2$ снова
    используем~\eqref{eq:StructConst-Stand-F4}. Полагаем также $N(q_{32},q_{20}) =
    -1$ и $N(q_{10},q_{20})=- 2$. Далее по тождеству Якоби получаем
    $N(p_{32},q_{20}) = 1$ и
    \begin{equation*}
        N(p_{32},q_{2,-1}) = 1,\quad
        N(p_{32},q_{30}) = -1,\quad
        N(q_{10},q_{31}) = 1.
    \end{equation*}
    Выбирая произвольно для оставшихся экстраспециальных пар корней $r$ и $s$ знаки
    структурных констант $N_{rs}$, получаем стандартную обертывающую алгебру~$R_2$.
\end{proof}

\section{Теоремы единственности}\label{sec:uniqtheorems}
%
В этом параграфе мы исследуем для классических типов $\Phi$ условия
однозначности специальных обертывающих алгебр нильтреугольной алгебры
$N\Phi(\mathbb{K})$.

\medskip

Ясно, что для любой подсистемы $\Psi$ системы корней $\Phi$ точная обертывающая
алгебры Ли $N\Psi(\mathbb{K})$ определена в алгебре $R_{\Phi}$ как подалгебра
\[
    R(\Psi)\coloneq\sum_{a \in \Psi^+}\mathbb{K}e_a.
\]

Когда в системе $\Phi$ все корни одной длины и базис $\Pi(\Psi)$ подсистемы
$\Psi$ лежит в базисе $\Pi(\Phi)$, нестандартный идеал подалгебры $R(\Psi)$
порождает в алгебре $R_{\Phi}$ также нестандартный идеал. Отсюда вытекает

\begin{lemma}\label{l:stand-subsystem}
    Пусть $\Pi(\Psi) \subset \Pi(\Phi)$ для подсистемы $\Psi$ системы корней $\Phi$
    одной длины. Если специальная обертывающая $R_{\Phi}$ алгебры Ли
    $N\Phi(\mathbb{K})$ стандартная, то подалгебра $R(\Psi)$ алгебры $R_{\Phi}$
    также стандартна.
\end{lemma}

Как и в \cite{JA-12}, неинцидентные корни $r,s\in\Phi^+$ называем
\emph{$p$-связанными} для простого корня $p$, если $r+p,s+p\in\Phi^+$.
Подсистему корней $\Psi(r,p,s)$ в $\Phi$ определяем равенством
\[
    \Psi=\Psi(r, p, s)\coloneq\Phi\cap (\mathbb{Z}r+\mathbb{Z}p+\mathbb{Z}s).
\]

Алгебры $R_{\Phi}$ типа $A_{2}$ ассоциативны, поскольку для них $R_{\Phi}^3=0$.
В системах корней $\Phi$ одной длины неразложимая подсистема $\Psi$ ранга 3
всегда типа $A_3$. Очевидна

\begin{lemma}\label{l:subsystem-rank3}
    Пусть $r,s\in\Phi^+$ "--- $p$-связанные корни для $p\in\Pi(\Phi)$. Если
    подсистема корней $\Psi=\Psi(r,p,s)$ имеет ранг $>2$, то либо она типа $A_3$ с
    базисом $\{r,p,s\}$, либо типа $B_3$ или $C_3$.
\end{lemma}

Замена $a\circ b\coloneq ba$ умножения в $R$ дает \emph{противоположную
    алгебру} $R^{(op)}$, антиизоморфную $R$.

\begin{lemma}\label{l:NT-opposite}
    Алгебра $NT(n,\mathbb{K})$ нижних нильтреугольных $(n\times n)$-матриц над
    полем $\mathbb{K}$ изоморфна своей противоположной алгебре
    $NT(n,\mathbb{K})^{(op)}$.
\end{lemma}
\begin{proof}
    Пусть $J=(j_{ab})$ "--- матрица перестановки с единицами на побочной диагонали,
    то есть $j_{a,n+1-a}=1$ и $j_{ab}=0$ иначе. Тогда $J^{-1}=J$.

    Рассмотрим линейное отображение
    \[
        \varphi\colon NT(n,\mathbb{K})\to NT(n,\mathbb{K}),\qquad \varphi(X)=JX^{t}J^{-1}=JX^tJ,
    \]
    где $X^t$ "--- транспонированная матрица. Если $X$ нижнетреугольна, то $X^t$
    верхнетреугольна, а сопряжение матрицей $J$ переводит верхнетреугольные матрицы
    в нижнетреугольные; значит, $\varphi(NT(n,\mathbb{K})) = NT(n,\mathbb{K})$.

    Для любых $X,Y\in NT(n,\mathbb{K})$ имеем
    \[
        \varphi(XY)=J(XY)^tJ=JY^tX^tJ=(JY^tJ)(JX^tJ)=\varphi(Y)\varphi(X),
    \]
    то есть $\varphi$ "--- антиавтоморфизм алгебры $NT(n,\mathbb{K})$.
    Следовательно, отображение $X\mapsto \varphi(X)$ задает изоморфизм
    $NT(n,\mathbb{K})\simeq NT(n,\mathbb{K})^{(op)}$.
\end{proof}

\begin{lemma}\label{l:stand-opposite}
    Если $R_{\Phi}$ "--- стандартная обертывающая алгебра, то $R_{\Phi}^{(op)}$
    также стандартна.
\end{lemma}
\begin{proof}
    Алгебра $R_{\Phi}^{(op)}$ также специальная по определению~\ref{def:RPhi} при
    замене $N_{rs}$ на $-N_{rs}$. Множества идеалов в $R_{\Phi}$ и в
    $R_{\Phi}^{(op)}$ совпадают. Отображение $x\mapsto -x$ задает изоморфизм алгебр
    Ли $R_{\Phi}^{(-)}$ и $(R_{\Phi}^{(op)})^{(-)}$, поскольку
    \[
        [-a,-b]_{R_{\Phi}^{(op)}}=ba-ab=-[a,b]_{R_{\Phi}}.
    \]
    При этом знаковая замена сохраняет корневые компоненты, множества углов идеалов
    и подпространства $Q(\mathcal{L})$, а значит, и стандартность.
\end{proof}

Отметим, что переход к противоположной алгебре $R_{\Phi}^{(op)}$ сохраняет и
стандартность, и ассоциативность.

Для любого кольца $A$ наряду с его односторонними аннуляторами выделяют
\emph{аннулятор}
\[
    \Ann(A) \coloneq \{\alpha\in A\ | \ \alpha A = 0 = A\alpha\}.
\]

\begin{lemma}\label{l:stand-A3}
    Пусть $\Psi=\Psi(r, p, s)$ "--- подсистема типа $A_3$ с базисом
    $\{r,p,s\}\subset\Phi^+$, где $r+p,s+p\in\Phi^+$. Тогда стандартность
    подалгебры $R(\Psi)$ в $R_{\Phi}$ равносильна условию $N_{s,p} = N_{p, r}$ и
    дает равенство $N_{s+p,r}=N_{s,r+p}$, причем $R(\Psi) \simeq NT(4,\mathbb{K})$,
    когда $N_{s+p,r} = N_{p,r}$. Если $N_{s+p,r}= -N_{p,r}$, то $R(\Psi)$ "---
    неассоциативная алгебра.
\end{lemma}

\begin{proof}
    В подсистеме $\Psi=\Psi(r,p,s)$ типа $A_3$ корень $r+p+s$ "--- единственный,
    представимый неоднозначно суммой специальной пары корней.

    Точную обертывающую алгебру $R(\Psi)$ алгебры Ли $N\Psi(\mathbb{K})$ порождают
    $\mathbb{K}e_r$, $\mathbb{K}e_s$ и $\mathbb{K}e_p$. Ненулевые структурные
    константы для $R(\Psi)$ типа $A_3$ равны $\pm 1$.

    Положим $I\coloneq\mathbb{K}e_p+\mathbb{K}e_{r+p}+\mathbb{K}e_{s+p}
    +\mathbb{K}e_{r+p+s}$. Аннулятор кольца $R(\Psi)$ есть идеал
    $\mathbb{K}e_{r+p+s}=R(\Psi)^3$, причем
    \[
        R(\Psi)=\mathbb{K}e_r+I+\mathbb{K}e_s.
    \]
    Идеал $I$ и его аннулятор в $R(\Psi)$ совпадают с аннулятором идеала
    $R(\Psi)^2$. Если по модулю $ R(\Psi)^3$ односторонний аннулятор идеала $I$
    (равносильно, элемента $e_p$) совпадает с $R(\Psi)$ и, например, $e_p
    e_r=e_{r+p},\ e_re_p=0$, то $\mathbb{K}(e_r+ e_s)$ порождает в $R(\Psi)$
    нестандартный идеал, как и в \crefPred{l:R-A3-D3}.

    Таким образом, при условии стандартности $R(\Psi)$ элементы $e_r, e_s$ лежат в
    разных односторонних аннуляторах элемента $e_p$. Это означает, что совпадают
    знаки констант $N_{p,r}$ и $N_{s,p}$, равносильно, $N_{p,r} N_{s,p}=1$.

    С учетом \crefRod{l:stand-opposite}, с точностью до перехода к противоположной
    алгебре, имеем
    \begin{equation}\label{eq:An-Nsp-Npr-1}
        N_{s,p} =N_{p,r}=1,\quad
        e_se_p=e_{s+p},\quad
        e_pe_r=e_{r+p},\quad
        e_re_p=0=e_pe_s.
    \end{equation}

    Тождество Якоби в кольце Ли $R(\Psi)^{(-)}$ дает равенства
    \[
        [e_s,[e_p,e_r]] = [[e_s,e_p],e_r],\quad N_{pr}N_{s,r+p}=N_{sp}N_{s+p,r}.
    \]
    Отсюда $N_{s,r+p}=N_{s+p,r}\ (=\pm 1)$, то есть $N_{s,r+p}N_{s+p,r} = 1$. В
    случае
    \begin{equation*}
        N_{s+p,r} = 1, \quad
        e_{s+p}e_r=e_{r+s+p}=e_{s}e_{r+p}, \quad
        e_re_{s+p}=0=e_{r+p}e_s,
    \end{equation*}
    приходим к изоморфизму алгебр $R(\Psi)\simeq NT(4,\mathbb{K})$, действующему по правилу:
    \[
        e_p \mapsto e_{32}, \ \ e_r \mapsto e_{21}, \ \ e_s \mapsto
        e_{43}, \ \ e_{r+p} \mapsto e_{31}, \ \ e_{s+p} \mapsto e_{42}, \
        \ e_{s+r+p } \mapsto e_{41}.
    \]

    В оставшемся случае, когда $N_{s+p,r} = -1$, имеем $e_p(e_re_s)=0$ и
    $(e_pe_r)e_s = e_{s+r+p}$. Поэтому алгебра $R(\Psi)$ неассоциативная; по модулю
    аннулятора она изоморфна алгебре $NT(4,\mathbb{K})/\mathbb{K}e_{41}$. Лемма
    доказана.
\end{proof}

    Если специальная обертывающая алгебра $R_\Phi$ алгебры Ли $N\Phi(\mathbb{K})$
    типа $A_n$ $(n>2)$ стандартна, то $R_\Phi\simeq NT(n+1, \mathbb{K})$ по модулю
    третьих степеней алгебр.
\end{lemma}
\begin{proof}

    $\Psi=\Psi(r,p,s)$ типа $A_3$ с базисом $\{r,p,s\}\subset\Pi(\Phi)$, по лемме
    \ref{l:stand-subsystem}, свойство стандартности $R_\Phi$ наследуется
    подалгеброй $R(\Psi)$ "--- точной обертывающей алгебры Ли $N\Psi(\mathbb{K})$.

    \medskip

    Ясно, что пара $p$-связанных корней $r,s\in\Phi^+$ с простым корнем $p$

    экстраспециальны.

    Пусть $\Psi=\Psi(r, p,s)$ "--- подсистема в $\Phi$ с базисом из простых корней
    $r,p,s$. По леммам~\ref{l:stand-one-corner} и~\ref{l:stand-A3} для стандартной

Критерий стандартности специальных обертывающих алгебр $R_\Phi$ алгебры Ли
$N\Phi(\mathbb{K})$ типа $A_{n-1}\ (n>3)$ дает

% \begin{theorem}[{{\cite[теорема 1]{imm2020}}}]\label{th:Stand-An}
\begin{theorem}\label{th:Stand-An}
    %<*th:Stand-An>
    Специальная обертывающая алгебра $R_\Phi$ алгебры Ли $N\Phi(\mathbb{K})$ типа
    $A_{n-1}$ $(n>3)$ стандартна тогда и только тогда, когда по модулю аннулятора
    изоморфна $NT(n,\mathbb{K})/\mathbb{K}e_{n1}$.
    %</th:Stand-An>
\end{theorem}

\begin{proof}
    Обозначим $Z=\Ann(R_\Phi)=\mathbb{K}e_{n1}$ и положим
    $\nu_{ijk}=N_{r_{ij},r_{jk}}\in\{-1,1\}$ при $1\leq k<j<i\leq n$. Константы
    $\nu_{ijk}$ и равенства между ними рассматриваем в $\mathbb{Z}$.

    Пусть алгебра $R_\Phi$ стандартна. Для $1\leq l<k<j<i\leq n$ с $j=k+1$ возьмем
    корни $r=r_{kl}$, $p=r_{jk}$ и $s=r_{ij}$. Если $\nu_{jkl}\neq\nu_{ijk}$, то
    $N_{pr}=N_{ps}$, и по таблице специального умножения подпространство
    \[
        I=\mathbb{K}(e_r+e_s)+\mathbb{K}(e_{r+p}+e_{s+p})+
        \sum_{\mathclap{\substack{q\in\Phi^+,\ q>r\ \text{или}\ q>s\\
                    q\neq r+p,\ s+p}}}\mathbb{K}e_q
    \]
    является двусторонним идеалом алгебры $R_\Phi$. Корень $r$ является углом
    идеала $I$, но $e_{r+p}\notin I$, что противоречит стандартности.
    Следовательно,
    \begin{equation}\label{eq:Stand-An-linked}
        \nu_{ijk}=\nu_{jkl}\qquad
        (1\leq l<k<j<i\leq n,\ j=k+1).
    \end{equation}
    В частности, константы $\nu_{t+2,t+1,t}$ $(1\leq t\leq n-2)$ имеют общее
    значение $\nu\in\{-1,1\}$. По тождеству Якоби
    \begin{equation}\label{eq:Stand-An-Jacobi}
        \nu_{ijk}\nu_{ikl}=\nu_{ijl}\nu_{jkl}\qquad
        (1\leq l<k<j<i\leq n).
    \end{equation}
    Индукцией по $i-k$ докажем, что $\nu_{ijk}=\nu$ при $(i,k)\neq(n,1)$.
    Случай $i-k=2$ уже установлен. При $i-k>2$ подстановка
    $(i,j+1,j,k)$ в \eqref{eq:Stand-An-Jacobi} и индуктивное предположение дают
    $\nu_{ijk}=\nu_{i,j+1,k}$ для $j+1<i$, поэтому константа $\nu_{ijk}$ не
    зависит от $j$. Если $i<n$, то по \eqref{eq:Stand-An-linked}
    $\nu_{i,i-1,k}=\nu_{i+1,i,i-1}=\nu$; если $k>1$, то
    $\nu_{i,k+1,k}=\nu_{k+1,k,k-1}=\nu$.

    Таким образом, по модулю $Z$ специальная таблица умножения совпадает с
    матричной при $\nu=1$ и с противоположной матричной при $\nu=-1$.
    Антиавтоморфизм из доказательства \crefRod{l:NT-opposite} сохраняет
    $\mathbb{K}e_{n1}$, поэтому матричный фактор изоморфен своей противоположной
    алгебре. Необходимость доказана.

    Обратно, предположим, что $R_\Phi/Z\simeq NT(n,\mathbb{K})/\mathbb{K}e_{n1}$.
    Выберем в $R_\Phi$ корневые векторы $f_{ij}=\pm e_{ij}$ так, чтобы
    $[f_{ij},f_{jk}]=f_{ik}$ при $1\leq k<j<i\leq n$. Положим
    $f_{ij}f_{jk}=c_{ijk}f_{ik}$, где $c_{ijk}=1$ при $\nu_{ijk}=1$ и $c_{ijk}=0$
    при $\nu_{ijk}=-1$; тогда $f_{jk}f_{ij}=(c_{ijk}-1)f_{ik}$.

    При $n=4$ неравенство $\nu_{432}\neq\nu_{321}$ дало бы четырехмерный
    односторонний аннулятор фактора, тогда как у матричного фактора оба
    односторонних аннулятора трехмерны. Следовательно, $\nu_{432}=\nu_{321}$ и
    $c_{432}=c_{321}$. При $n\geq5$ ассоциативность фактора $R_\Phi/Z$, примененная
    к $f_{ij},f_{jk},f_{kl}$ и их перестановкам, дает
    \[
        c_{ijk}=c_{ijl}=c_{ikl}=c_{jkl}\qquad
        (1\leq l<k<j<i\leq n,\ (i,l)\neq(n,1)).
    \]
    На пересекающихся наборах индексов $\{1,\ldots,n-1\}$ и $\{2,\ldots,n\}$ эти
    равенства дают одно общее значение $c\in\{0,1\}$ для всех троек с
    $(i,k)\neq(n,1)$. При $n=4$ общее значение $c$ определяется равенством
    $c_{432}=c_{321}$. Отождествим линейное пространство $R_\Phi$ с
    $NT(n,\mathbb{K})$, сопоставив $f_{ij}$ матричным единицам. Это отождествление
    сохраняет коммутатор и корневые компоненты. Таблица умножения $R_\Phi$ по
    модулю $Z$ совпадает с таблицей матричного умножения $xy$ при $c=1$ или
    умножения $-yx$ при $c=0$.

    Рассмотрим произвольный идеал кольца $R_\Phi$ и обозначим его образ при
    указанном отождествлении через $H$. Сохранение коммутатора означает, что $H$
    является идеалом кольца Ли $NT(n,\mathbb{K})^{(-)}$.

    Если $H\not\subset Z$, то $[NT(n,\mathbb{K})^{(-)},H]$ "--- ненулевой
    $\mathbb{K}$-линейный идеал, содержащийся в $H$. Нильпотентность алгебры Ли
    $NT(n,\mathbb{K})^{(-)}$ и одномерность ее центра дают $Z\subset H$, поэтому
    совпадение таблиц умножения по модулю $Z$ сохраняет идеальность $H$ в матричном
    кольце. При $H\subset Z$ идеальность $H$ в матричном кольце очевидна.

    Таким образом, $H$ является идеалом матричного кольца и по
    \cite[следствие~4.2]{vL74} стандартен.
\end{proof}

\begin{remark}\label{remark:Stand-Non-Assoc-An}
    Замена соотношений $e_{nj}e_{j1}= e_{n1}$ и $e_{j1}e_{nj}=0$ новыми
    \[
        e_{nj}e_{j1}=0,\quad e_{j1}e_{nj}=e_{n1}\qquad (j=2,3,\ldots,n-1)
    \]
    приводит алгебру $NT(n,\mathbb{K})\ (n>3)$ к неассоциативной стандартной
    обертывающей алгебре типа $A_{n-1}$. Обе алгебры по модулю аннулятора
    $\mathbb{K}e_{n1}$ изоморфны, поэтому стандартность обертывающей алгебры не
    влечет ассоциативности.
\end{remark}

Существование стандартных обертывающих алгебр $R_{\Phi}$ для типов $B_n$ и
$C_n$ установлено в \cite[теорема~2]{imm2020}. Более точное описание
нестандартных идеалов дает следующая теорема.

% \begin{theorem}[{{\cite[теорема~4]{smj2023}}}]\label{th:classical-Exc-RDn}
\begin{theorem}\label{th:classical-Exc-RDn}
    Пусть $R_{\Phi}$ есть алгебра классического типа из списка \eqref{eq:RAn-RDn}.
    Если $s$ "--- угол идеала $H$ кольца $R_{\Phi}$ и $Q(s) \not\subset H$, то
    $\Phi$ типа $D_n$, $s=r_{iv}$, $2 \leq i<n$, $v=\pm 1$, причем идеалы
    $T(r_{i+1,1})$ и $T(r_{i+1,-1})$ оба не лежат в $H$.
\end{theorem}

\begin{proof}
    Тип $A_n$ исключается стандартностью всех идеалов кольца $NT(n+1,\mathbb{K})$
    \cite[следствие~4.2]{vL74}. Для остальных типов сохраняем индексацию корней из
    \S~\ref{sec:enveloping}. Запишем угол $s$ идеала $H$ в виде $s=r_{iv}$ и
    выберем элемент $h=\sum a_{kl}e_{kl}\in H$ с $a_{iv}\neq0$. Используем таблицу
    умножения из \crefRod{l:mult-RAn-RDn} и \crefVin{l:stand-one-corner}. Идеал $H$
    содержит подпространства $\mathbb{K}(e_ph)$ и $\mathbb{K}(he_p)$ для любого
    базисного элемента $e_p$ $(p\in\Phi^+)$.

    В алгебрах $RB_n$, $RC_n$ при $v<0$ и в $RD_n$ при $v<-1$ произведение
    $e_{i+1,i}h$ дает $T(r_{i+1,v})\subset H$ при $i<n$. Произведение
    $e_{|v|+1,|v|}h$ дает $T(r_{i,v-1})\subset H$ при $|v|+1<i$ для типов $B_n$,
    $D_n$ и при $|v|<i$ для типа $C_n$. Сумма полученных идеалов равна $Q(s)$.

    При $v>1$ в алгебрах $RB_n$ и $RD_n$ произведения $e_{i+1,i}h$ при $i<n$ и
    $he_{v,v-1}$ дают соответственно $T(r_{i+1,v})\subset H$ и $T(r_{i,v-1})\subset
    H$. В типе $D_n$ при $v=2$ произведение $he_{2,-1}$ дает также
    $T(r_{i,-1})\subset H$. Сумма указанных идеалов равна $Q(s)$.

    В алгебре $RB_n$ при $v=0$ произведения $e_{m0}h$ дают $T(r_{i,-m})\subset H$
    для $1\leq m<i$ и $T(r_{m,-i})\subset H$ для $i<m\leq n$. По модулю суммы этих
    идеалов произведение $e_{i+1,i}h$ при $i<n$ дает $T(r_{i+1,0})\subset H$. При
    $v=1$ сумма $he_{10}+e_{10}h$ дает $T(r_{i0})\subset H$, а произведение
    $e_{i+1,i}h$ при $i<n$ "--- $T(r_{i+1,1})\subset H$. В каждом случае сумма
    перечисленных идеалов равна $Q(s)$.

    В алгебре $RC_n$ при $v>0$ рассуждаем индукцией по $v$. Произведение
    $e_{i+1,i}h$ при $i<n$ дает $T(r_{i+1,v})\subset H$. При $v=1$ произведение
    $he_{1,-1}$ дает $T(r_{i,-1})\subset H$. При $v>1$ поправка $a_{v,1-v}e_{v,-v}$
    лежит в $H$: при $a_{v,1-v}\neq0$ выберем угол $u$ идеала $H$ с условием $u\leq
    r_{v,1-v}$. Если второй индекс угла $u$ положителен, этот индекс меньше $v$. По
    рассмотренному случаю отрицательного второго индекса или индукции имеем
    $\mathbb{K}e_{v,-v}\subset Q(u)\subset H$. Поэтому элемент
    $he_{v,v-1}+a_{v,1-v}e_{v,-v}$ порождает идеал с единственным углом
    $r_{i,v-1}$, откуда $T(r_{i,v-1})\subset H$. При любом $v>0$ сумма полученных
    идеалов равна $Q(s)$.

    Остается случай $R_\Phi=RD_n(\mathbb{K})$, $v=\pm1$. Введем идеалы
    $U_2=T(r_{3,-2})$ и $U_i=T(r_{i,-2})$ при $3\leq i\leq n$. При $i\geq3$
    произведение $e_{2,-v}h$ порождает идеал с единственным углом $r_{i,-2}$, а при
    $i=2$ имеем $e_{2,-v}(e_{32}h)=a_{2v}e_{3,-2}$. По \crefDat{l:stand-one-corner}
    получаем $U_i\subset H$. При $i=n$ имеем $Q(s)=U_i$, поэтому $i<n$. Таблица
    умножения дает
    \[
        e_{i+1,i}h\equiv
        a_{iv}e_{i+1,v}+a_{i,-v}e_{i+1,-v}\pmod{U_i},\qquad
        Q(s)=T(r_{i+1,v})+U_i.
    \]
    Из $U_i\subset H$ и второго равенства получаем $T(r_{i+1,v})\not\subset H$.
    Если $T(r_{i+1,-v})\subset H$, сравнение по модулю идеала $U_i$ дает
    $\mathbb{K}e_{i+1,v}\subset H$. По \crefDat{l:stand-one-corner} тогда
    $T(r_{i+1,v})\subset H$, противоречие.
\end{proof}

В отличие от типа $D_n$, в системах ${\Phi}$ типа $B_n$ и $C_n$ корни $r_{jv}$
и $r_{j, -v}$ всегда инцидентны, то есть в $R_{\Phi}$ один из идеалов
$T(r_{jv})$ и $T(r_{j, -v})$ лежит в другом.

\medskip

Далее идеал $T(r)$ обертывающей алгебры $R_{\Phi}$ классического типа
обозначаем при $r=r_{iv}$ через $T_{iv}$, кроме случая $v=1$ для типа $D_n$,
где
\[
    T_{i1}\coloneq T_{i,-1}+T(r_{i1}).
\]
Как показывают \cref{l:stand-one-corner,l:StructConst-vl90a_ru}, стандартны все
идеалы алгебры $R_{\Phi}$, лежащие в идеале вида $T_{i,-1}$ или "--- для типа
$B_n$ "--- в $T_{1 0}$.

Учитывая \crefVin{l:ReductBnCnDn} и \crefVin{th:Stand-An}, для алгебр
$R_{\Phi}$ типов $B_n$ и $C_n$ рассмотрим соответственно условия

\begin{equation}\label{IsomR-Bn}
    B_n:\ R_{\Phi}/T_{2,-1}\simeq NT(n+1,\mathbb{K});
\end{equation}
\begin{equation}\label{IsomR-Cn}
    C_n:\ R_{\Phi}/T_{2,-2}\simeq NT(n+1,\mathbb{K}).
\end{equation}

При выполнении соответствующего условия с точностью до перехода к
противоположной алгебре можно считать, что элементы $e_{jm}$ алгебры $R_{\Phi}$
при $j>m>0$ умножаются по обычным для матричных единиц правилам. Действительно,
для трех последовательных простых корневых элементов $x,y,z$ в ассоциативном
факторе имеем $(yx)z=y(xz)=0$. Если $yx\ne0$, то по определению специального
умножения $z(yx)\ne0$, откуда $zy\ne0$. Это согласует направление произведений
вдоль всей цепочки; остальные матричные произведения определяются
ассоциативностью. При таком выборе для ${\Phi}$ типа $B_n$, в силу
\crefRod{p:mult-enveloping},

\begin{equation}\label{IsomR-B3}
    e_{jm}e_{m0}=e_{j0},\qquad e_{m0}e_{j0}=- e_{j0}e_{m0}=c_{jm}
    e_{j,-m}\ \ (c_{jm}=\pm 1).
\end{equation}

\begin{remark}\label{remark:RBn-sign}
    Одновременная смена всех знаков $c_{jm}$ в \eqref{IsomR-B3} не меняет класс
    изоморфизма специальной обертывающей алгебры $R_\Phi$ типа $B_n$. Изоморфизм на
    алгебру с противоположными знаками $c_{jm}$ задает диагональная замена
    $e_{i,-j}\mapsto -e_{i,-j}$, тождественная на остальных базисных элементах. Эта
    замена меняет знак лишь у тех произведений, где элемент $e_{i,-j}$ стоит в
    результате, но не среди сомножителей, "--- а это только
    $e_{m0}e_{j0}=c_{jm}e_{j,-m}$ (у прочих произведений элементы $e_{i,-j}$ либо
    не участвуют, либо стоят и в сомножителе, и в результате).
\end{remark}

\begin{lemma}\label{l:Subsistem-B3}
    Алгебра $R_{\Phi}$ типа $B_3$ с условием \eqref{IsomR-Bn} стандартная. С
    точностью до изоморфизма и перехода к противоположной алгебре она определена
    однозначно, когда подалгебра $R(\Psi)$ для подсистемы $\Psi$ типа $A_3$ в
    $\Phi$ ассоциативная.
\end{lemma}

\begin{proof}
    При необходимости переходя к противоположной алгебре, считаем выполненными
    равенства \eqref{IsomR-B3}. По \crefDat{remark:RBn-sign} дополнительно считаем
    $e_{20}e_{10}=e_{2,-1}$. Все длинные корни системы корней $\Phi$ типа $B_3$
    образуют единственную подсистему $\Psi$ типа $A_3$. С учетом равенств
    \eqref{IsomR-B3}, условие \eqref{IsomR-Bn} легко дает стандартность алгебры
    $R_{\Phi}$.

    Базис в $\Psi$ образуют корни $r_{2,-1}$, $r_{32}$ и $r_{21}$, то есть
    \[
        \Psi= \Psi(r_{2,-1}, r_{32}, r_{21}).
    \]
    При выбранном умножении фактора имеем $e_{32} e_{21}= e_{31}$, $e_{21} e_{32} =
    0$. Поэтому стандартность подалгебры $R(\Psi)$ по \crefDat{l:stand-A3}
    равносильна равенствам
    \[
        e_{32}e_{2, - 1}=0,\ \ e_{2,-1}e_{32} = e_{3,-1}.
    \]
    Так как $[e_{20},e_{10}]=2e_{2,-1}$, тождество Якоби и \cref{def:RPhi} дают
    %
    \[ [e_{30},e_{10}]=[[e_{32},e_{20}],e_{10}]=[e_{32},[e_{20},e_{10}]]=
        -2e_{3,-1},\ \ e_{30}e_{10}=-e_{3,-1}.\]
    %
    Далее,
    %
    \[[e_{2,-1},e_{31}]=[e_{2,-1},[e_{32},e_{21}]]=[[e_{2,-1},e_{32}],e_{21}]=[e_{3,-1},e_{21}]
        %
        =\pm e_{3,-2},\]
    %
    то есть либо $e_{2,-1}e_{31} = e_{3,-2}$ и $e_{31}e_{2,-1} = 0$, либо
    $e_{2,-1}e_{31} = 0$ и $e_{31}e_{2,-1} = e_{3,-2}$.
    %
    Ассоциативность подалгебры $R(\Psi)$ допускает только первый случай (см. также
    \crefVin{remark:Stand-Non-Assoc-An}), так как
    \[
        e_{31}e_{2,-1}=(e_{32}e_{21})e_{2,-1}=e_{32}(e_{21}e_{2,-1})=0.
    \]
    При этом
    %
    \[e_{2,-1}(e_{32}e_{21})=e_{2,-1}e_{31}=e_{3,-2}=
        (e_{2,-1}e_{32})e_{21}=e_{3,-1}e_{21}. \]
    %
    Отсюда
    \[ [e_{30},e_{20}]=[e_{30},[e_{21},e_{10}]]=[e_{21},[e_{30},e_{10}]]=
        2e_{3,-2},\ \ e_{30}e_{20}=e_{3,-2}.\]
    %
    Таким образом, таблица умножения определена однозначно.
\end{proof}

\begin{lemma}\label{l:Subsistem-Bn}
    Пусть $R_n=R_{\Phi}$ есть стандартная обертывающая алгебра типа $B_n$ $(n > 3)$
    с условием \eqref{IsomR-Bn}. С точностью до перехода к противоположной алгебре
    имеем
    \[
        R_{\Phi}/T_{n,-1}\simeq RB_n(\mathbb{K})/T_{n,-1}.
    \]
\end{lemma}

\begin{proof}
    При необходимости переходя к противоположной алгебре, считаем выполненными
    равенства \eqref{IsomR-B3}. По \crefDat{remark:RBn-sign} дополнительно считаем
    $e_{20}e_{10}=e_{2,-1}$. В системе корней ${\Phi}$ типа $B_n$ $(n>3)$ выделим
    подсистемы $\Psi_{m,j,i,k}$ типа $B_4$ с базисом $\{r_{m0}, r_{jm}, r_{ij},
    r_{ki}\}$. Корни $r_{j,-m}$, $r_{ij}$, $r_{ki}$ образуют базис подсистемы
    $\Psi$ типа $A_3$, то есть
    \[
        \Psi=\Psi(r_{j,-m},r_{ij},r_{ki}),\qquad 1\leq m<j<i<k\leq n.
    \]

    Пусть сначала $i=j+1$. Если $e_{ij}e_{j,-m}=0$, то идеал, порожденный
    $e_{j,-m}+e_{ki}$, нестандартен: поскольку $r_{ij}$ прост, проекция этого
    идеала на $\mathbb{K}e_{i,-m}+\mathbb{K}e_{kj}$ совпадает с
    $\mathbb{K}(e_{i,-m}+e_{kj})$. Следовательно, $e_{ij}e_{j,-m}=e_{i,-m}$. Для
    $i>j+1$ используем разложение $e_{ij}=[e_{i,i-1},e_{i-1,j}]$ и тождество Якоби.
    Индукция по $i-j$ дает
    \[
        e_{jv}e_{ij}=0,\quad e_{ij}e_{jv}=e_{iv}=[e_{ij},e_{jv}]
        \qquad(n>i>j>|v|=m>0).
    \]

    Корни $r_{i,-m}$ и $r_{jm}$ инцидентны в системе $\Psi_{m,j,i,k}$, однако в ее
    подсистеме типа $D_4$ длинных корней не инцидентны. Вместе с $r_{kj}$ они
    образуют базис подсистемы $\Psi=\Psi(r_{i,-m},r_{jm},r_{kj})$ типа $A_3$. При
    $j=m+1$ тот же довод с идеалом, порожденным $e_{i,-m}+e_{kj}$, дает
    $e_{jm}e_{i,-m}=e_{i,-j}$. Для $j>m+1$ применяем тождество Якоби к разложению
    $e_{jm}=[e_{j,j-1},e_{j-1,m}]$. Индукцией по $j-m$ получаем
    \[
        e_{i,-m}e_{jm}=0,\quad e_{jm}e_{i,-m}=e_{i,-j}=[e_{jm},e_{i,-m}]
        \qquad(m<j<i<n).
    \]
    Еще одно применение тождества Якоби дает
    \[
        e_{i,-j}=[e_{jm},e_{i,-m}]=[e_{jm},[e_{ij},e_{j,-m}]]
        =[e_{j,-m},[e_{ij},e_{jm}]]=[e_{j,-m},e_{im}],
    \]
    откуда $e_{j,-m}e_{im}=e_{i,-j}$ и $e_{im}e_{j,-m}=0$.

    Так как $[e_{20},e_{10}]=2e_{2,-1}$, при $2<j<n$ имеем
    \[
        [e_{j0},e_{10}]=[[e_{j2},e_{20}],e_{10}]
        =[e_{j2},[e_{20},e_{10}]]=2e_{j,-1},\quad e_{j0}e_{10}=e_{j,-1}.
    \]
    Для корней $r_{i,-j}$ при $1<j<i<n$ также получаем
    \[
        [e_{i0},e_{j0}]=[e_{i0},[e_{j1},e_{10}]]
        =[e_{j1},[e_{i0},e_{10}]]=2e_{i,-j},\quad e_{i0}e_{j0}=e_{i,-j}.
    \]

    Поскольку подсистема $\Psi_{m,j,i,k}$ при $i<k\leq n$ выбирается произвольно,
    таблицы умножения алгебр $R_\Phi$ и $RB_n(\mathbb{K})$ совпадают по модулю
    идеала $T_{n,-1}$.
\end{proof}

\begin{remark}\label{remark:graph-automorphisms}

    \cite{vl90a_ru}), алгебра Ли $N\Phi (\mathbb{K})$ допускает графовый

    автоморфизм $\theta$ алгебры Ли $N\Phi(\mathbb{K})$ порядка 2 определяется

\end{remark}

\begin{lemma}\label{l:Bn-factor-from-embedding}
    Пусть $R_n$ и $R_{n+1}$ "--- специальные обертывающие алгебры типов $B_n$ и
    $B_{n+1}$, где $n\geq3$. Если $R_{n+1}$ стандартна и существует вложение
    $R_n\hookrightarrow R_{n+1}$, сохраняющее корневые компоненты при естественном
    включении $B_n\subset B_{n+1}$, то $R_n/T_{2,-1}\simeq NT(n+1,\mathbb{K})$.
\end{lemma}

\begin{proof}
    Положим $\overline R_m=R_m/T_{2,-1}$ при $m=n,n+1$. Алгебры $\overline R_m$
    "--- специальные обертывающие алгебры типов $A_m$, корневые порядки которых
    совпадают с ограничениями порядков $B_m$.

    Пусть $H$ "--- произвольный идеал кольца $\overline R_{n+1}$, а $\widetilde H$
    "--- его прообраз в $R_{n+1}$. Из определения $T_{2,-1}=T(r_{2,-1})$ следует,
    что углам $H$ соответствуют в точности те углы $r$ идеала $\widetilde H$, для
    которых $e_r\notin T_{2,-1}$. Стандартность $\widetilde H$ влечет стандартность
    $H$, поэтому алгебра $\overline R_{n+1}$ стандартна.

    По \crefDat{th:Stand-An} алгебра $\overline R_{n+1}/\Ann(\overline R_{n+1})$
    ассоциативна, а аннулятор $\Ann(\overline R_{n+1})$ натянут на образ корневого
    вектора $e_{n+1,0}$. Композиция заданного вложения с естественной проекцией
    \[
        R_n\hookrightarrow R_{n+1}\longrightarrow
        \overline R_{n+1}/\Ann(\overline R_{n+1})
    \]
    имеет ядро $T_{2,-1}\subset R_n$, поскольку вложение сохраняет корневые
    компоненты, а корень $r_{n+1,0}$ не принадлежит $B_n$. Следовательно,
    $\overline R_n$ ассоциативна и, по \crefDat{th:Stand-Graded-An}, изоморфна
    $NT(n+1,\mathbb{K})$, то есть выполнено \eqref{IsomR-Bn}.
\end{proof}

Сейчас мы можем указать условия однозначности обертывающих алгебр $R_{\Phi}$
классических типов. Для типа $B_n$ справедлива следующая теорема.

% \begin{theorem}[{{\cite[теорема~5]{smj2023}}}]\label{th:Stand-Bn}
\begin{theorem}\label{th:Stand-Bn}
    %<*th:Stand-Bn>
    Если стандартные обертывающие алгебры $R_n=R_\Phi$ типа $B_n$ при
    $n=3,4,\ldots$ допускают вложения $R_n\hookrightarrow R_{n+1}$, сохраняющие
    корневые компоненты при естественных включениях $B_n\subset B_{n+1}$, то либо
    $R_n\simeq\nobreak RB_n(\mathbb{K})$ при всех $n$, либо
    $R_n^{(op)}\simeq\nobreak RB_n(\mathbb{K})$ при всех $n$.
    %</th:Stand-Bn>
\end{theorem}

\begin{proof}
    По \crefDat{l:Bn-factor-from-embedding} условие \eqref{IsomR-Bn} выполнено при
    всех $n$.

    Вложения сохраняют корневые компоненты, а значит, и то, какое из произведений
    $e_{21}e_{10}$ и $e_{10}e_{21}$ ненулевое. Заменив при необходимости все $R_n$
    противоположными алгебрами, считаем выполненными равенства \eqref{IsomR-B3} при
    всех $n$.

    При любом $n\geq 3$ образ $R_n$ в $R_{n+1}$ натянут на элементы $e_{iv}$ с
    $0\leq |v|<i\leq n$ и пересекается с идеалом $T_{n+1,-1}$ по нулю. Поэтому
    естественная проекция $R_{n+1}\to R_{n+1}/T_{n+1,-1}$ инъективна на этом
    образе. Из доказательства \crefRod{l:Subsistem-Bn} для $R_{n+1}$ следует, что
    умножение на этой проекции совпадает с умножением в $RB_n(\mathbb{K})$ с
    точностью до выбора знака $c=\pm 1$. Ввиду \crefRod{remark:RBn-sign}, этот знак
    не влияет на класс изоморфизма алгебры.
\end{proof}

Естественные вложения $RD_n(\mathbb{K})\hookrightarrow RD_{n+1}(\mathbb{K})$

\section{Ассоциативные точные обертывающие алгебры}\label{sec:associative-envelopes}

Напомним некоторые сведения из теории некоммутативных колец.

Ассоциативное кольцо называется \emph{(правым) артиновым}, если всякая строго
убывающая последовательность его правых идеалов конечна
\cite[\S~1.3]{herstein-ru}. Ясно, что всякая конечномерная $\mathbb{K}$-алгебра
с единицей является артиновой. Для артинова кольца $R$ \emph{радикал Джекобсона
    $J(R)$} совпадает с радикалом Веддерберна, т.е. является его наибольшим
нильпотентным идеалом \cite[лемма~1.2.2, теорема~1.3.1]{herstein-ru}.

Артиново кольцо $R$ называем \emph{полупростым}, если $J(R)=0$. Факторкольцо
$R/J(R)$ артинова кольца $R$ также артиново и имеет нулевой радикал
\cite[теорема~1.2.4]{herstein-ru}, поэтому полупросто.

Через $M_n(D)$ обозначим полное кольцо $(n \times n)$-матриц над ассоциативным
кольцом с делением (телом) $D$.

\begin{theorem}[{{\cite[теорема~2.1.7]{herstein-ru}}}]\label{th:wedderburn-artin}
    Пусть $R$ "--- полупростое кольцо. Тогда $ R \simeq \prod_{i=1}^r M_{n_i}(D_i)
    $
    %
    для подходящих тел $D_1,\dots,D_r$ и положительных целых чисел $n_1,\dots,n_r$.
\end{theorem}

Далее нам потребуется \emph{функтор поднятия поля скаляров} \cite[\S\
4.1.7]{linalg1986}. Пусть $\mathbb{K}$ "--- подполе поля $\mathbb{E}$, $R$ "---
$\mathbb{K}$-алгебра. Рассмотрим тензорное произведение векторных
$\mathbb{K}$-пространств $R\otimes_{\mathbb{K}}\mathbb{E}$. Это
$\mathbb{K}$-пространство порождается элементами $r\otimes a$ ($r\in R$,
$a\in\mathbb{E}$), называемыми \emph{чистыми тензорами}, с определяющими
соотношениями $\mathbb{K}$-билинейности. Введем на
$R\otimes_{\mathbb{K}}\mathbb{E}$ структуру векторного пространства над
$\mathbb{E}$, определив умножение на скаляры $a \in \mathbb{E}$ формулой
\[
    a(r \otimes b) = r \otimes (ab)\quad (r \in R,\ a,b \in \mathbb{E}).
\]
Структуру алгебры над $\mathbb{E}$ задаем умножением на чистых тензорах
формулой
\[
    (r\otimes a)(s \otimes b) = (rs) \otimes (ab)\quad (r,s \in R,\ a,b \in \mathbb{E}),
\]
с продолжением по билинейности (\cite[\S\ 5.4]{bresar2025}).

Через $\mathfrak{g}^j$ обозначаем члены нижнего центрального ряда алгебры Ли
$\mathfrak{g}$:
\[
    \mathfrak{g}^1=\mathfrak{g},\qquad
    \mathfrak{g}^{j+1}=[\mathfrak{g},\mathfrak{g}^j].
\]

\begin{lemma}\label{l:extension-scalars-nilpotent-lie}
    Пусть $R$ "--- ассоциативная $\mathbb{K}$-алгебра, $R^{(-)}$ нильпотентна,
    $\mathbb{E}$ "--- расширение поля $\mathbb{K}$. Тогда $\mathbb{E}$-алгебра Ли
    $(R\otimes_{\mathbb{K}}\mathbb{E})^{(-)}$ нильпотентна.
\end{lemma}
\begin{proof}
    Скобка Ли в $(R\otimes_{\mathbb{K}}\mathbb{E})^{(-)}$ на чистых тензорах имеет
    вид
    \[
        [r\otimes a,s\otimes b]=[r,s]\otimes(ab),
    \]
    так что
    $(R\otimes_{\mathbb{K}}\mathbb{E})^{(-)}=R^{(-)}\otimes_{\mathbb{K}}\mathbb{E}$
    как алгебры Ли над $\mathbb{E}$. Пусть $R^{(-)}$ нильпотентна класса $m$, то
    есть $(R^{(-)})^{m+1}=0$. Тогда
    \[
        \left(R^{(-)}\otimes_{\mathbb{K}}\mathbb{E}\right)^{m+1} =
        \left(R^{(-)}\right)^{m+1}\otimes_{\mathbb{K}}\mathbb{E}=0.\qedhere
    \]
\end{proof}

\begin{lemma}[{{\cite[следствие~1 к теореме~4.3.3]{herstein-ru}}}]\label{l:csa-splitting-field}
    Пусть $A$ "--- конечномерная центральная простая ассоциативная
    $\mathbb{K}$-алгебра. Тогда существует конечное сепарабельное расширение полей
    $\mathbb{E}/\mathbb{K}$ такое, что алгебра $A\otimes_{\mathbb{K}}\mathbb{E}
    \simeq M_n(\mathbb{E})$ для некоторого $n\ge 1$.
\end{lemma}
\begin{lemma}\label{l:lie-nilpotent-semisimple-commutative}
    Пусть $R$ "--- конечномерная полупростая ассоциативная $\mathbb{K}$-алгебра.
    Если $R^{(-)}$ нильпотентна, то $R$ коммутативна.
\end{lemma}
\begin{proof}
    По теореме Веддерберна--Артина имеем разложение
    \[
        R \simeq \prod_{i=1}^r M_{n_i}(D_i).
    \]
    Так как проекции $R^{(-)}\to M_{n_i}(D_i)^{(-)}$ являются эпиморфизмами алгебр
    Ли, из нильпотентности $R^{(-)}$ следует нильпотентность $M_{n_i}(D_i)^{(-)}$
    для всех $i$.

    Покажем, что $n_i=1$. Действительно, если $n_i\ge 2$, то в $M_{n_i}(D_i)$
    определены матричные единицы $e_{pq}$ и $[e_{11},e_{12}]=e_{12}$.
    Следовательно, $\operatorname{ad}e_{11}$ не является нильпотентным оператором,
    что противоречит нильпотентности $M_{n_i}(D_i)^{(-)}$. Значит, $n_i=1$ для всех
    $i$ и потому
    \[
        R\simeq \prod_{i=1}^r D_i.
    \]

    Остается показать, что каждое тело $D_i$ коммутативно. Зафиксируем $D = D_i$ и
    обозначим через $F = Z(D)$ его центр. Тогда $D$ является центральной простой
    $F$-алгеброй. По \crefDat{l:csa-splitting-field} существует конечное
    сепарабельное расширение полей $E/F$, для которого $ D\otimes_F E\simeq M_m(E)
    $ при некотором $m\ge 1$. По \crefDat{l:extension-scalars-nilpotent-lie}, из
    нильпотентности $D^{(-)}$ следует нильпотентность $(D\otimes_F E)^{(-)}\simeq
    M_m(E)^{(-)}$, но при $m\ge 2$ алгебра Ли $M_m(E)^{(-)}$ не нильпотентна по уже
    рассмотренному примеру с $e_{11}$ и $e_{12}$. Значит, $m=1$ и потому
    $D\otimes_F E\simeq E$. Отсюда следует, что $D=F$, то есть $D$ коммутативно.

    Таким образом, все сомножители $D_i$ коммутативны, и потому алгебра $R$
    коммутативна.
\end{proof}

\begin{lemma}[{{\cite[теорема~1.3.2]{herstein-ru}}}]\label{l:idempotent-non-nilpotent-ideal}
    Пусть $R$ "--- артиново ассоциативное кольцо и $I$ "--- ненулевой
    ненильпотентный правый идеал в $R$. Тогда $I$ содержит ненулевой идемпотент.
\end{lemma}

\begin{lemma}\label{l:assoc-envelope-nilpotent}
    Пусть $\mathfrak{g}$ "--- конечномерная нильпотентная алгебра Ли такая, что
    $\dim Z(\mathfrak{g})=1$ и $Z(\mathfrak{g})\subset
    [\mathfrak{g},\mathfrak{g}]$. Тогда всякая ассоциативная точная обертывающая
    алгебра алгебры Ли $\mathfrak{g}$ является нильпотентной.
\end{lemma}
\begin{proof}
    Пусть $R$ "--- ассоциативная точная обертывающая алгебра для $\mathfrak{g}$, и
    предположим, что $R$ не нильпотентна. Применяя
    \crefVin{l:idempotent-non-nilpotent-ideal} к правому идеалу $R$ в конечномерной
    унитализации $\mathbb{K}1\oplus R$, получаем ненулевой идемпотент $e\in R$.

    Обозначим через $L_e$ и $R_e$ левое и правое умножение на $e$ в $R$. Так как
    $e^2=e$, имеем $L_e^2=L_e$, $R_e^2=R_e$ и $L_eR_e=R_eL_e$. Поэтому
    \[
        (\operatorname{ad}e)^3=(L_e-R_e)^3=L_e-R_e=\operatorname{ad}e.
    \]
    С другой стороны, алгебра Ли $R^{(-)}\simeq \mathfrak{g}$ нильпотентна, значит
    $\operatorname{ad}e$ "--- нильпотентный оператор. Следовательно,
    $\operatorname{ad}e=0$, и потому $e$ лежит в центре $R$.

    Отображение $L_e\colon R\to R$ "--- ненулевой идемпотентный эндоморфизм,
    поэтому
    \[
        R=eR\oplus\ker L_e.
    \]
    Так как $e$ центральный, то и $eR$, и $\ker L_e$ являются двусторонними
    идеалами алгебры $R$. Следовательно,
    \[
        R^{(-)}=(eR)^{(-)}\oplus (\ker L_e)^{(-)}
    \]
    есть прямая сумма лиевых идеалов. Если бы оба слагаемых были ненулевыми, то,
    как нильпотентные алгебры Ли, они имели бы ненулевые центры. Тогда $\dim
    Z(R^{(-)})\ge 2$, что противоречит $\dim Z(\mathfrak{g})=1$. Значит, $\ker
    L_e=0$, и потому $R=eR$.

    Так как $e$ центральный и $R=eR$, то для любого $x\in R$ имеем $x=ey=ye$ для
    некоторого $y\in R$, откуда $ex=xe=x$. Следовательно, $e$ есть единица алгебры
    $R$.

    Факторалгебра $R/J(R)$ "--- конечномерная полупростая ассоциативная алгебра.
    Так как $R^{(-)}\simeq \mathfrak{g}$ нильпотентна, то и $(R/J(R))^{(-)}$ тоже
    нильпотентна. Тогда, по \crefDat{l:lie-nilpotent-semisimple-commutative},
    алгебра $R/J(R)$ должна быть коммутативной, и потому $ [R,R]\subset J(R) $. В
    частности, $1\notin [R,R]$.

    С другой стороны, при изоморфизме $R^{(-)}\simeq \mathfrak{g}$ элемент $1$
    лежит в $Z(\mathfrak{g})$. Так как $Z(\mathfrak{g})\subset
    [\mathfrak{g},\mathfrak{g}]$, получаем $1\in [R,R]$, противоречие.
    Следовательно, $R$ нильпотентна.
\end{proof}

\begin{remark}\label{remark:assoc-envelope-center}
    В \crefPred{l:assoc-envelope-nilpotent} условие $\dim Z(\mathfrak{g})=1$
    избыточно. Пусть $R$ "--- ассоциативная точная обертывающая алгебры
    $\mathfrak{g}$. Если $R$ не нильпотентна, то, как в доказательстве леммы, в $R$
    найдется ненулевой центральный идемпотент $e$. Гомоморфизм $R\to eR$, $x\mapsto
    ex$, переводит включение $e\in[R,R]$ во включение $e\in[eR,eR]$. По
    \crefDat{l:lie-nilpotent-semisimple-commutative}, факторалгебра $(eR)/J(eR)$
    коммутативна, поэтому $[eR,eR]\subset J(eR)$. Однако $e$ "--- единица алгебры
    $eR$, поэтому $e\notin J(eR)$, что противоречит включению $e\in[eR,eR]\subset
    J(eR)$.
\end{remark}

Для типа $A_n$ алгебра Ли $N\Phi(\mathbb{K})$ изоморфна
$NT(n+1,\mathbb{K})^{(-)}$ с центром $\mathbb{K}e_{n+1,1}$. При $n\ge2$ имеем
$e_{n+1,1}=[e_{n+1,2},e_{2,1}]$, поэтому из
\crefRod{l:assoc-envelope-nilpotent} следует

\begin{corollary}\label{corollary:assoc-envelope-nilpotent-An}
    Пусть $R$ "--- ассоциативная точная обертывающая $\mathbb{K}$-алгебра для
    алгебры Ли $N\Phi(\mathbb{K})$ типа $A_n$ $(n \ge 2)$. Тогда $R$ нильпотентна.
\end{corollary}

Для ассоциативной алгебры $R$ через $R^j$ обозначаем линейную оболочку
произведений $j$ элементов алгебры $R$.

\begin{lemma}\label{l:assoc-squares-third}
    Пусть $R$ --- нильпотентная ассоциативная алгебра над полем $\mathbb{K}$
    характеристики, отличной от $2$. Если $x^2\in(R^{(-)})^3$ для каждого $x\in R$,
    то $R^3=0$.
\end{lemma}
\begin{proof}
    Положим $I=(R^{(-)})^3$. Поляризация \cite[п.~2.3.9]{linalg1986} дает
    \[
        x\circ y=\frac12\bigl((x+y)^2-x^2-y^2\bigr)\in I,
    \]
    и по тождеству
    \[
        xyz=x\circ(yz)-y\circ(zx)+z\circ(xy)
    \]
    имеем $R^3\subset I$. Обратное включение следует из определения $I$, поэтому
    $R^3=I$. Далее,
    \[
        [x,yz]=2\bigl((x\circ y)z-y(x\circ z)\bigr)\in R^4.
    \]
    Значит, $I=[R,[R,R]]\subset[R,R^2]\subset R^4$, откуда $R^3=R^4$.
    Нильпотентность алгебры $R$ влечет $R^3=0$.
\end{proof}

Через $C_{\mathfrak{g}}(x)=\{y\in\mathfrak{g}\mid[x,y]=0\}$ обозначаем
централизатор элемента $x\in\mathfrak{g}$.

\begin{lemma}\label{l:assoc-centralizer-squares}
    Пусть $R$ --- конечномерная нильпотентная ассоциативная алгебра над полем
    $\mathbb{K}$ характеристики, отличной от $2$, а $I$ --- идеал алгебры Ли
    $R^{(-)}$ коразмерности не меньше $2$. Предположим, что существует $v\in R$,
    для которого при каждом $x\in R$ включение
    \[
        C_{R^{(-)}}(x+tv)\subset\mathbb{K}(x+tv)+I
    \]
    выполняется для бесконечного числа $t\in\mathbb{K}$. Тогда $x^2\in I$ для всех
    $x\in R$.
\end{lemma}
\begin{proof}
    Пусть $\pi\colon R\to R/I$ --- каноническая проекция. Зафиксируем $x\in R$. В
    ассоциативной алгебре каждый элемент коммутирует со своим квадратом, поэтому
    векторы $\pi((x+tv)^2)$ и $\pi(x+tv)$ линейно зависимы для бесконечного числа
    $t\in\mathbb{K}$. В любом базисе $R/I$ миноры второго порядка матрицы координат
    векторов $\pi((x+tv)^2)$ и $\pi(x+tv)$ являются многочленами от $t$. Каждый
    минор обращается в нуль при бесконечном числе значений $t$, поэтому
    тождественно равен нулю. Подставляя $t=0$, получаем
    $\pi(x^2)\in\mathbb{K}\pi(x)$ при $x\notin I$.

    Для $y\in I$ определим линейное отображение $T_y\colon R\to R/I$ формулой
    $T_y(x)=\pi(yx)$. Если $x\notin I$, то
    \[
        4T_y(x)=\pi\bigl((x+y)^2-(x-y)^2\bigr)\in\mathbb{K}\pi(x),
    \]
    поскольку $[x,y]\in I$. Для $z\in I$ имеем
    $T_y(z)=T_y(x+z)-T_y(x)\in\mathbb{K}\pi(x)$ при любом $x\notin I$. Так как
    $\dim(R/I)\geq2$, отсюда $T_y(z)=0$. Следовательно, левое умножение на $y$
    сохраняет $I$. Индуцированный на $R/I$ оператор левого умножения на $y$
    нильпотентен и сохраняет каждую прямую, поэтому равен нулю. Получаем $yR\subset
    I$, а условие $[R,I]\subset I$ дает $Ry\subset I$. Значит, $I$ --- идеал
    ассоциативной алгебры $R$.

    В нильпотентной факторалгебре $R/I$ равенство $\bar x^2=\lambda\bar x$ для
    ненулевого элемента $\bar x$ влечет $\lambda=0$, поэтому все квадраты в $R$
    лежат в $I$.
\end{proof}

\begin{proposition}\label{p:non-A-no-associative-envelope}
    Если тип системы корней $\Phi$ отличен от $A_n$ и $\Char\mathbb{K}\neq2$,
    причем для типа $G_2$ также $\Char\mathbb{K}\neq3$, то алгебра Ли
    $N\Phi(\mathbb{K})$ не имеет ассоциативной точной обертывающей алгебры.
\end{proposition}
\begin{proof}
    При расширении скаляров точная обертывающая сохраняется, поэтому считаем поле
    бесконечным. Пусть $R^{(-)}=\mathfrak{n}=N\Phi(\mathbb{K})$, где $R$
    ассоциативна. Запишем $\Pi=\{\alpha_1,\ldots,\alpha_n\}$ и положим
    \[
        \mathfrak{n}_j=\sum_{\mathclap{\height(r)=j}}\mathbb{K} e_r,\qquad
        e_i=e_{\alpha_i},\qquad e=\sum_{i=1}^n e_i.
    \]
    При указанных ограничениях векторы $e_i$ порождают $\mathfrak{n}$
    \cite[лемма~3.6.2, теорема~4.2.1]{carter72}, поэтому
    $\mathfrak{n}^j=\bigoplus_{k\geq j}\mathfrak{n}_k$.

    Для каждого простого корня $s\in\Pi$ выберем $r\in\Pi$, соединенный с $s$
    ребром диаграммы Дынкина. Тогда $[e_r,e_s]=N_{rs}e_{r+s}$, где $N_{rs}\neq0$.
    Если элемент $h=\sum_{\beta\in\Phi^+}d_\beta e_\beta$ централен, то коэффициент
    при $e_{r+s}$ в коммутаторе $[e_r,h]$ равен $N_{rs}d_s$. Из равенства
    $[e_r,h]=0$ получаем $N_{rs}d_s=0$, откуда $d_s=0$. Следовательно, центр
    алгебры $\mathfrak{n}$ содержится в $[\mathfrak{n},\mathfrak{n}]$, поэтому по
    \crefDat{l:assoc-envelope-nilpotent} с учетом
    \crefRod{remark:assoc-envelope-center} алгебра $R$ нильпотентна.

    Покажем, что
    \begin{equation}\label{eq:non-A-low-degree-kernels}
        \ker(\ad e|_{\mathfrak{n}_1})=\mathbb{K} e,\qquad
        \ker(\ad e|_{\mathfrak{n}_2})=0.
    \end{equation}
    Равенство $[e,\sum_i b_i e_i]=0$ дает $b_i=b_j$ для любых соседних
    вершин $i,j$ диаграммы Дынкина. По связности диаграммы Дынкина все
    коэффициенты $b_i$ равны, что доказывает первое равенство
    в~\eqref{eq:non-A-low-degree-kernels}.

    Базис $\mathfrak{n}_2$ образуют коммутаторы $[e_i,e_j]$, где $i<j$ и вершины
    $\alpha_i,\alpha_j$ соединены ребром диаграммы Дынкина. Пусть
    \[
        z=\sum_{\substack{i<j\\i,j\text{ соседние}}}a_{ij}[e_i,e_j],
        \qquad [e,z]=0.
    \]
    Положим $a_{ji}=-a_{ij}$. Если $i,j,k$ --- последовательные вершины пути в
    диаграмме, то по тождеству Якоби коэффициент при
    $e_{\alpha_i+\alpha_j+\alpha_k}$ в коммутаторе $[e,z]$ равен
    $\pm(a_{jk}-a_{ij})$. Следовательно, $a_{ij}=a_{jk}$. Если диаграмма имеет
    кратное ребро, обозначим его концы через $i,j$ так, чтобы
    $2\alpha_i+\alpha_j\in\Phi^+$. Коэффициент при $e_{2\alpha_i+\alpha_j}$ в
    $[e,z]$ равен $\pm2a_{ij}$, откуда $a_{ij}=0$ \cite[теорема~4.2.1]{carter72}. В
    типах $D,E$ для трех соседей $i,k,m$ вершины $j$ получаем
    \[
        a_{ji}+a_{jk}=a_{ji}+a_{jm}=a_{jk}+a_{jm}=0,
    \]
    поэтому $a_{ji}=a_{jk}=a_{jm}=0$. Из равенств $a_{ij}=a_{jk}$ и связности
    диаграммы следует $a_{ij}=0$ для каждого ребра. Значит, $z=0$, что доказывает
    второе равенство.

    Положим $I=\mathfrak{n}^3=\bigoplus_{j\geq3}\mathfrak{n}_j$ и через $U$
    обозначим множество элементов $N\Phi(\mathbb{K})$ с ненулевыми проекциями на
    все простые корневые подпространства $\mathbb{K} e_\alpha$ $(\alpha\in\Pi)$,
    т.\:е.
    \[
        U=\left\{\sum_{r\in\Phi^+}c_r e_r\in\mathfrak{n}\ \middle|\
        c_\alpha\neq0\text{ для всех }\alpha\in\Pi\right\}.
    \]
    Через $x_j$ обозначим проекцию элемента $x\in\mathfrak{n}$ на подпространство
    $\mathfrak{n}_j$ в разложении $\mathfrak{n}=\bigoplus_{k\geq1}\mathfrak{n}_k$.
    Для $x=\sum_{r\in\Phi^+}c_r e_r\in U$ определим диагональный автоморфизм $D_x$
    формулой
    \[
        D_x(e_r)=\Bigl(\prod_i c_{\alpha_i}^{m_i}\Bigr)e_r,
        \qquad r=\sum_i m_i\alpha_i\in\Phi^+.
    \]
    Тогда $D_x(e)=x_1$ и $D_x(\mathfrak{n}_j)=\mathfrak{n}_j$, поэтому
    равенства~\eqref{eq:non-A-low-degree-kernels} справедливы и при замене $e$ на
    $x_1$. Если $x\in U$, $y\in\mathfrak{n}$ и $[x,y]=0$, то из компоненты высоты
    $2$ получаем $[x_1,y_1]=0$, откуда $y_1=\lambda x_1$. Компонента высоты $3$
    равенства $[x,y-\lambda x]=0$ дает $[x_1,y_2-\lambda x_2]=0$, откуда
    $y_2=\lambda x_2$. Следовательно,
    \[
        C_{\mathfrak{n}}(x)\subset\mathbb{K} x+I\qquad(x\in U).
    \]
    При каждом $x\in\mathfrak{n}$ имеем $x+te\in U$ для всех $t\in\mathbb{K}$,
    кроме конечного числа значений. Поскольку $\dim(\mathfrak{n}/I)=2n-1\geq3$, по
    \crefDat{l:assoc-centralizer-squares} все квадраты в $R$ лежат в $I$. По
    \crefDat{l:assoc-squares-third} получаем $R^3=0$, откуда $\mathfrak{n}^3=0$. Но
    оператор $\ad e$ инъективен на ненулевом подпространстве $\mathfrak{n}_2$,
    поэтому $0\neq[e,\mathfrak{n}_2]\subset\mathfrak{n}^3$. Противоречие.
\end{proof}

\chapter[Градуированные точные обертывающие алгебры нильтреугольных алгебр Шевалле]{Градуированные точные обертывающие алгебры\\нильтреугольных алгебр Шевалле}\label{ch:graded}

\section{Градуированные алгебры}\label{sec:graded-algebras}
Базис Шевалле алгебры $\mathfrak{L}(\Phi,{\mathbb{K}})$ ранга $n$ задает на ней
стандартную \emph{$\mathbb{Z}\Phi$-градуировку Картана} целочисленной решеткой
корней $\mathbb{Z}\Phi \simeq \mathbb{Z}^n$. Компонента с нулевой степенью в
этой градуировке есть подалгебра $H$, натянутая на картановские элементы базиса
Шевалле, а остальные нетривиальные компоненты являются одномерными
$H$-инвариантными подалгебрами в $\mathfrak{L}(\Phi,{\mathbb{K}})$. В
частности, на подалгебре $N\Phi(\mathbb{K})$ все нетривиальные компоненты
индуцированной градуировки являются одномерными.

Как показывает \cref{th:Stand-Graded-An} в \S~\ref{sec:graded-An}, алгебра
$NT(n, \mathbb{K})$ является единственной ассоциативной точной обертывающей
алгеброй для алгебры Ли $N\Phi(\mathbb{K})$ типа $A_{n-1}$ $(n \geq 4)$,
допускающей градуировку, согласованную с градуировкой Картана.

\medskip

Пусть $G$ "--- абелева группа. Будем говорить, что на алгебре $A$ задана
\emph{$G$\nobreakdash-градуировка $\Gamma$}, если $A$ представлена в виде
прямой суммы подпространств
\[
    \Gamma\colon A = \bigoplus_{g \in G}A_g
\]
и включение $A_{g}A_{h} \subset A_{g+h}$ выполняется для всех $g, h \in G$.
\emph{Носителем} градуировки $\Gamma$ называем множество $\Supp\Gamma \coloneq
\{g \in G \mid A_g \neq 0\}$.

\medskip

Градуировки алгебр Ли абелевыми группами систематически изучались
в~\cite{patera1989}, см. также \cite{kochetov2013, latyshev1988}.

\medskip

Произвольный элемент $a \in A_g$ будем называть \emph{однородным степени $g$} и
писать $\mathrm{deg}_\Gamma(a) = g$. Отметим, что при таком определении $0 \in
A$ является однородным элементом произвольной степени. Подпространство $A_g$
называется \emph{однородной компонентой степени $g$}. При фиксированной
градуировке $\Gamma$ алгебру $A$ будем называть \emph{градуированной}.

Элемент $a \in A$ однозначно представим в виде суммы $\sum_{g \in G}a_g$, где
элементы $a_g \in A_g$ называются \emph{однородными компонентами $a$}.
Подалгебра $R \subset A$ называется градуированной, если она содержит
однородные компоненты всех своих элементов, т.\;е.
\[ R = \bigoplus_{g \in G} (A_g \cap R).\]
Если $I$ является градуированным идеалом алгебры $A$, то фактор-алгебра $A/I$
наследует градуировку~\cite[\S~1.3]{moody95}, задаваемую формулой
\[
    (A/I)_g\coloneq (A_g+I)/I = \{a+I \mid a\in A_g\}.
\]

Рассмотрим две градуировки алгебры $A$
\begin{equation*}
    \Gamma_1 \colon A = \bigoplus_{g \in G}A_g
    \quad \text{и} \quad
    \Gamma_2 \colon A = \bigoplus_{h \in H}A_h.
\end{equation*}
Градуировку $\Gamma_2$ называем \emph{уточнением} градуировки $\Gamma_1$, если
каждая ее ненулевая однородная компонента содержится в компоненте $\Gamma_1$.
Уточнение называем \emph{собственным}, если одно из этих включений строгое
\cite[определение~1.24]{kochetov2013}.

Градуировка $\Gamma$ называется \emph{тонкой}, если она не допускает
собственных уточнений \cite[определение 2]{patera1989}.

\begin{lemma}\label{l:graded-fine-existence}
    У всякой градуировки конечномерной алгебры существует тонкое уточнение.
\end{lemma}
\begin{proof}
    При каждом собственном уточнении число ненулевых однородных компонент
    увеличивается и не превосходит $\dim A$. Поэтому последовательное уточнение за
    конечное число шагов приводит к тонкой градуировке.
\end{proof}

Напомним, что $\Ann(A)=\{a\in A\mid aA=Aa=0\}$. Для произвольной алгебры $A$
через $A^j$ обозначаем линейную оболочку всех произведений $j$ элементов
алгебры $A$ при всевозможных расстановках скобок. Алгебра $A\neq0$ называется
\emph{нильпотентной} класса нильпотентности $m$, если $A^m\neq0$ и $A^{m+1}=0$.
В частности, $A^m\subset\Ann(A)\neq0$.

\begin{lemma}\label{l:graded-ann}
    Аннулятор $\Ann(A)$ градуированной алгебры $A=\bigoplus_{g\in G}A_g$ является
    ее градуированным идеалом.
\end{lemma}
\begin{proof}
    Пусть $x=\sum_g x_g\in\Ann(A)$, где $x_g\in A_g$. Для любого $y\in A_h$ имеем
    \[
        xy=\sum_g x_gy=0.
    \]
    При различных $g$ слагаемые $x_gy\in A_{g+h}$ лежат в различных однородных
    компонентах, поэтому $x_gy=0$ для всех $g$. Так как $h\in G$ и $y\in A_h$
    произвольны, получаем $x_gA=0$. Аналогично из $Ax=0$ следует $Ax_g=0$. Значит,
    каждая компонента $x_g$ принадлежит $\Ann(A)$, что и требовалось.
\end{proof}

Рассмотрим две градуированные алгебры
\[
    \Gamma_1\colon A=\bigoplus_{g\in G}A_g
    \quad\text{и}\quad
    \Gamma_2\colon B=\bigoplus_{h\in H}B_h.
\]
Градуировки $\Gamma_1$ и $\Gamma_2$ называются \emph{эквивалентными}, если
существуют изоморфизм алгебр $\phi\colon A\to B$ и биекция носителей
$\alpha\colon\Supp\Gamma_1\to\Supp\Gamma_2$ такие, что
${\phi(A_g)=B_{\alpha(g)}}$ для всех $g\in\Supp\Gamma_1$
\cite[определение~1.14]{kochetov2013}.

$\mathbb{Z}\Phi$-градуировку Ли-допустимой алгебры $R$ называем
\emph{согласованной с градуировкой Картана}, если существует сохраняющий
степени изоморфизм алгебр Ли $\phi\colon R^{(-)}\to N\Phi(\mathbb{K})$.

\section{Градуированные точные обертывающие алгебры типа \texorpdfstring{$A_n$}{Aₙ}}\label{sec:graded-An}
%
На алгебре $N\Phi(\mathbb{K})$ зафиксируем градуировку Картана $\Gamma_C$
относительно
% положительной части базиса Шевалле
ее корневого базиса $\{e_r\mid r\in\Phi^+\}$:
\begin{equation*}%\label{eq:An-Cartan-Grading}
    \Gamma_C\colon N\Phi(\mathbb{K})=
    \bigoplus_{r\in\Phi^+}\mathbb{K}e_r
    %= \bigoplus_{\mathclap{1\leq j<i\leq n+1}}\mathbb{K}e_{ij}
    .
\end{equation*}

\begin{lemma}\label{l:enveloping-roots-grading}
    %<*l:enveloping-roots-grading>
    Пусть $R$ "--- точная обертывающая $\mathbb{K}$-алгебра алгебры Ли
    $N\Phi(\mathbb{K})$. Если на $R$ задана градуировка $\Gamma$, согласованная с
    градуировкой Картана, то $R$ можно реализовать на линейном пространстве
    $N\Phi(\mathbb{K})$ так, что $e_re_s \in \mathbb{K}e_{r+s}$ при $r+s \in
    \Phi^{+}$, $e_re_s = 0$ при $r+s \not\in \Phi^{+}$, и коммутатор в $R^{(-)}$
    совпадает с коммутатором в $N\Phi(\mathbb{K})$.
    %</l:enveloping-roots-grading>
\end{lemma}
\begin{proof}
    Запишем заданную градуировку в виде
    \begin{equation*}
        \Gamma\colon R=\bigoplus_{g\in\mathbb{Z}\Phi}R_g.
    \end{equation*}
    По определению согласованности существует сохраняющий степени изоморфизм
    алгебр Ли $\phi\colon R^{(-)}\to N\Phi(\mathbb{K})$, поэтому
    $\phi(R_r)=\mathbb{K}e_r$ для $r\in\Phi^+$ и $R_g=0$ при
    $g\not\in\Phi^+$. Перенесем умножение $R$ на линейное пространство
    $N\Phi(\mathbb{K})$ посредством $\phi$. Включение $R_rR_s\subset R_{r+s}$
    дает требуемые условия на произведения $e_re_s$, а совпадение коммутаторов
    следует из того, что $\phi$ является изоморфизмом алгебр~Ли.
\end{proof}

Для системы корней $\Phi$ типа $A_n$ будем использовать ее стандартную
реализацию~\cite[таблица I]{bourbaki72}:
\begin{equation}\label{eq:An-Root-System}
    \begin{gathered}
        A_n=\{\epsilon_i-\epsilon_j \mid 1\leq i\neq j\leq n+1\},\qquad
        A_n^+=\{\epsilon_i-\epsilon_j \mid 1\leq j<i\leq n+1\},\\
        \Pi(A_n)=\{\alpha_i\coloneq \epsilon_{i+1}-\epsilon_i \mid 1\leq i\leq n\}.
    \end{gathered}
\end{equation}
%
В этой реализации базисные векторы $e_r=e_{\epsilon_i-\epsilon_j}$ алгебры
$N\Phi(\mathbb{K})$ удобно отождествлять с матричными единицами $e_{ij}$ в
$\mathfrak{sl}_{n+1}(\mathbb{K})$ (см., например, \cite[\S\ 11.2.1]{carter72}):
\begin{equation}\label{eq:An-matrix-realization}
    e_{\epsilon_i-\epsilon_j} \longleftrightarrow e_{ij} \qquad (1 \le j < i \le n+1).
\end{equation}

% \begin{theorem}[{{\cite[теорема~2]{jsfu2026}}}]\label{th:Stand-Graded-An}
\begin{theorem}\label{th:Stand-Graded-An}
    %<*th:Stand-Graded-An>
    Если ассоциативная точная обертывающая алгебра $R$ алгебры Ли
    $N\Phi(\mathbb{K})$ типа $A_n\ (n \ge 3)$ допускает градуировку, согласованную
    с градуировкой Картана, то $R \simeq NT(n+1,\mathbb{K})$.
    %</th:Stand-Graded-An>
\end{theorem}
\begin{proof}
    По \crefDat{l:enveloping-roots-grading} реализуем алгебру $R$ на пространстве
    $N\Phi(\mathbb{K})$ в базисе $\{e_{ij}\}$ из \eqref{eq:An-matrix-realization},
    сохраняя матричный коммутатор. Положим $x_i=e_{i+1,i}$ $(1\leq i\leq n)$.
    Корневая градуировка дает $x_ix_j=0$ при $|i-j|\neq1$.

    Разложим ненулевой вложенный коммутатор
    \[
        e_{n+1,1}=[x_n,[x_{n-1},\dots,[x_2,x_1]\dots]]
    \]
    в сумму произведений, содержащих элементы $x_1,\dots,x_n$ ровно по одному разу.
    В ненулевом произведении соседние индексы различаются на единицу, поэтому
    ненулевыми могут быть лишь $x_n\cdots x_1$ и $x_1\cdots x_n$. Хотя бы одно из
    этих произведений ненулевое. При необходимости заменим умножение в $R$ на
    $-yx$; такая замена сохраняет коммутатор, ассоциативность и корневую
    градуировку. Считаем теперь, что $x_n\cdots x_1\neq0$.

    Для $1\leq i\leq n-2$ имеем $x_{i+2}x_{i+1}x_i\neq0$, а значит, оба
    произведения $x_{i+2}x_{i+1}$ и $x_{i+1}x_i$ ненулевые. Одномерность корневых
    компонент дает скаляры $\lambda,\mu\in\mathbb{K}$ такие, что
    \[
        x_{i+1}x_{i+2}=\lambda x_{i+2}x_{i+1},\qquad
        x_ix_{i+1}=\mu x_{i+1}x_i.
    \]
    По ассоциативности и равенству $x_{i+2}x_i=0$ получаем
    \[
        \begin{aligned}
            \lambda x_{i+2}x_{i+1}x_i
             & =(x_{i+1}x_{i+2})x_i=x_{i+1}(x_{i+2}x_i)=0, \\
            \mu x_{i+2}x_{i+1}x_i
             & =x_{i+2}(x_ix_{i+1})=(x_{i+2}x_i)x_{i+1}=0.
        \end{aligned}
    \]
    Следовательно, $\lambda=\mu=0$. Поскольку $n\geq3$, такие тройки охватывают все
    соседние пары, поэтому $x_ix_{i+1}=0$ при $1\leq i<n$. Вместе с корневой
    градуировкой это дает $x_ix_j=0$ при $i\neq j+1$.

    Из равенств $e_{ij}=[x_{i-1},e_{i-1,j}]$ при $i>j+1$ теперь следует
    \[
        e_{ij}=x_{i-1}\cdots x_j\qquad(1\leq j<i\leq n+1).
    \]
    Для любых базисных элементов $e_{ij},e_{kl}$ получаем
    \[
        e_{ij}e_{kl}=(x_{i-1}\cdots x_j)(x_{k-1}\cdots x_l)
        =\begin{cases}
            e_{il}, & j=k,     \\
            0,      & j\neq k.
        \end{cases}
    \]
    Таким образом, получено матричное умножение. Если умножение заменялось на
    $-yx$, то исходная алгебра изоморфна $NT(n+1,\mathbb{K})^{(op)}$ посредством
    $x\mapsto-x$ и, по \crefDat{l:NT-opposite}, снова изоморфна
    $NT(n+1,\mathbb{K})$.
\end{proof}
\begin{remark}\label{remark:LieAdmissible-Nonunique-Assoc-A3}
    Если же в \crefPred{th:Stand-Graded-An} не требовать существования градуировки,
    согласованной с градуировкой Картана, то алгебра $R$, вообще говоря, не обязана
    быть изоморфной $NT(n+1,\mathbb{K})$.

    Действительно, в алгебре $NT(4,\mathbb{K})$ заменим умножение на новое $\ast$,
    полагая $x\ast y=xy$ для всех пар базисных элементов, кроме пары
    $(e_{32},e_{21})$, и дополнительно задавая
    \[
        e_{32}\ast e_{21}=e_{31}+e_{41},\qquad e_{21}\ast e_{32}=e_{41}.
    \]
    Тогда все коммутаторы базисных элементов совпадают с коммутаторами в
    $NT(4,\mathbb{K})$; значит, $(NT(4,\mathbb{K}),\ast)^{(-)}\simeq
    N\Phi(\mathbb{K})$ типа $A_3$. Ассоциативность умножения $\ast$ сохраняется,
    поскольку билинейная поправка $\delta(x,y)=x\ast y-xy$ принимает значения в
    аннуляторе $\mathbb{K}e_{41}$ и обращается в нуль, если хотя бы один из ее
    аргументов $x,y$ принадлежит $NT(4,\mathbb{K})^2$.

    Полученная ассоциативная алгебра неизоморфна $NT(4,\mathbb{K})$: размерность
    левого аннулятора является инвариантом. В $NT(4,\mathbb{K})$ имеем
    $\Ann_{\ell}=\langle e_{21},e_{31},e_{41}\rangle$ и $\dim\Ann_{\ell}=3$, тогда
    как в $(NT(4,\mathbb{K}),\ast)$ имеем $\Ann_{\ell}=\langle
    e_{31},e_{41}\rangle$ и $\dim\Ann_{\ell}=2$.
\end{remark}

\begin{remark}\label{remark:graded-graph-auto}
    С другой стороны, условие существования градуировки, согласованной с
    градуировкой Картана, выделяет весьма широкий класс точных обертывающих алгебр:
    по \crefDat{l:enveloping-roots-grading} такая алгебра реализуется на
    пространстве $N\Phi(\mathbb{K})$ с умножением $e_re_s = c_{rs}e_{r+s}$, и,
    обратно, любой выбор коэффициентов $c_{rs} \in \mathbb{K}$ с условием
    $c_{rs}-c_{sr}=N_{rs}$ задает такую алгебру.

    Нетривиальная симметрия графа Кокстера системы $\Phi$ типа $A_2$ индуцирует
    \cite[предложение 12.2.3]{carter72} автоморфизм $\theta$ алгебры Ли
    $N\Phi(\mathbb{K})$:
    \[
        \theta(e_{21})=e_{32}, \qquad \theta(e_{32})=e_{21}, \qquad
        \theta(e_{31})=-e_{31}.
    \]
    Автоморфизм точной обертывающей алгебры с умножением $e_re_s = c_{rs}e_{r+s}$
    назовем \emph{графовым}, если он переводит каждую компоненту $\mathbb{K}e_r$ в
    $\mathbb{K}e_{\bar{r}}$, где $\bar{\phantom{r}}$ "--- перестановка корней,
    продолжающая симметрию графа. Ни одна специальная обертывающая алгебра $R_\Phi$
    типа $A_2$ графового автоморфизма не допускает: если, скажем,
    $e_{32}e_{21}=e_{31}$ и $e_{21}e_{32}=0$, то для графового автоморфизма
    $\varphi$ получаем противоречие
    \[
        0 \neq \varphi(e_{32}e_{21}) = \varphi(e_{32})\varphi(e_{21}) \in
        \mathbb{K}(e_{21}e_{32}) = 0.
    \]
    При $2\in\mathbb{K}^*$ переопределение
    \[
        e_{32}e_{21}=\tfrac{1}{2}e_{31}, \qquad e_{21}e_{32}=-\tfrac{1}{2}e_{31}
    \]
    дает ассоциативную точную обертывающую алгебру $R$ с той же градуировкой;
    $\theta$ "--- ее графовый автоморфизм. Алгебры $NT(3,\mathbb{K})$ и $R$ не
    изоморфны, поэтому условие $n \geq 3$ в \crefPred{th:Stand-Graded-An}
    существенно.
\end{remark}

\chapter[Левосимметрические структуры на нильтреугольных алгебрах Шевалле]{Левосимметрические структуры на нильтреугольных\\алгебрах Шевалле}\label{ch:pre-lie}

Для классических типов, отличных от $A_n$, при $\Char\mathbb{K}\neq2$ алгебра
Ли $N\Phi(\mathbb{K})$ не имеет ассоциативной точной обертывающей алгебры
(\cref{p:non-A-no-associative-envelope}). В этой главе для $N\Phi(\mathbb{K})$
типов $B_n$, $C_n$ и $D_n$ точные обертывающие алгебры строятся в классе
левосимметрических алгебр, расширяющем класс ассоциативных алгебр; для типа
$A_{n-1}$ ассоциативная точная обертывающая $NT(n,\mathbb{K})$ указана в
\S~\ref{sec:graded-An}.

\medskip

В \S~\ref{sec:pre-lie-algs} развивается общий аппарат: левосимметрические
алгебры и их полупрямые суммы. Здесь же (\cref{l:nonsing-deriv-pre-lie})
левосимметрическая структура на $N\Phi(\mathbb{K})$ произвольного типа строится
по невырожденному дифференцированию, однако лишь при ограничении на
характеристику поля $\mathbb{K}$. Основная конструкция главы
(\S~\ref{sec:semidirect-classical}) свободна от этого ограничения: для типов
$B_n$, $C_n$ и $D_n$ левосимметрическая точная обертывающая алгебра над
произвольным полем строится как полупрямая сумма алгебры $NT(n,\mathbb{K})$ и
подходящего идеала алгебры Ли $N\Phi(\mathbb{K})$ (\cref{th:BnCnDn-pre-lie}).

\section{Левосимметрические алгебры и их полупрямые суммы}\label{sec:pre-lie-algs}
%
Через $L_x\colon y\mapsto xy$ $(x,y \in R)$ обозначим линейные операторы левого
умножения в алгебре $R$.
\begin{definition}\label{def:pre-lie-algebra}
    %<*def:pre-lie-algebra>
    Алгебра $R$ над полем $\mathbb{K}$ называется \emph{левосимметрической} или
    \emph{прелиевой}, если ее умножение удовлетворяет тождеству
    \begin{equation*}
        (xy)z-x(yz)=(yx)z-y(xz) \qquad (x,y,z\in R)
    \end{equation*}
    или, эквивалентно, отображение $L\colon x \mapsto L_x$ задает представление
    алгебры Ли $R^{(-)}$ в $\mathfrak{gl}(R)$:
    \begin{equation*}
        [L_x,L_y]=L_{[x,y]} \mathrlap{\qquad (x,y \in R).}
    \end{equation*}
    %</def:pre-lie-algebra>
\end{definition}

Как и в ассоциативном случае, всякая левосимметрическая алгебра является
Ли-допустимой: если положить $[x,y]\coloneq xy-yx$, то $(R,[\ ,\ ])$ является
алгеброй Ли.

Левосимметрическую точную обертывающую алгебру для алгебры Ли $\mathfrak{g}$
также называют \emph{аффинной} или \emph{левосимметрической структурой} на
$\mathfrak{g}$. В обзоре Д.~Бурде \cite{burde-pre-lie-survey} отмечено, что
полупростая конечномерная алгебра Ли над полем характеристики 0 не допускает
левосимметрических структур. В \cite[\S~4]{milnor-left-sym} Дж.~Милнор поставил
вопрос о полных левоинвариантных аффинных структурах на разрешимых группах Ли,
связанный с существованием левосимметрических структур. Д.~Бурде и
Ф.~Груневальд \cite{burde-counter-example} построили примеры нильпотентных
алгебр Ли, не допускающих левосимметрических структур.

\medskip

Дифференцирование $D$ алгебры Ли $\mathfrak{g}$ называется
\emph{невырожденным}, если $\ker D = 0$. По
\cite[следствие~3.4]{burde-pre-lie-survey}, наличие невырожденного
дифференцирования в конечномерной алгебре Ли влечет существование на ней
левосимметрической структуры. В следующей лемме приведем явную конструкцию
такого умножения.
\begin{lemma}\label{l:nonsing-deriv-pre-lie}
    Пусть $D$ "--- невырожденное дифференцирование конечномерной алгебры Ли
    $\mathfrak{g}$. Тогда формула $x \cdot y \coloneq D^{-1}[x, Dy]$ задает на
    $\mathfrak{g}$ левосимметрическое умножение, для которого $(\mathfrak{g},
    \cdot)^{(-)} = \mathfrak{g}$.
\end{lemma}
\begin{proof}
    Из определения $D(x \cdot y) = [x, Dy]$. Поскольку $D$ "--- дифференцирование,
    $D[x, y] = [Dx, y] + [x, Dy] = [x, Dy] - [y, Dx]$, откуда
    \[
        x \cdot y - y \cdot x = D^{-1}([x, Dy] - [y, Dx]) = D^{-1} D[x, y] = [x, y],
    \]
    то есть $(\mathfrak{g}, \cdot)^{(-)} = \mathfrak{g}$. Далее, применяя $D$ к
    выражению $(x \cdot y) \cdot z - x \cdot (y \cdot z)$, получаем $[x \cdot y,
    Dz] - [x, [y, Dz]]$, поэтому
    \[
        D\bigl((x \cdot y) \cdot z - x \cdot (y \cdot z) - (y \cdot x) \cdot z
        + y \cdot (x \cdot z)\bigr)
        = [x \cdot y - y \cdot x, Dz] - [[x, y], Dz] = 0
    \]
    по доказанному и тождеству Якоби. Невырожденность $D$ дает левосимметрическое
    тождество.
\end{proof}
%
Диагональное отображение $D(e_r) \coloneq \height(r) e_r$ ($r \in \Phi^+$)
является дифференцированием алгебры $N\Phi(\mathbb{K})$ и невырождено при
$\Char\mathbb{K} = 0$ либо $\Char\mathbb{K} > \height(\rho)$, где $\rho$ "---
максимальный в $\Phi^+$ корень. По \crefDat{l:nonsing-deriv-pre-lie} формула
\[
    e_r \cdot e_s = D^{-1}[e_r, D e_s] =
    \frac{\height(s)}{\height(r) + \height(s)}\, [e_r, e_s]
    \qquad (r, s, r+s \in \Phi^+)
\]
вместе с условием $e_r\cdot e_s=0$ при $r+s\notin\Phi^+$ задает
левосимметрическую структуру на $N\Phi(\mathbb{K})$.

\medskip

Для построения левосимметрических точных обертывающих алгебр для алгебр Ли
$N\Phi(\mathbb{K})$ типов $B_n$, $C_n$ и $D_n$ над произвольным полем
$\mathbb{K}$ используем их разложения в полупрямые суммы подалгебры
$NA_{n-1}(\mathbb{K})$ с подходящим идеалом.

\begin{definition}[{{\cite[\S~1.2.11]{vinberg-onishik}}}]\label{def:semidirect-lie}
    \emph{Полупрямой суммой $\mathfrak{g}\ltimes_{\phi} \mathfrak{h}$} алгебр
    Ли $\mathfrak{g}$ и $\mathfrak{h}$ называется прямая сумма векторных
    пространств $\mathfrak{g}$ и $\mathfrak{h}$, снабженная операцией
    коммутирования по формуле
    \[
        [(x,a),(y,b)] = \bigl([x,y],\ \phi_x(b)-\phi_y(a) + [a,b]\bigr),
    \]
    где $\phi\colon\mathfrak{g}\to\Der(\mathfrak{h})$ "--- некоторый гомоморфизм
    алгебры Ли $\mathfrak{g}$ в алгебру Ли дифференцирований $\Der(\mathfrak{h})$ и
    $\phi_x = \phi(x)$.
\end{definition}
%
Полупрямые суммы алгебр Ли возникают, например, в теореме Леви--Мальцева о
разложении конечномерной алгебры Ли над полем характеристики 0 в полупрямую
сумму разрешимого идеала и полупростой подалгебры
\cite[\S~6.1]{vinberg-onishik}.

По аналогии с \crefTvor{def:semidirect-lie}, введем понятие полупрямой суммы
для левосимметрических алгебр.
\begin{definition}\label{def:semidirect-pre-lie}
    %<*def:semidirect-pre-lie>
    \emph{Полупрямой суммой}
    $R\ltimes_{\tau} I$ левосимметрических алгебр $R$ и $I$ назовем прямую сумму
    векторных пространств $R$ и $I$, снабженную операцией умножения по формуле
    \[
        (x,a)(y,b) = (xy,\ \tau_x(b) + ab),
    \]
    где $\tau\colon R^{(-)} \to \Der(I)$ "--- некоторый гомоморфизм алгебры Ли
    $R^{(-)}$ в алгебру Ли дифференцирований $\Der(I)$ и $\tau_x = \tau(x)$.
    %</def:semidirect-pre-lie>
\end{definition}
%
Всякое дифференцирование левосимметрической алгебры $I$ является
дифференцированием алгебры Ли $I^{(-)}$, то есть $\Der(I)\subset
\Der(I^{(-)})$. Таким образом, при рассмотрении полупрямой суммы
левосимметрических алгебр $R\ltimes_{\tau} I$ можно одновременно рассматривать
и полупрямую сумму ассоциированных алгебр Ли $R^{(-)}$ и $I^{(-)}$ с тем же
действием $\tau$.

\medskip

Как показывают следующие две теоремы, полупрямая сумма ассоциативных алгебр
может быть неассоциативной, но всегда является левосимметрической.
% \begin{theorem}[{{\cite[теорема~3]{jsfu2026}}}]\label{th:semidirect-assoc-cond}
\begin{theorem}\label{th:semidirect-assoc-cond}
    %<*th:semidirect-assoc-cond>
    Пусть $R$ и $I$ "--- ассоциативные алгебры и $\tau\colon R^{(-)}\to\Der(I)$
    "--- гомоморфизм алгебры Ли $R^{(-)}$ в $\Der(I)$. Алгебра $R\ltimes_{\tau} I$
    ассоциативна тогда и только тогда, когда выполнены условия:
    \begin{enumerate}
        \item[\textup{(A)}] $\tau_{xy}=\tau_x\tau_y$ для всех $x,y\in R$;
        \item[\textup{(B)}] $\tau_x(I)\subset \Ann_r(I)$ для всех $x\in R$.
    \end{enumerate}
    %</th:semidirect-assoc-cond>
\end{theorem}
\begin{proof}
    Пусть выполняются условия (A) и (B). Так как $R$ ассоциативна, достаточно
    проверить совпадение вторых компонент в равенстве
    \begin{equation}\label{eq:semidirect-assoc}
        ((x,a)(y,b))(z,c)=(x,a)((y,b)(z,c))
    \end{equation}
    %
    для всех $x,y,z \in R$ и $a,b,c \in I$. По определению умножения в
    $R\ltimes_{\tau} I$ имеем
    \[
        (x,a)(y,b)=(xy,\ \tau_x(b)+ab),\qquad (y,b)(z,c)=(yz,\ \tau_y(c)+bc).
    \]
    Поэтому вторая компонента левой части равна
    \[
        \tau_{xy}(c)+(\tau_x(b)+ab)c=\tau_{xy}(c)+\tau_x(b)c+(ab)c,
    \]
    а вторая компонента правой части равна
    \[
        \tau_x(\tau_y(c)+bc)+a(\tau_y(c)+bc)=\tau_x\tau_y(c)+\tau_x(bc)+a\tau_y(c)+a(bc).
    \]
    По \textup{(A)} имеем $\tau_x\tau_y(c) = \tau_{xy}(c)$. По условию
    $\tau_x\in\Der(I)$ и \textup{(B)} имеем
    \[
        \tau_x(bc) + a\tau_y(c) = \tau_x(b)c+b\,\tau_x(c) = \tau_x(b)c.
    \]
    Наконец, так как $I$ ассоциативна, то $a(bc)=(ab)c$. Следовательно, вторые
    компоненты совпадают, что и доказывает ассоциативность $R\ltimes_{\tau} I$.

    \medskip

    Пусть теперь алгебра $R\ltimes_{\tau} I$ ассоциативна. Подставляя в тождество
    ассоциативности \eqref{eq:semidirect-assoc} тройку $(x,0),(y,0),(0,c)$,
    получаем
    \[
        ((x,0)(y,0))(0,c)=(xy,0)(0,c)=(0,\tau_{xy}(c)),
    \]
    \[
        (x,0)((y,0)(0,c))=(x,0)(0,\tau_y(c))=(0,\tau_x\tau_y(c)),
    \]
    откуда следует (A).

    \medskip

    Далее, подставляя в \eqref{eq:semidirect-assoc} тройку $(x,0),(0,b),(0,c)$,
    получаем
    \[
        ((x,0)(0,b))(0,c)=(0,\tau_x(b))(0,c)=(0,\tau_x(b)c),
    \]
    \[
        (x,0)((0,b)(0,c))=(x,0)(0,bc)=(0,\tau_x(bc)),
    \]
    то есть $\tau_x(bc)=\tau_x(b)c$ для всех $b,c\in I$. Так как
    $\tau_x\in\Der(I)$, имеем
    \[
        \tau_x(bc)=\tau_x(b)c+b\,\tau_x(c),
    \]
    и потому $b\,\tau_x(c)=0$ для всех $x \in R$ и $b,c\in I$, что равносильно (B).
\end{proof}
% \begin{theorem}[{{\cite[теорема~4]{jsfu2026}}}]\label{th:semidirect-pre-lie}
\begin{theorem}\label{th:semidirect-pre-lie}
    %<*th:semidirect-pre-lie>
    Пусть $R,I$ "--- левосимметрические алгебры, $\tau\colon R^{(-)}\to\Der(I)$
    "--- гомоморфизм алгебры Ли $R^{(-)}$ в $\Der(I)$. Тогда алгебра
    $R\ltimes_{\tau} I$ является левосимметрической. Более того, $(R\ltimes_{\tau}
    I)^{(-)}$ изоморфна полупрямой сумме $R^{(-)}\ltimes_\tau I^{(-)}$, причем
    \[
        [(x,a),(y,b)]=\bigl([x,y],\ \tau_x(b)-\tau_y(a)+[a,b]\bigr).
    \]
    %</th:semidirect-pre-lie>
\end{theorem}
\begin{proof}
    Непосредственным вычислением находим формулу коммутатора в $(R\ltimes_{\tau}
    I)^{(-)}$:
    \[
        [(x,a),(y,b)]=\bigl([x,y],\ \tau_x(b)-\tau_y(a)+[a,b]\bigr),
    \]
    и тем самым получаем изоморфизм $(R\ltimes_{\tau} I)^{(-)} \simeq
    R^{(-)}\ltimes_\tau I^{(-)}$.

    \medskip

    Докажем левосимметричность $R\ltimes_{\tau} I$.

    \medskip

    Для $(x,a) \in R\ltimes_{\tau} I$ оператор $L_{(x,a)}$ действует на
    $R\ltimes_{\tau} I$ по формуле
    \[
        L_{(x,a)}(y,b)=(xy,\ \tau_x(b)+ab),
    \]
    то есть $L_{(x,a)}|_R = L_x$ и $L_{(x,a)}|_I = \tau_x+L_a$.

    \medskip

    Так как $R$ и $I$ левосимметрические, получаем $[L_x,L_y]=L_{[x,y]}$ и
    $[L_a,L_b]=L_{[a,b]}$. Кроме того, поскольку $\tau$ "--- гомоморфизм алгебры Ли
    $R^{(-)}$ в $\Der(I)$, имеем $[\tau_x,\tau_y]=\tau_{[x,y]}$ и
    $[\tau_x,L_a]=L_{\tau_x(a)}$.

    Пользуясь полученными тождествами, в ограничениях на $R$ и $I$ получаем
    \begin{align*}
        \restr{[L_{(x,a)}, L_{(y,b)}]}{R}
         & = [L_x, L_y] = L_{[x,y]}                                        \\
         & = \restr{L_{[(x,a),(y,b)]}}{R},                                 \\
        \restr{[L_{(x,a)}, L_{(y,b)}]}{I}
         & = [\tau_x + L_a, \tau_y + L_b]                                  \\
         & = [\tau_x, \tau_y] + [L_a, \tau_y] + [\tau_x, L_b] + [L_a, L_b] \\
         & = \tau_{[x,y]} - L_{\tau_y(a)} + L_{\tau_x(b)} + L_{[a,b]}      \\
         & = \tau_{[x,y]} + L_{[a,b] + \tau_x(b) - \tau_y(a)}              \\
         & = \restr{L_{([x,y],\ \tau_x(b) - \tau_y(a) + [a,b])}}{I}        \\
         & = \restr{L_{[(x,a),(y,b)]}}{I}.
    \end{align*}

    Поскольку оба оператора $[L_{(x,a)}, L_{(y,b)}]$ и $L_{[(x,a),(y,b)]}$
    сохраняют подпространства $R$ и $I$ и совпадают на них, получаем
    \[
        [L_{(x,a)},L_{(y,b)}]=L_{[(x,a),(y,b)]}.\qedhere
    \]
\end{proof}
% \begin{corollary}[{{\cite[следствие~4.1]{jsfu2026}}}]\label{corollary:enveloping-semidirect}
\begin{corollary}\label{corollary:enveloping-semidirect}
    Пусть $\mathfrak{g} = \mathfrak{l}\ltimes_{\phi}\mathfrak{h}$ "--- полупрямая
    сумма алгебр Ли, $R$ и $I$ "--- левосимметрические точные обертывающие алгебры
    для $\mathfrak{l}$ и $\mathfrak{h}$ соответственно. Пусть также задан
    гомоморфизм алгебр Ли $\tau\colon R^{(-)}\to\Der(I)$ такой, что индуцированное
    $\tau$ действие алгебры Ли $R^{(-)}$ на $I^{(-)}$ совпадает с действием $\phi$.
    Тогда алгебра $R\ltimes_{\tau} I$ является левосимметрической точной
    обертывающей алгеброй алгебры Ли $\mathfrak{g}$.
\end{corollary}
\begin{proof}
    По \crefDat{th:semidirect-pre-lie} алгебра $R\ltimes_{\tau}I$
    левосимметрическая, а значит, Ли-допустима. Кроме того,
    \[
        (R\ltimes_{\tau}I)^{(-)}\simeq
        R^{(-)}\ltimes_\tau I^{(-)}\simeq
        \mathfrak{l}\ltimes_{\phi}\mathfrak{h} = \mathfrak{g},
    \]
    откуда $R\ltimes_{\tau}I$ является точной обертывающей алгеброй алгебры Ли
    $\mathfrak{g}$.
\end{proof}

\section{Левосимметрические структуры для типов \texorpdfstring{$B_n$}{Bₙ}, \texorpdfstring{$C_n$}{Cₙ} и \texorpdfstring{$D_n$}{Dₙ}}\label{sec:semidirect-classical}
%
Используем \crefVin{th:semidirect-pre-lie} и ее
\crefVin{corollary:enveloping-semidirect} для построения нильпотентных
градуированных левосимметрических точных обертывающих алгебр для
$N\Phi(\mathbb{K})$ типов $B_n$, $C_n$ и $D_n$.

\medskip

Пусть $\Psi = A_{n-1}$ из \eqref{eq:An-Root-System}. Системы положительных
корней типов $D_n$, $C_n$ и $B_n$ можно представить в виде объединения
(\cite[таблицы I--IV]{bourbaki72})
\[
    \begin{aligned}
        D_n^+ & = \Psi^+ \cup \{\epsilon_i+\epsilon_j \mid 1\le j<i\le n\}, \\
        C_n^+ & = D_n^+ \cup \{2\epsilon_i \mid 1\le i\le n\},              \\
        B_n^+ & = D_n^+ \cup \{\epsilon_i \mid 1\le i\le n\}.
    \end{aligned}
\]
%
Всюду в этом параграфе считаем $n \geq 2$. Базис системы корней $\Phi$ типов
$D_n$, $C_n$ и $B_n$ запишем в виде $\Pi(\Phi) = \Pi(\Psi) \cup \{r_\Phi\}$,
где
\[
    r_\Phi\coloneq
    \begin{cases}
        \epsilon_2 + \epsilon_1, & \Phi = D_n, \\
        2\epsilon_1,             & \Phi = C_n, \\
        \epsilon_1,              & \Phi = B_n.
    \end{cases}
\]

Ясно, что $N\Psi = \langle e_r\mid r \in \Psi^+ \rangle \simeq
NT(n,\mathbb{K})^{(-)}$ является подалгеброй в $N\Phi(\mathbb{K})$ для системы
корней $\Phi$ ранга $n$.

\medskip

Обозначая через $\phi$ присоединенное действие $NT(n,\mathbb{K})^{(-)}$ на
идеале $T(r_\Phi)$, получаем разложение $N\Phi(\mathbb{K})$ в полупрямую сумму
алгебр Ли
\begin{equation*}
    N\Phi(\mathbb{K})\simeq NT(n,\mathbb{K})^{(-)} \ltimes_\phi T(r_\Phi).
\end{equation*}

По \crefDat{corollary:enveloping-semidirect} \crefRod{th:semidirect-pre-lie},
чтобы построить точную обертывающую алгебру $N\Phi(\mathbb{K})$ типов $D_n$,
$C_n$ и $B_n$ в виде полупрямой суммы ассоциативных алгебр, достаточно выбрать
на идеале $T(r_\Phi)$ ассоциативное умножение
\[
    \cdot_\Phi\colon T(r_\Phi) \times T(r_\Phi) \to T(r_\Phi)
\]
такое, что $(T(r_\Phi), \cdot_\Phi)^{(-)} = T(r_\Phi)$. Кроме того,
присоединенное действие $\phi$ алгебры Ли $NT(n,\mathbb{K})^{(-)}$ на
$T(r_\Phi)$ должно быть согласовано с $\cdot_\Phi$: оператор $\phi_x$ для
каждого $x\in NT(n,\mathbb{K})$ должен быть дифференцированием ассоциативной
алгебры $\bigl(T(r_\Phi),\cdot_\Phi\bigr)$.

\medskip

Для типов $D_n$ и $C_n$ идеал $T(r_\Phi)$ является абелевым и в качестве
$\cdot_\Phi$ можем взять нулевое умножение.

\medskip

Рассмотрим тип $B_n$. Коммутант алгебры Ли $T(\epsilon_1)$ централен, и
умножение
\[
    e_r \cdot_{B_n} e_s \coloneq
    \begin{cases}
        \dfrac12[e_r,e_s], & \Char \mathbb{K}\neq 2, \\[4pt]
        0,                 & \text{иначе}
    \end{cases}
    \mathrlap{\qquad\bigl(e_r,e_s \in T(\epsilon_1)\bigr)}
\]
дает ассоциативную точную обертывающую алгебру $\bigl(T(\epsilon_1),
\cdot_{B_n}\bigr)$ для $T(\epsilon_1)$.

% \begin{theorem}[{{\cite[теорема~5]{jsfu2026}}}]\label{th:BnCnDn-pre-lie}
\begin{theorem}\label{th:BnCnDn-pre-lie}
    %<*th:BnCnDn-pre-lie>
    Полупрямая сумма ассоциативных алгебр
    \[
        R\Phi'(\mathbb{K}) \coloneq NT(n,\mathbb{K}) \ltimes_{\phi} \bigl(T(r_\Phi), \cdot_\Phi\bigr)
    \]
    является левосимметрической точной обертывающей алгебры $N\Phi(\mathbb{K})$
    типа $D_n$, $C_n$ или $B_n$.
    %</th:BnCnDn-pre-lie>
\end{theorem}

Базис $\{e_r \mid r \in \Phi^+\}$ задает на алгебре $R\Phi'(\mathbb{K})$
градуировку решеткой корней $\mathbb{Z}\Phi$, согласованную с градуировкой
Картана: $\mathbb{K}e_r \cdot \mathbb{K}e_s \subset \mathbb{K}e_{r+s}$
(полагаем $e_{r+s}=0$ при $r+s\notin\Phi^+$), поскольку однородны матричное
умножение в $NT(n,\mathbb{K})$, присоединенное действие $\phi$ и умножение
$\cdot_\Phi$. Кроме того, произведение любых $m$ базисных элементов лежит в
компоненте корня высоты не менее $m$, поэтому $R\Phi'(\mathbb{K})^{m}=0$ при $m
> \height(\rho)$ и алгебра $R\Phi'(\mathbb{K})$ нильпотентна.

\chapter[Проблемы перечисления идеалов нильтреугольных алгебр Шевалле]{Проблемы перечисления идеалов нильтреугольных\\алгебр Шевалле}\label{ch:enumeration}
Проблемы комбинаторного перечисления стандартных идеалов и всех идеалов алгебр
Ли $N\Phi(\mathbb{F}_q)$ классических типов записаны в 2001 году \cite[проблемы
1 и 2]{sigsam2001}.

\medskip

Согласно~\cite[\S~4]{JA-12}, стандартный идеал $H$ в $N\Phi(\mathbb{K})$
характеризуют множество $\mathcal{L}=\mathcal{L}(H)$ его углов и \emph{фрейм}
$\mathcal{F}(H)$, определяемый условиями
%
\[\mathcal{F} (H)\subset \sum_{s\in \mathcal{L}}\mathbb{K}e_{s},\quad \mathcal{F}(H)=H \mod
    Q(\mathcal{L}).\]
%
Таким образом, стандартность $H$ равносильна условию $H=Q(\mathcal{L}) +
\mathcal{F}(H)$.

Известные перечисления (\cite[следствие 4.3]{vL74}, \cite[теорема
2.1.2]{Egorychev77}, \cite{Eg-Lev-1996}) нормальных подгрупп групп
$U\Phi(\mathbb{K})$ над полем $\mathbb{K}$, инвариантных относительно
диагональных автоморфизмов, редуцируются к перечислению стандартных идеалов $H$
колец Ли $N\Phi(\mathbb{K})$ с фреймом $\mathcal{F}(H) = \sum_{s\in
    \mathcal{L}} \mathbb{K}e_{s}$, то есть вида $T(\mathcal{L})$. Тем самым они
редуцируются к перечислению определенных путей в решетках, зависящих от выбора
типа алгебры $N\Phi(\mathbb{K})$.

\medskip

Родственная задача хорошо изучена в характеристике $0$: идеалы борелевской
подалгебры простой алгебры Ли $\mathfrak{L}(\Phi, \mathbb{C})$, лежащие в ее
нильрадикале $N\Phi(\mathbb{C})$ (\emph{$\ad$-нильпотентные}, или
\emph{$B$-инвариантные}, идеалы), "--- это в точности идеалы $T(\mathcal{L})$ с
множеством углов $\mathcal{L}$ "--- антицепью в $\Phi^+$ \cite{cellini-papi-1}.
Полное число таких идеалов равно обобщенному числу Каталана $\prod_{i=1}^n (h +
e_i + 1)/(e_i + 1)$, где $h$ "--- число Кокстера, $e_i$ "--- экспоненты системы
$\Phi$ \cite[теорема~5.2]{fomin2007} (доказательство, единое для всех типов
$\Phi$, дано в~\cite[теорема~1]{cellini-papi-2}), а число идеалов с $m$ углами,
то есть $m$-элементных антицепей $B(\Phi, m)$, "--- $m$-му числу Нараяны типа
$\Phi$ \cite[теорема~5.9]{fomin2007}. Для классических типов числа $B(\Phi, m)$
выписаны в \crefPred{th:enum-stand-id-classical}, для исключительных "--- в
таблице~\cref{table:Bmn}; о дальнейших связях с областями конфигурации
гиперплоскостей Ши, непересекающимися (non-crossing) разбиениями и элементами
аффинной группы Вейля см.~\cite{eS05,fomin2007} и приведенные там ссылки.

\medskip

Интерес к проблемам перечисления идеалов алгебры Ли $N\Phi(\mathbb{F}_q)$
независимо возникает в теории суперхарактеров и в комбинаторике унипотентных
групп: по теореме В.\,М.~Левчука \cite[теорема~1]{vL76} нормальные подгруппы
унитреугольной группы $UT_n(\mathbb{F}_q)$ в точности совпадают с идеалами
кольца Ли $N\Phi(\mathbb{F}_q)$ типа $A_{n-1}$, а при простом $q$ "--- с
идеалами алгебры Ли. Ассоциативный аналог этого соответствия установил
Э.~Марберг \cite{margberg2011}: при простом $q$ супернормальные подгруппы
(суперхарактерный аналог нормальных) отвечают идеалам ассоциативной алгебры
$NT(n,\mathbb{F}_q)$, то есть стандартным идеалам из проблемы~1. Л.~Ганьон
\cite{gagnon2021}, отталкиваясь от той же теоремы, описывает идеалы алгебры Ли
типа $A_{n-1}$ плотными сплайсами невложенных разбиений и матроидами.

\section{Координатно полные подпространства \texorpdfstring{$\mathbb{F}_q^m$}{F₍q₎ᵐ} и их перечисление}\label{s:coordinate-complete}
%
Пусть $\mathbb{K}^m$ "--- пространство строк длины $m$ над $\mathbb{K}$.
Подпространство $S$ в $\mathbb{K}^m$ назовем \emph{координатно полным}, если
оно не лежит ни в одной координатной гиперплоскости $H_i = \{x \in \mathbb{K}^m
\mid x_i = 0\}$ $(1 \leq i \leq m)$. Координатно полные подпространства
$\mathbb{F}_q^m$ также называют невырожденными линейными кодами в теории
кодирования \cite[\S\ 1.1.2]{acodes2007}.

\medskip

Число всех координатно полных $t$-мерных подпространств пространства
$\mathbb{F}_q^m$ обозначим через $V_q(m,t)$. В~\cite[лемма 2]{kriv-lev2015}
найдена формула
\[
    V_q(m,t) = \mathlarger{\sum}_{\raisebox{-0.6ex}{$\scriptstyle \mathclap{1=j_1<j_2<\ldots
                < j_{t}\leq m}$}}  \ \frac{(q^t-1)^{m-j_{t}}}{(q-1)^{t-j_{t}}}\cdot
    \prod_{k=2}^{t-1}\left(\frac{q^k-1}{q-1}\right)^{\mathrlap{j_{k+1}-j_k-1}}
    .
\]
Применяя к ней разработанный \cite{Egorychev77} метод интегрального
представления комбинаторных сумм, включающих $q$-биномиальные коэффициенты
\[
    \qbinom{n}{k} \coloneq\prod_{i=0}^{k-1}\frac{1-q^{n-i}}{1-q^{i+1}},
\]
Г.\:П.~Егорычев установил \cite[леммы~3 и 4]{IzvIrkGU-2016} при $1\leq t\leq m$
рекуррентное соотношение
\begin{equation}\label{eq:vmt-rec-egorychev}
    V_q(m,t) = \sum_{k=0}^{m-t} (q^t -1)^k \times
    V_q(m-1-k,t-1)
\end{equation}
и следующее утверждение.

\begin{lemma}[Г.\:П. Егорычев]
    При $1\leq t\leq m$ справедлива следующая формула
    \begin{equation}\label{eq:vmt-comb-egorychev}
        V_q(m,t) = \sum_{k=0}^{m-t} (-1)^{m-t-k} q^k \binom{m-1}{t+k-1}
        \qbinom{t+k-1}{k}.
    \end{equation}
\end{lemma}

Для формул \eqref{eq:vmt-rec-egorychev} и \eqref{eq:vmt-comb-egorychev}
в~\cite[заключение]{IzvIrkGU-2016} отмечена потребность в
алгебраически-комбинаторном доказательстве. Рекуррентная формула
\eqref{eq:vmt-rec-egorychev} получается перенесением схемы перечисления
канонических базисов из \crefRod{l:LieIdeal} (\S~\ref{sec:canonicalbase}).

% \begin{lemma}[{{\cite[лемма 2.3]{smj2023}}}]\label{l:vmt-comb}
\begin{lemma}\label{l:vmt-comb}
    Справедлива формула
    \begin{equation}\label{eq:vmt-comb}
        V_q(m,t) = \sum_{k=t}^{m} (-1)^{m-k} \binom{m}{k}
        \qbinom{k}{t}.
    \end{equation}
\end{lemma}

\begin{proof}
    Число всех $t$-мерных подпространств в $\mathbb{F}_q^m$ равно $\qbinom{m}{t}$
    \cite[предложение~1.3.18]{stanley-ec1-ru}. Обозначим через $\hat{H}_i$
    множество $t$-мерных подпространств, лежащих в координатной гиперплоскости
    $H_i$. Для любого набора попарно различных индексов $i_1, \ldots, i_k$
    пересечение $H_{i_1}\cap\ldots\cap H_{i_k}$ состоит из векторов с нулевыми
    координатами $i_1,\ldots,i_k$ и потому изоморфно $\mathbb{F}_q^{m-k}$.
    Следовательно, $|\hat{H}_{i_1}\cap\ldots\cap\hat{H}_{i_k}| = \qbinom{m-k}{t}$,
    причем это значение зависит лишь от числа $k$, а не от выбора индексов. Так как
    число $k$-элементных подмножеств в $\{1,\ldots,m\}$ равно $\binom{m}{k}$,
    формула включений-исключений дает
    \begin{equation}\label{eq:vmt-comb-old}
        V_q(m,t) = \sum_{k=0}^{m-t} (-1)^k \binom{m}{k}
        \qbinom{m-k}{t}.
    \end{equation}
    %
    Подстановка $j=m-k$ в \eqref{eq:vmt-comb-old} завершает доказательство леммы.
\end{proof}

\begin{remark}
    К формуле \eqref{eq:vmt-comb-old} также приводят $q$-аналог $\qbinom{n}{k} =
    \qbinom{n}{n-k}$ известной комбинаторной формулы и подстановка в
    \eqref{eq:vmt-comb-egorychev} тождества
    \[
        q^k\qbinom{t+k-1}{k} = \qbinom{t+k}{t} - \qbinom{t+k-1}{t}.
    \]
    %Это дает решение задачи 1 из \cite{IzvIrkGU-2016}.
\end{remark}

Отметим в заключение параграфа равенство чисел $V_q(n,s)$ числам $A_n^{(s)}$ из
\cite[формула~(9)]{carlitz1933}:
\[
    A_n^{(s)} = \sum_{k=s}^{n}(-1)^{n-k}\binom{n}{k}\qbinom{k}{s}.
\]
Числа $A_n^{(s)}$ возникли у Л.~Карлитца при перечислении подполей абелевых
расширений поля $\mathbb{Q}$. Пусть $K/\mathbb{Q}$ "--- расширение Галуа с
элементарной абелевой группой
$\mathrm{Gal}(K/\mathbb{Q})\simeq{(\mathbb{Z}/q\mathbb{Z})}^n$ ($q$ "---
простое), порожденной автоморфизмами $\sigma_1,\ldots,\sigma_n$ поля $K$.
Обозначим через $K_i$ неподвижное поле автоморфизма $\sigma_i$; тогда
$K_1,\ldots,K_n$ "--- максимальные промежуточные поля степени $q^{n-1}$ над
$\mathbb{Q}$. Число $A_n^{(s)}$ подсчитывает те промежуточные поля $E$
($\mathbb{Q}\subset E \subset K$, $[E:\mathbb{Q}]=q^s$), которые не содержатся
ни в одном из подполей $K_i$.

\medskip

По соответствию Галуа промежуточное поле $E$ с $[E:\mathbb{Q}]=q^s$ отвечает
подгруппе $H = \mathrm{Gal}(K/E)$ группы
$\mathrm{Gal}(K/\mathbb{Q})\simeq\mathbb{F}_q^n$ индекса $q^s$; так как
подгруппы элементарной абелевой $q$-группы суть подпространства, $H$ "---
подпространство $\mathbb{F}_q^n$ коразмерности~$s$. Подгруппа
$\mathrm{Gal}(K/K_i)=\langle\sigma_i\rangle$ "--- одномерное подпространство
$\mathbb{F}_q e_i$ (отождествляем $\sigma_i$ с базисным вектором $e_i$).
Поэтому условие $E\not\subset K_i$ для всех $i$ равносильно тому, что $H$ не
содержит ни одного стандартного базисного вектора $e_i$. Переходя к
ортогональному дополнению $H^\perp$ относительно билинейной формы $\sum_i x_i
y_i$, получаем биекцию таких подпространств с $s$-мерными координатно полными
подпространствами в $\mathbb{F}_q^n$, откуда $A_n^{(s)} = V_q(n,s)$. Отметим
также восходящее к Карлитцу тождество $V_q(n,s) = (q-1)^{n-s}\,S_q[n,s]$, где
$S_q[n,s]$ "--- $q$-числа Стирлинга второго рода \cite[предложение~4.2,
тождество~(4.7)]{anne1993}.

\section{Проблема перечисления стандартных идеалов} \label{sec:Problem-1}
Ясно, что в алгебре Ли $N\Phi(\mathbb{K})$ любой стандартный идеал $H$ с
множеством углов $\mathcal{L}=\{r_1,\ldots,r_m\}$ записывается в виде
\begin{equation}\label{eq:hls}
    H(\mathcal{L},S) = Q(\mathcal{L}) + \{ a_1e_{r_1} + a_2e_{r_2}
    + \dots + a_me_{r_m} \,|\, (a_1,a_2,\dots,a_m)\in S \}
\end{equation}
%
для однозначно определенного координатно полного подпространства $S$ в
$\mathbb{K}^m$.

\medskip

Как и в \S~\ref{s:coordinate-complete}, число всех координатно полных
$t$-мерных подпространств пространства $\mathbb{K}^m$ при
$\mathbb{K}=\mathbb{F}_q$ обозначим через $V_q(m,t)$, а через $B(\Phi,m)$ "---
число всех $m$-элементных антицепей в $\Phi^+$. Из биективности соответствия
\eqref{eq:hls} между стандартными идеалами и парами $(\mathcal{L}, S)$ сразу же
вытекает

\begin{lemma}\label{l:enum-stand-id-reduct-BV}
    Число стандартных идеалов алгебры Ли $N\Phi(\mathbb{F}_q)$ ранга $n$ равно
    \[
        1+\sum\limits_{m=1}^{n}B(\Phi,m)\, \ \sum\limits_{t=1}^{m}V_q(m,t).
    \]
\end{lemma}

\begin{remark}\label{remark:N-at-q1-antichains}
    Значение перечисляющего многочлена из \crefRod{l:enum-stand-id-reduct-BV} при
    $q = 1$ равно
    \[
        \sum_{m=0}^n B(\Phi, m),
    \]
    то есть полному числу антицепей в частичном порядке $\Phi^+$.

    Действительно, при $q = 1$ гауссов коэффициент $\qbinom{k}{t}$ переходит в
    $\binom{k}{t}$, и правая часть \eqref{eq:vmt-comb} в силу ортогональности
    биномиальных коэффициентов принимает вид
    \[
        \sum_{k=t}^{m} (-1)^{m-k} \binom{m}{k} \binom{k}{t} = \delta_{m, t},
    \]
    где $\delta_{m, t}$ "--- символ Кронекера. Поэтому внутренняя сумма
    $\sum_{t=1}^{m} V_q(m, t)$ обращается при $q = 1$ в $1$, а весь многочлен "---
    в $\sum_{m=0}^n B(\Phi, m)$ (с учетом $B(\Phi, 0) = 1$). Единственное ненулевое
    слагаемое при $t = m$ отвечает случаю $S = \mathbb{K}^m$ в \eqref{eq:hls}, то
    есть идеалам $H = T(\mathcal{L})$.
\end{remark}

Полное решение проблемы~1 из \cite{sigsam2001} дает следующая теорема.
% \begin{theorem}[{{\cite[теорема 1]{smj2023}}}]\label{th:enum-stand-id-classical}
\begin{theorem}\label{th:enum-stand-id-classical}
    %<*th:enum-stand-id-classical>
    Число стандартных идеалов алгебры Ли $N\Phi(\mathbb{F}_q)$ классического типа
    $\Phi = A_{n-1}$, $B_n$, $C_n$, $D_n$ равно
    %
    \begin{equation*}
        1+\sum_{m=1}^n B(\Phi,m) \sum_{t=1}^m
        \sum_{k=0}^{m-t} (-1)^k \binom{m}{k} \qbinom{m-k}{t},
    \end{equation*}
    где $B(\Phi,m)=\binom{n}{m}^2$ для типов $B_n$ и $C_n$, а также
    \[
        B(A_{n-1},m) = \frac{1}{n}\binom{n}{m}\binom{n}{m+1}, \quad B(D_n,m) =
        \binom{n}{m}\left( \binom{n-1}{m} + \binom{n-2}{m-2} \right).
    \]
    %</th:enum-stand-id-classical>
\end{theorem}
\begin{proof}
    В~\cite{vmj2015} проблема~1 из \cite{sigsam2001} исследовалась для типа
    $A_{n-1}$. Числа $B(\Phi,m)=B(A_{n-1},m)$ вычислены ранее
    \cite[теорема~2.1.2]{Egorychev77} (см. также \cite{vL74} и
    \cite{Tolasov-1977}):
    \[
        B(A_{n-1},m) = \frac{1}{n}\binom{n}{m}\binom{n}{m+1}.
    \]

    \Cref{l:vmt-comb,l:enum-stand-id-reduct-BV} вместе с указанной
    формулой для числа $B(A_{n-1},m)$ завершают доказательство
    \crefRod{th:enum-stand-id-classical} для типа $A_{n-1}$: лиев ранг здесь
    равен $n-1$, и слагаемое с $m = n$ в сумме теоремы обращается в нуль,
    так как $B(A_{n-1},n) = 0$.

    \medskip

    Числа $B(\Phi,m)$ исследовались для классических типов в различных ситуациях с
    использованием метода интегрального представления комбинаторных сумм
    (\cite{Egorychev77}, \cite{Eg-Lev-1996}, \cite{sigsam2001} и др.). Эти числа
    указаны явно в выписанном в \crefPred{th:enum-stand-id-classical} виде для
    типов $B_n$ и $C_n$ в~\cite[предложение~17]{reiner97}, а для типа $D_n$ "---
    в~\cite[теорема~1.2]{reiner04}.

    \medskip

    Учитывая \cref{l:vmt-comb,l:enum-stand-id-reduct-BV}, завершаем доказательство
    теоремы \ref{th:enum-stand-id-classical}.
\end{proof}

\section{Специальный канонический базис идеала}\label{sec:canonicalbase}

В связи с проблемой 2 из \cite{sigsam2001} перечисления всех идеалов алгебр Ли
$N\Phi(\mathbb{F}_q)$ классических типов, в \cite[\S\ 4]{vmj2015}
рассматривались специальные канонические базисы лиевых идеалов алгебр
$NT(n,\mathbb{K})$.

\medskip

Аналогичные базисы подалгебр в алгебрах Ли $N\Phi(\mathbb{K})$ выявляет

\begin{lemma}\label{l:LieIdeal}
    Пусть $H$ "--- ненулевая подалгебра алгебры Ли $N\Phi(\mathbb{K})$ над полем
    $\mathbb{K}$ и $\mathcal{L} = \mathcal{L}(H)= \{r_1, r_2, \dots, r_m\}$. Тогда
    любой базис пересечения $Q(\mathcal{L})\cap H$ можно дополнить до базиса в $H$
    элементами

    \begin{equation}\label{eq:base1'}
        \alpha_i=  \sum_{j=1}^m a_{ij}e_{r_j}+\alpha_i' ,\quad
        i=1,2,\dots, t,\quad  t= \dim_{\mathbb{K}} H / H \cap Q(\mathcal{L}),
    \end{equation}
    где $||a_{ij}||$ "--- $(t\times m)$-матрица ранга $t$ над $\mathbb{K}$ без
    нулевых столбцов $(t \leq m)$ и $\alpha_i' \in Q(\mathcal{L})$. Кроме того, при фиксированном
    упорядочении $\mathcal{L}$ для однозначно определенных номеров
    $j_1=1<j_2<\cdots <j_t \leq m$ можно считать

    \begin{equation}\label{eq:base-1}
        a_{i,j_i}=1\ (1\leq i\leq t),\quad a_{ik}=0\ (k<j_i),\quad
        a_{i,j_k}=0,\ 1\leq i<k\leq t.
    \end{equation}
\end{lemma}

\begin{proof}
    Ясно, что $Q(\mathcal{L})+H=Q(\mathcal{L})+\mathcal{F}(H)$ есть стандартный
    идеал в $N\Phi(\mathbb{K})$, имеющий вид $H(\mathcal{L},S)$ из \eqref{eq:hls}.
    Элементы $\alpha_i\in H$ и равенства \eqref{eq:base-1} получаем, используя
    известный изоморфизм
    \[
        (H+Q(\mathcal{L})) / Q(\mathcal{L}) \simeq H / H \cap Q(\mathcal{L}).
    \]

    Уточним матрицу $||a_{ij}||$ из \eqref{eq:base1'}. Очевидно, базисный элемент
    $\alpha_1\in H$ всегда можно выбрать так, что $a_{11}=1$; положим $j_1=1$ и
    обозначим через $H_1$ подалгебру элементов из $H$ с нулевым коэффициентом при
    $e_{r_1}$. Если $\alpha_i$, $j_i$ и $H_i$ уже определены при $1\leq i<t$, то
    элемент $\alpha_{i+1}\in H_i$ выбираем с наименьшим возможным номером $j_{i+1}$
    первой ненулевой координаты; можно считать $a_{i+1,k}=1$ при $k=j_{i+1}$. Через
    $H_{i+1}$ обозначаем подалгебру элементов из $H_i$ с нулевым коэффициентом при
    $e_{r_k}, \ k=j_{i+1}$.

    Продолжая аналогично, на $t$-м шаге выбираем в $H_{t-1}$ элемент $\alpha_{t}$ с
    коэффициентом $a_{t,j_t}=1$ и $H_{t}=H \cap Q(\mathcal{L})$. Элементарные
    преобразования дают оставшиеся в~\eqref{eq:base-1} равенства $a_{i,j_k}=0$
    последовательно для $i=1,2, \cdots , t-1$ и $i<k\leq t$. Это завершает
    доказательство леммы.
\end{proof}

Уточним базис \eqref{eq:base1'} для нестандартного идеала $H$ обертывающей
алгебры $RD_n(\mathbb{K})$. По \crefDat{th:classical-Exc-RDn}, $Q(r)\subset H$
для всех его углов $r$, кроме двух симметричных углов $s$ и $\bar{s}$. Простой
корень $p \neq \bar{p}$, инцидентный с $s$, и простой корень $p_0= \bar{p}_0$
такой, что $p+p_0\in\Phi^+$, определены однозначно, причем $T(s_0 + \bar{p})
\subset H$, где полагаем $s_0=s+p_0$ при $s=p$ и $s_0=s$ при $s\neq p$.

\medskip

Если неинцидентные с $s$ и $\bar{s}$ простые корни $p_j= \bar{p}_j$ выберем
так, что
%
\begin{equation}\label{eq:r-stepsId}
    s_i=s+p_1+p_2+\cdots + p_i\in \Phi^+, \quad 1\leq i< n-\height(s),
\end{equation}
%
то $Q(s_{i})\subset H$ при $i=n-1-\height(s)$, поэтому существует наименьший
номер $k$ с условиями $Q(s_k) \subset H$ и $1\leq k\leq n-1-\height(s)$.
Записывая $\mathcal{L}$ в виде
\begin{equation}\label{eq:R-stepsId}
    \mathcal{L}=\{r_1=s, r_2=\bar{s}, r_3, \dots, r_m\}\quad (2\leq
    m=|\mathcal{L}| <n),
\end{equation}
при $3\leq j\leq m$ имеем $\bar{r}_j= r_j$. Сопоставим $H$, по лемме
\ref{l:LieIdeal}, $(t\times m)$-матрицу $||a_{ij}||$ над $\mathbb{K}$ ранга $t$
без нулевых столбцов $(1\leq t<m)$ с условиями \eqref{eq:base-1}. Тогда $a_{12}=c\neq 0$ и
\[
    H \cap Q(\mathcal{L}) =T(s_0 + \bar{p}) + Q(s_k) + Q(\bar{s}_k)+
    \sum_{j=1}^k \mathbb{K}(e_{s_j} + ce_{\bar{s}_j})+ \sum_{j=3}^m Q(r_j)
\]
(последнее слагаемое при $m=2$ отбрасываем). Упрощая в
\eqref{eq:base1'} элементы $\alpha_i$ по модулю $H \cap Q(\mathcal{L})$,
находим $d_i\in \mathbb{K}$ такие, что $\alpha_i'= d_ie_{s_k}, \
i=1,2, \dots, t.$

\medskip

Таким образом, идеал $H$ представляется в виде

\begin{equation}\label{eq:IdH0}
    Id\ \{ \mathcal{L}, k, t, ||a_{ij}||,\ c, d_1, \dots, d_t \}\ \coloneq
    \sum_{i=1}^t \ \mathbb{K}(d_ie_{s_k}+\sum_{j=1}^m a_{ij}e_{r_j})+
\end{equation}
\[
    +T(s_0 + \bar{p})+ Q(s_k)+ Q(\bar{s}_k)+
    \sum_{j=1}^k \mathbb{K}(e_{s_j} + ce_{\bar{s}_j})+\sum_{j=3}^m Q(r_j),
\]
где корни $s_i$ определены в \eqref{eq:r-stepsId}. Его однозначно характеризуют
множество углов $\mathcal{L}=\mathcal{L}(H)$ вида \eqref{eq:R-stepsId},
параметры $k, t$ с условиями
\[
    1\leq k\leq n-1-\height(s),\quad 1\leq t= \dim_{\mathbb{K}} H / H \cap Q(\mathcal{L}) <|\mathcal{L}|=m <n,
\]
элементы $c\neq 0$, $d_1, \dots , d_t$ из $\mathbb{K}$ и $(t\times m)$-матрица
$||a_{ij}||$ над $\mathbb{K}$ ранга $t$ без нулевых столбцов с условиями
$a_{12}=c$ и \eqref{eq:base-1}. Тем самым доказана

% \begin{theorem}[{{\cite[теорема 6]{smj2023}}}]\label{th:ExIdForm}
\begin{theorem}\label{th:ExIdForm}
    %<*th:ExIdForm>
    В обертывающей алгебре $RD_n(\mathbb{K})$ всякий нестандартный идеал $H$ имеет
    множество углов $\mathcal{L}=\mathcal{L}(H)$ вида \eqref{eq:R-stepsId} и
    представляется как идеал \eqref{eq:IdH0}, однозначно характеризуемый набором
    $\mathcal{L}, k, t, ||a_{ij}||,\ c, d_1, \dots, d_t$.
    %</th:ExIdForm>
\end{theorem}

\section{Метод коэффициентов и перечисление идеалов алгебры \texorpdfstring{$RD_n(\mathbb{F}_q)$}{RDₙ(F₍q₎)}}\label{sec:egorychev-RDn}
%
\Cref{th:enum-stand-id-classical}, решая проблему 1 из \cite{sigsam2001}
о числе всех стандартных идеалов алгебры Ли $N\Phi(\mathbb{F}_q)$ классического
типа над полем $\mathbb{K}=\mathbb{F}_q$, дает и число всех идеалов ее обертывающей алгебры
$R_{\Phi}(\mathbb{F}_q)$, когда она стандартна.

В силу теорем \ref{th:enum-stand-id-classical} и \ref{th:classical-Exc-RDn},
перечисление идеалов любой стандартной обертывающей алгебры из
\crefRod{p:mult-enveloping} классических типов $A_n$, $B_n$ и $C_n$ завершено.
Остается перечислить нестандартные идеалы обертывающей алгебры
$RD_n(\mathbb{F}_q)$ при $n\geq3$.

\medskip

Нестандартные идеалы алгебры $RD_n(\mathbb{F}_q)$, по \crefDat{th:ExIdForm},
исчерпываются идеалами вида~\eqref{eq:IdH0}, которые характеризуются однозначно
набором $\mathcal{L}, k, t, ||a_{ij}||,\ c, d_1, \dots, d_t$. Записывая
$\mathcal{L}$ вида \eqref{eq:R-stepsId} в матричном представлении, полагаем
\[
    e_{s}=e_{i1},\quad e_{\bar{s}}=e_{i,-1},\quad e_{s_k}=e_{i+k,1}\quad (2\leq i\leq n-k).
\]
Через $B(m,n,i,k)$ обозначим число всевозможных $m$-элементных антицепей вида
\eqref{eq:R-stepsId}, содержащих корни $s$, $\bar{s}$ и не содержащих корней в
строках с номерами $j$, $i < j \leq i+k$. Нам потребуется

\begin{lemma}\label{l:rdn}
    Число нестандартных идеалов алгебры $RD_n(\mathbb{F}_q)$ равно
    \begin{equation}\label{eq:enum-Dn-1}
        \sum_{m=2}^{n-1} \sum_{i=2}^{n-1} \sum_{k=1}^{n-i}
        B(m,n,i,k)\sum_{t=1}^{m-1}q^t(q-1)V_q(m-1,t).
    \end{equation}
\end{lemma}

\begin{proof}
    При фиксированных $|\mathcal{L}|= m$ и $k$ число способов выбора параметра
    $c\neq 0$ в $\mathbb{K}=\mathbb{F}_q$ и наборов $(d_1, \dots , d_t)$ над
    $\mathbb{K}$ равно $q^t(q-1)$. Согласно доказательству \crefRod{l:LieIdeal},
    число возможностей выбора матриц $||a_{ij}||$, требуемых для канонических
    базисов идеалов, совпадает с $V_q(m-1,t)$ при $1 \leq t \leq m-1$: ввиду
    $a_{12}=c$ и условий \eqref{eq:base-1} второй столбец матрицы равен
    $(c,0,\dots,0)^T$, а остальные столбцы образуют каноническую
    $(t\times(m-1))$-матрицу без нулевых столбцов. Отсюда для числа нестандартных
    идеалов алгебры $RD_n(\mathbb{F}_q)$ получаем формулу \eqref{eq:enum-Dn-1}.
\end{proof}

Второе и третье суммирования в комбинаторной сумме~\eqref{eq:enum-Dn-1}
проведем с помощью метода коэффициентов \cite{Egorychev77}. Напомним некоторые
понятия.

Пусть $L$ "--- множество формальных степенных рядов Лорана над полем
$\mathbb{C}$, содержащих конечное число членов с отрицательными степенями. По
определению, целое число $k$ есть \emph{порядок} монома $c_kw^k\ \neq 0$ и
$L_k$ "--- множество рядов Лорана из $L$, у которых наименьший порядок мономов
равен $k$. Под \emph{оператором формального вычета} ряда $C(w) =
\sum_{k}c_kw^k$ из $L$ понимают
\[
    \res_wC(w) \coloneq c_{-1}.
\]

Оператор формального вычета $\res$ (контурный интеграл) имеет ряд свойств (см.,
например, \cite{Egorychev77}). Из них наиболее часто используются следующие
два.

\medskip

\textbf{Правило линейности.} Для любых $A(w)$ и $B(w)$ из $L$ и
$a,b \in \mathbb{C}$ имеем
\[
    a\res_wA(w) + b\res_wB(w) = \res_w\{aA(w) + bB(w)\}.
\]

\textbf{Правило подстановки.} Если либо $f \in L_m \
(m=1,2,\dots)$ и $A(w) \in L$, либо $A(w)$ "--- полином и $f(w)\in
L$, то
%
\[\sum_k (f(w))^k\res_z\{A(z)z^{-k-1}\} = A(f(w)).\]

Если $C(w)=\sum_kc_kw^k$ из $L$ есть производящая функция для
последовательности $\{c_k\}$, то $c_k=\res_wC(w)w^{-k-1}$ для любого $k$. Когда
$n,k = 0,1,2,\dots$, приходим к биномиальным коэффициентам
%
\begin{equation}\label{integral_formula}
    \binom{n}{k} = \res_w\left\{\frac{(1+w)^n}{w^{k+1}}\right\} = \frac{1}{2\pi
        i}\int_{\lvert
        w \rvert=\theta} \frac{(1+w)^n}{w^{k+1}}\,dw,\ \theta > 0,
\end{equation}
\[
    \begin{aligned}
        (-1)^k\binom{-n}{k} = \binom{n+k-1}{k}
         & = \res_w\left\{\frac{(1-w)^{-n}}{w^{k+1}}\right\} \\
         & = \frac{1}{2\pi i}\int_{\lvert w \rvert=\theta}
        \frac{(1-w)^{-n}}{w^{k+1}}\,dw,\quad 0 < \theta < 1.
    \end{aligned}
\]

Для суммирования по $i$ и $k$ используем следующую лемму.
%
\begin{lemma}\label{l:B'-Dn}
    %<*l:B'-Dn>
    Справедлива следующая формула суммирования
    \[
        \sum_{i=2}^{n-1} \sum_{k=1}^{n-i} B(m,n,i,k) =
        \binom{n-2}{m-2}\binom{n-1}{m}.
    \]
    %</l:B'-Dn>
\end{lemma}

\begin{proof}
    Обозначим через $B^{(-)}(u,l)$ число $l$-элементных антицепей в $D_n^+$-матрице
    таких, что последний корень расположен левее $(-1)$-го столбца и выше $(u+3)$-й
    строки, а через $B^{(+)}(u,n-u-1,l)$ "--- число $l$-элементных антицепей таких,
    что первый корень расположен правее $1$-го столбца и ниже $(n-u)$-й строки. В
    этих обозначениях
    %
    \begin{equation}\label{Bmnik}
        B(m,n,i,k)= \sum_{l=0}^m B^{(-)}(i-3,l)
        B^{(+)}(n-i-k,i+k-1,m-2-l).
    \end{equation}
    %
    Применяя \cite[Теорема~10.14.1]{Krattenthaler2015}, находим

    \begin{equation}
        B^{(+)}(u,v,l) =
        \binom{u+v}{l}\binom{u}{l} - \binom{u+v-1}{l-1}\binom{u+1}{l+1}.
    \end{equation}
    Несложная индукция по $l$ дает также равенство
    \begin{eqnarray}\label{bminus}
        B^{(-)}(u,l) = \binom{u+1}{2l}.
    \end{eqnarray}

    В силу~\eqref{Bmnik}-\eqref{bminus}, второе и третье суммирования
    в~\eqref{eq:enum-Dn-1} приводят к 3-кратной комбинаторной сумме с биномиальными
    коэффициентами. При этом, как обычно, вычисления проводим с помощью свойств
    оператора $\res$, путем последовательного нахождения интегрального
    представления рассматриваемых выражений и вычисления полученных интегралов. В
    соответствии с \cite[теорема 10.14.1]{Krattenthaler2015} находим представление
    чисел $B^+(u,v,l)$. Пользуясь интегральной формулой~\eqref{integral_formula},
    для каждого из биномиальных коэффициентов находим
    \begin{gather*}
        B^{(+)}(u,v,l) =
        \binom{u+v}{l}\binom{u}{l} - \binom{u+v-1}{l-1}\binom{u+1}{l+1} = \\
        = \res_{z,w}\left\{ \frac{(1+z)^{u+v}}{z^{l+1}}\frac{(1+w)^u}{w^{l+1}}
        \right\}
        - \res_{z,w}\left\{ \frac{(1+z)^{u+v-1}}{z^l}\frac{(1+w)^{u+1}}{w^{(l+1)+1}}
        \right\} = \\
        = \res_{z,w}\left\{ \frac{(1+z)^{u+v-1}(1+w)^u}{z^{l+1}w^{l+2}}(w-z) \right\}
    \end{gather*}
    и тем самым
    \begin{equation*}
        \begin{gathered}
            B^{(+)}(n-i-k, i+k-1, m - 2 - l) = \\
            \res_{z,w}\left\{ \frac{(1+z)^{n-2}(1+w)^{n-i-k}}{z^{(m-2-l) +
                    1}w^{(m-1-l)
                    + 1}}(w-z) \right\}.
        \end{gathered}
    \end{equation*}

    Отсюда, в силу \eqref{Bmnik}-\eqref{bminus}, получаем
    \begin{gather*}
        S \coloneq \sum_{i=2}^{n-1}\sum_{k=1}^{n-i}B(m,n,i,k) = \\
        = \sum_{i=2}^{n-1}\sum_{k=1}^{n-i}\sum_{l=0}^{m}B^{(-)}(i-3, l)
        B^{(+)}(n-i-k, i+k-1, m-2-l) =
    \end{gather*}
    (занесение знака суммы за знак $\res_{z,w}$ и добавление нулевых
    слагаемых)
    \begin{gather*}
        \sum_{i=2}^{n-1}\sum_{k=1}^{n-i}\res_{z,w}\left\{
        \frac{(1+z)^{n-2}(1+w)^{n-i-k}(w-z)}{z^{(m-2)+1}w^{(m-1)+1}}
        \times \left[ \sum_{l=0}^{m}(zw)^l \binom{i-2}{2l} \right]\right\}=
    \end{gather*}
    (суммирование в квадратных скобках и формула геометрической
    прогрессии)
    \begin{gather*}
        \sum_{i=2}^{n-1}\sum_{k=1}^{n-i}\res_{z,w}\left\{
        \frac{(1+z)^{n-2}(1+w)^{n-i-k}(w-z)}{z^{(m-2)+1}w^{(m-1)+1}}
        \times \frac{(1 + \sqrt{zw})^{i-2} + (1 - \sqrt{zw})^{i-2}}{2}
        \right\}
        = \\
        \frac{1}{2}\res_{z,w} \biggl\{
        %
        \frac{(1+z)^{n-2}(1+w)^{n-2}(w-z)}{z^{(m-2)+1}w^{(m-1)+1}} \times \\
        \left[ \sum_{i=2}^{\infty} \left(
        \frac{1+\sqrt{zw}}{1+w}\right)^{i-2} + \sum_{i=2}^{\infty} \left(
        \frac{1-\sqrt{zw}}{1+w}\right)^{i-2} \right] \times
        \sum_{k=1}^{\infty}\left(\frac{1}{1+w}\right)^k
        %
        \biggr\} =
    \end{gather*}
    ($\lvert w\rvert \gg 1$, $\lvert (1 + \sqrt{zw})/(1+w)\rvert < 1$,
    $\lvert (1-\sqrt{zw})/(1+w) \rvert < 1$, суммирование по индексам
    $i$ и $k$, формула геометрической прогрессии)
    \begin{gather*}
        \frac{1}{2}\res_{z,w}\biggl\{
        \frac{(1+z)^{n-2}(1+w)^{n-2}(w-z)}{z^{(m-2)+1}w^{(m-1)+1}} \times \\
        \left[ 1/(1 - \frac{1+ \sqrt{zw}}{1+w}) + 1 / (1 - \frac{1 -
            \sqrt{zw}}{1+w}) \right] \times \frac{1}{1+w} \times \frac{1}{1 -
            \frac{1}{1+w}}
        \biggr\} = \\
        \frac{1}{2}\res_{z,w}\left\{
        \frac{(1+z)^{n-2}(1+w)^{n-1}(w-z)}{z^{(m-2)+1}w^{(m-1)+1}} \times
        \left[ \frac{1}{w-\sqrt{zw}} + \frac{1}{w + \sqrt{zw}} \right]
        \times \frac{1}{w}
        \right\} = \\
        \frac{1}{2}\res_{z,w}\left\{
        \frac{(1+z)^{n-2}(1+w)^{n-1}(w-z)}{z^{(m-2)+1}w^{(m-1)+1}} \times
        \frac{2w}{w^2-zw} \times \frac{1}{w}
        \right\} = \\
        \res_{z,w}\left\{
        \frac{(1+z)^{n-2}(1+w)^{n-1}}{z^{(m-2)+1}w^{m+1}} \right\} =
        \\
        \res_{z}\left\{ \frac{(1+z)^{n-2}}{z^{(m-2)+1}} \right\} \times
        \res_{w}\left\{ \frac{(1+w)^{n-1}}{w^{m+1}} \right\} =
        \binom{n-2}{m-2}\binom{n-1}{m}.
    \end{gather*}
    Тем самым доказательство \crefRod{l:B'-Dn} завершается.
\end{proof}

Основной результат параграфа дает следующая теорема.

% \begin{theorem}[{{\cite[теорема 7]{smj2023}}}]\label{th:Enum-Dn}
\begin{theorem}\label{th:Enum-Dn}
    %<*th:Enum-Dn>
    Число нестандартных идеалов алгебры $RD_n(\mathbb{F}_q)$ $(n \geq 3)$ равно
    \[
        (q-1)\sum_{m=2}^{n-1}\binom{n-2}{m-2}\binom{n-1}{m}
        \sum_{t=1}^{m}\sum_{k=t}^{m}(-1)^{m-k}\binom{m}{k}
        \genfrac{[}{]}{0pt}{}{k}{t}_{\mathrlap{q}}.
    \]
    %</th:Enum-Dn>
\end{theorem}

\begin{proof}
    Тождество $V_q(m,t)=(q-1)^{m-t}S_q[m,t]$ (\S~\ref{s:coordinate-complete}) и
    стандартное рекуррентное соотношение
    $S_q[m,t]=S_q[m-1,t-1]+\frac{q^t-1}{q-1}\,S_q[m-1,t]$ для $q$-чисел Стирлинга
    второго рода после умножения на $(q-1)^{m-t}$ и суммирования по $t$ дают при
    $m\ge2$ тождество
    \[
        \sum_{t=1}^{m} V_q(m,t) = \sum_{t=1}^{m-1} q^t\,V_q(m-1,t).
    \]
    Подставляя это равенство в \eqref{eq:enum-Dn-1} и применяя
    \crefVin{l:B'-Dn,l:vmt-comb}, получаем требуемую формулу.
\end{proof}

\begin{remark}\label{remark:problem-2}
    В отличие от идеалов алгебры $RD_n(\mathbb{K})$, для идеалов $H$ алгебры Ли
    $N\Phi(\mathbb{K})$ классического типа возрастает число пар $p$-связанных углов
    $r,s$ в $H$. Решение проблемы 2 из \cite{sigsam2001} сложнее даже для типа
    $A_n$; с возрастанием ранга $\Phi$ может неограниченно возрастать и число
    углов, $p$-связанных не с одним, а с двумя углами в $H$.
\end{remark}

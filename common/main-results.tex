%% Согласно ГОСТ Р 7.0.11-2011, п. 5.3.3 и 9.2.3.
\begin{enumerate}
    \item Установлено, что стандартная обертывающая алгебры Ли $N\Phi(\mathbb{K})$
          существует для всех типов $\Phi$, кроме $D_n$ $(n\geq 4)$ и $E_n$ $(n=6,7,8)$;
          для типа $F_4$ построены как стандартные, так и нестандартные специальные
          обертывающие алгебры.

    \item Для алгебр Ли $N\Phi(\mathbb{K})$ классических типов исследованы условия
          однозначности специальных обертывающих алгебр $R_{\Phi}$. Специальная
          обертывающая алгебра типа $A_{n-1}$ $(n>3)$ стандартна тогда и только тогда,
          когда она или ее противоположная алгебра по модулю аннулятора изоморфна
          $NT(n,\mathbb{K})/\mathbb{K}e_{n1}$; ассоциативная специальная обертывающая
          единственна и изоморфна $NT(n,\mathbb{K})$, тогда как стандартность сама по
          себе однозначности не дает. Для типа $B_n$ $(n\geq3)$ рассмотрены стандартные
          обертывающие алгебры $R_{\Phi}$ с условием $R_{\Phi}/T(p)^2\simeq
          NT(n+1,\mathbb{K})$, где $p$ "--- короткий простой корень, а $T(p)^2$ "---
          квадрат идеала $T(p)$ в $R_{\Phi}$. Монотонность относительно вложений,
          сохраняющих корневые компоненты при естественных включениях $B_n\subset
          B_{n+1}$, однозначно выделяет среди них алгебру $RB_n(\mathbb{K})$ с точностью
          до перехода к противоположной алгебре.

    \item Доказано, что алгебра $NT(n+1,\mathbb{K})$ "--- единственная с точностью до
          изоморфизма ассоциативная точная обертывающая алгебры Ли $N\Phi(\mathbb{K})$
          типа $A_n$ $(n\geq 3)$, допускающая градуировку, согласованную с градуировкой
          Картана.

    \item Для алгебр Ли $N\Phi(\mathbb{K})$ типов $B_n$, $C_n$ и $D_n$ над произвольным
          полем построены левосимметрические точные обертывающие алгебры в виде
          полупрямых сумм алгебры $NT(n,\mathbb{K})$ и подходящего идеала алгебры Ли
          $N\Phi(\mathbb{K})$.

    \item Решена задача~\textbf{(A)}: найдено число стандартных идеалов алгебры Ли
          $N\Phi(q)$ над конечным полем для всех классических и исключительных типов
          $\Phi$. Перечислены нестандартные идеалы точной обертывающей алгебры $RD_n(q)$.
          Тем самым для каждого классического типа получено перечисление всех идеалов
          одной из специальных обертывающих алгебр. Для перечисления нестандартных
          идеалов алгебры $RD_n(q)$ методом коэффициентов вычислена трехкратная сумма
          биномиальных коэффициентов.
\end{enumerate}

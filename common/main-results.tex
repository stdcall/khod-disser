%% Список основных результатов работы.
%% Подключается в двух местах: Dissertation/conclusion.tex (заключение
%% диссертации) и Synopsis/content.tex (содержание автореферата).
%% Согласно ГОСТ Р 7.0.11-2011, п. 5.3.3 и 9.2.3.
\begin{enumerate}
    \item Установлено, что стандартная обертывающая алгебра алгебры Ли
          $N\Phi(\mathbb{K})$ существует тогда и только тогда, когда тип системы корней
          $\Phi$ отличен от $D_n$ $(n\geq4)$ и $E_6,E_7,E_8$; для типа $F_4$ построены
          как стандартные, так и нестандартные специальные обертывающие алгебры.

    \item Для алгебр Ли $N\Phi(\mathbb{K})$ классических типов исследованы условия
          однозначности специальных обертывающих алгебр $R_{\Phi}$. Специальная
          обертывающая алгебра типа $A_{n-1}$ $(n>3)$ стандартна тогда и только тогда,
          когда по модулю аннулятора изоморфна $NT(n,\mathbb{K})/\mathbb{K}e_{n1}$. Для
          типа $B_n$ $(n\geq3)$ рассмотрены стандартные обертывающие алгебры $R_{\Phi}$ с
          условием $R_{\Phi}/T(p)^2\simeq NT(n+1,\mathbb{K})$, где $p$ "--- короткий
          простой корень, а $T(p)^2$ "--- квадрат идеала $T(p)$ в $R_{\Phi}$. Наличие
          вложений, сохраняющих корневые компоненты при естественных включениях
          $B_n\subset B_{n+1}$, выделяет среди них алгебру $RB_n(\mathbb{K})$ с точностью
          до перехода к противоположной алгебре.

    \item Доказано, что алгебра $NT(n+1,\mathbb{K})$ "--- единственная с точностью до
          изоморфизма ассоциативная точная обертывающая алгебры Ли $N\Phi(\mathbb{K})$
          типа $A_n$ $(n\geq 3)$, допускающая градуировку, согласованную с градуировкой
          Картана.

    \item Построены левосимметрические точные обертывающие алгебры для алгебр Ли
          $N\Phi(\mathbb{K})$ типов $B_n$, $C_n$ и $D_n$ над произвольным полем в виде
          полупрямых сумм алгебры $NT(n,\mathbb{K})$ и подходящего идеала алгебры Ли
          $N\Phi(\mathbb{K})$.

    \item Решена задача~\textbf{(А)}: найдено число стандартных идеалов алгебры Ли
          $N\Phi(\mathbb{F}_q)$ для всех классических и исключительных типов $\Phi$.
          Перечислены нестандартные идеалы точной обертывающей алгебры
          $RD_n(\mathbb{F}_q)$. Тем самым для каждого классического типа получено
          перечисление всех идеалов одной из специальных обертывающих алгебр. Для
          перечисления нестандартных идеалов алгебры $RD_n(\mathbb{F}_q)$ методом
          коэффициентов вычислена трехкратная сумма биномиальных коэффициентов.
\end{enumerate}

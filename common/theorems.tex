% Проверка правильности префиксов лейблов

\RequirePackage{xstring}

% Сохраняем оригинальную команду \label
\let\originallabel\label

% Переменная для хранения типа текущего теоремного окружения
\let\currenttheoremtype\relax

% Флаг: нужно ли пропустить проверку (например, внутри equation)
\newif\ifskiptheoremcheck
\skiptheoremcheckfalse

% Вспомогательная команда для проверки префикса
\newcommand{\checklabelprefix}[2]{%
    % #1 - ожидаемый префикс, #2 - лейбл
    \StrBefore{#2}{:}[\labelprefix]%
    \IfStrEq{\labelprefix}{}{%
        % Нет двоеточия - ошибка
        \errmessage{Label #2 must have prefix. Expected: #1:#2 (see README.md)}%
        \originallabel{#2}%
    }{%
        % Есть префикс - проверяем правильность
        \IfStrEq{\labelprefix}{#1}{%
            % Правильный префикс - создаем лейбл
            \originallabel{#2}%
        }{%
            % Неправильный префикс - ошибка
            \errmessage{Label #2 has wrong prefix \labelprefix:. Expected: #1: (see README.md)}%
            \originallabel{#2}%
        }%
    }%
}

% Переопределяем \label один раз глобально
\renewcommand{\label}[1]{%
    \ifskiptheoremcheck
        % Проверка отключена (внутри equation и т.д.) - просто создаем лейбл
        \originallabel{#1}%
    \else
        \ifx\currenttheoremtype\relax
            % Вне теоремных окружений - просто вызываем оригинальный \label
            \originallabel{#1}%
        \else
            % Внутри теоремного окружения - проверяем префикс
            \checklabelprefix{\currenttheoremtype}{#1}%
        \fi
    \fi
}

% Вспомогательные команды для регистрации хуков
\newcommand{\RegisterTheoremLabelCheck}[2]{%
    \AtBeginEnvironment{#1}{\def\currenttheoremtype{#2}}%
    \AtEndEnvironment{#1}{\let\currenttheoremtype\relax}%
}
\newcommand{\SkipLabelCheckIn}[1]{%
    \BeforeBeginEnvironment{#1}{\skiptheoremchecktrue}%
    \AfterEndEnvironment{#1}{\skiptheoremcheckfalse}%
}

% Устанавливаем тип окружения при входе и сбрасываем при выходе
\RegisterTheoremLabelCheck{theorem}{th}
\RegisterTheoremLabelCheck{lemma}{l}
\RegisterTheoremLabelCheck{proposition}{p}
\RegisterTheoremLabelCheck{corollary}{corollary}
\RegisterTheoremLabelCheck{definition}{def}
\RegisterTheoremLabelCheck{remark}{remark}

% Отключаем проверку при входе в математические и другие окружения с лейблами
\SkipLabelCheckIn{equation}
\SkipLabelCheckIn{equation*}
\SkipLabelCheckIn{align}
\SkipLabelCheckIn{align*}
\SkipLabelCheckIn{gather}
\SkipLabelCheckIn{gather*}
\SkipLabelCheckIn{multline}
\SkipLabelCheckIn{multline*}
\SkipLabelCheckIn{figure}
\SkipLabelCheckIn{table}
